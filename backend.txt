// backend/api/admin/blogs/[id].cjs
const { ObjectId } = require("mongodb");
const sanitizeHtml = require("sanitize-html");
const { connectToDatabase } = require("../../lib/db.cjs");
const { getAdminFromRequest } = require("../../lib/auth.cjs");
const { setCors, handleOptions } = require("../../lib/cors.cjs");
const { slugify, ensureUniqueSlug } = require("../../lib/blogSlug.cjs");
const { deleteImage } = require("../../lib/cloudinary.cjs");

// same sanitize config as index.cjs — extract to lib/sanitize.cjs if you want DRY
const SANITIZE_OPTS = {
  allowedTags: [
    "p", "br", "strong", "em", "u", "s",
    "h1", "h2", "h3", "ul", "ol", "li",
    "blockquote", "code", "pre", "a", "hr", "img",
  ],
  allowedAttributes: {
    a: ["href", "target", "rel"],
    img: ["src", "alt", "width", "height"],
  },
  allowedSchemes: ["http", "https", "mailto", "tel", "data"],
  transformTags: {
    a: sanitizeHtml.simpleTransform("a", {
      target: "_blank",
      rel: "noopener noreferrer",
    }),
  },
};

function cleanHtml(html) {
  if (!html) return "";
  return sanitizeHtml(String(html), SANITIZE_OPTS);
}

function cleanTags(tags) {
  if (!Array.isArray(tags)) return [];
  return [
    ...new Set(
      tags
        .map((t) => String(t || "").trim().toLowerCase())
        .filter((t) => t && t.length <= 30)
    ),
  ].slice(0, 10);
}

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  const decoded = getAdminFromRequest(req);
  if (!decoded) return res.status(401).json({ message: "Unauthorized" });

  const id = req.query?.id || req.params?.id;
  if (!id || !ObjectId.isValid(id)) {
    return res.status(400).json({ message: "Invalid blog ID" });
  }

  const { db } = await connectToDatabase();
  const blogs = db.collection("blogs");
  const _id = new ObjectId(id);

  try {
    const existing = await blogs.findOne({ _id });
    if (!existing) return res.status(404).json({ message: "Blog not found" });

    // ---------- GET ----------
    if (req.method === "GET") {
      return res.status(200).json(existing);
    }

    // ---------- UPDATE ----------
    if (req.method === "PUT" || req.method === "PATCH") {
      const {
        title, excerpt, content, cover, author, tags, status,
      } = req.body || {};

      const update = { updatedAt: new Date() };

      if (title !== undefined) {
        const trimmed = String(title).trim();
        if (!trimmed) {
          return res.status(400).json({ message: "Title cannot be empty" });
        }
        update.title = trimmed;
        // Re-slug only if the title actually changed
        if (trimmed !== existing.title) {
          const base = slugify(trimmed);
          update.slug = await ensureUniqueSlug(db, base, _id);
        }
      }

      if (excerpt !== undefined) {
        update.excerpt = String(excerpt).trim().slice(0, 300);
      }
      if (content !== undefined) {
        update.content = cleanHtml(content);
      }
      if (author !== undefined) {
        update.author = String(author || "").trim().slice(0, 60);
      }
      if (tags !== undefined) {
        update.tags = cleanTags(tags);
      }

      if (cover !== undefined) {
        update.cover = cover
          ? {
              url: String(cover.url || ""),
              publicId: String(cover.publicId || ""),
            }
          : null;

        // Delete old cover from Cloudinary if it changed
        if (
          existing.cover?.publicId &&
          existing.cover.publicId !== update.cover?.publicId
        ) {
          deleteImage(existing.cover.publicId).catch((e) =>
            console.error("Cloudinary cleanup error:", e)
          );
        }
      }

      if (status !== undefined) {
        const isPublished = status === "published";
        update.status = isPublished ? "published" : "draft";
        // Set publishedAt only the first time it goes live
        if (isPublished && !existing.publishedAt) {
          update.publishedAt = new Date();
        }
        // If moving back to draft, keep the old publishedAt (so re-publish keeps history)
      }

      await blogs.updateOne({ _id }, { $set: update });
      const updated = await blogs.findOne({ _id });
      return res.status(200).json(updated);
    }

    // ---------- DELETE ----------
    if (req.method === "DELETE") {
      if (existing.cover?.publicId) {
        deleteImage(existing.cover.publicId).catch((e) =>
          console.error("Cloudinary cleanup error:", e)
        );
      }
      await blogs.deleteOne({ _id });
      return res.status(200).json({ message: "Blog deleted" });
    }

    return res.status(405).json({ message: "Method not allowed" });
  } catch (err) {
    console.error("Admin blog [id] error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};

const { connectToDatabase } = require("../../lib/db.cjs");

const SANITIZE_OPTS = {
  allowedTags: [
    "p", "br", "strong", "em", "u", "s",
    "h1", "h2", "h3", "ul", "ol", "li",
    "blockquote", "code", "pre", "a", "hr", "img",
  ],
  allowedAttributes: {
    a: ["href", "target", "rel"],
    img: ["src", "alt", "width", "height"],
  },
  allowedSchemes: ["http", "https", "mailto", "tel", "data"],
  transformTags: {
    a: sanitizeHtml.simpleTransform("a", {
      target: "_blank",
      rel: "noopener noreferrer",
    }),
  },
};

function cleanHtml(html) {
  if (!html) return "";
  return sanitizeHtml(String(html), SANITIZE_OPTS);
}

function cleanTags(tags) {
  if (!Array.isArray(tags)) return [];
  return [
    ...new Set(
      tags
        .map((t) => String(t || "").trim().toLowerCase())
        .filter((t) => t && t.length <= 30)
    ),
  ].slice(0, 10);
}

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  const decoded = getAdminFromRequest(req);
  if (!decoded) return res.status(401).json({ message: "Unauthorized" });

  try {
    const { db } = await connectToDatabase();
    const blogs = db.collection("blogs");

    // ---------- LIST ----------
    if (req.method === "GET") {
      const { status, search } = req.query || {};
      const filter = {};
      if (status && status !== "all") filter.status = status;
      if (search) {
        const rx = new RegExp(String(search).trim(), "i");
        filter.$or = [{ title: rx }, { excerpt: rx }, { tags: rx }];
      }

      const list = await blogs
        .find(filter, {
          projection: {
            title: 1, slug: 1, excerpt: 1, cover: 1, status: 1,
            tags: 1, author: 1, publishedAt: 1, createdAt: 1, views: 1,
          },
        })
        .sort({ createdAt: -1 })
        .limit(500)
        .toArray();

      return res.status(200).json(list);
    }

    // ---------- CREATE ----------
    if (req.method === "POST") {
      const {
        title, excerpt, content, cover, author, tags, status,
      } = req.body || {};

      if (!title || !String(title).trim()) {
        return res.status(400).json({ message: "Title is required" });
      }
      const trimmedTitle = String(title).trim();
      const baseSlug = slugify(trimmedTitle);
      const slug = await ensureUniqueSlug(db, baseSlug);

      const now = new Date();
      const isPublished = status === "published";

      const doc = {
        title: trimmedTitle,
        slug,
        excerpt: String(excerpt || "").trim().slice(0, 300),
        content: cleanHtml(content),
        cover: cover
          ? {
              url: String(cover.url || ""),
              publicId: String(cover.publicId || ""),
            }
          : null,
        author: String(author || "Halesi Books Team").trim().slice(0, 60),
        tags: cleanTags(tags),
        status: isPublished ? "published" : "draft",
        publishedAt: isPublished ? now : null,
        views: 0,
        createdAt: now,
        updatedAt: now,
      };

      const result = await blogs.insertOne(doc);
      return res.status(201).json({ _id: result.insertedId, ...doc });
    }

    return res.status(405).json({ message: "Method not allowed" });
  } catch (err) {
    console.error("Admin blogs error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};

// backend/api/admin/messages/[id].cjs
const { ObjectId } = require("mongodb");
const { connectToDatabase } = require("../../lib/db.cjs");
const { getAdminFromRequest } = require("../../lib/auth.cjs");
const { setCors, handleOptions } = require("../../lib/cors.cjs");
const { sendReplyToUser } = require("../../lib/mailer.cjs");

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  const decoded = getAdminFromRequest(req);
  if (!decoded) return res.status(401).json({ message: "Unauthorized" });

  const id = req.query?.id || req.params?.id;
  if (!id || !ObjectId.isValid(id)) {
    return res.status(400).json({ message: "Invalid ID" });
  }

  const { db } = await connectToDatabase();
  const messages = db.collection("contact_messages");
  const _id = new ObjectId(id);

  try {
    const msg = await messages.findOne({ _id });
    if (!msg) return res.status(404).json({ message: "Message not found" });

    if (req.method === "GET") {
      // Mark as read when opened (only if not already replied)
      if (msg.status === "new") {
        await messages.updateOne(
          { _id },
          { $set: { status: "read", updatedAt: new Date() } }
        );
        msg.status = "read";
      }
      return res.status(200).json(msg);
    }

    if (req.method === "POST") {
      // Admin reply
      const { reply } = req.body || {};
      if (!reply || String(reply).trim().length < 2) {
        return res.status(400).json({ message: "Reply is required" });
      }

      const now = new Date();
      const entry = {
        from: "admin",
        text: String(reply).trim().slice(0, 4000),
        at: now,
        adminName: decoded.name || decoded.email || "Admin",
      };

      await messages.updateOne(
        { _id },
        {
          $push: { thread: entry },
          $set: { status: "replied", updatedAt: now },
        }
      );

      // Email the user
      sendReplyToUser({
        to: msg.email,
        userName: msg.name,
        subject: msg.subject,
        replyText: entry.text,
        originalMessage: msg.message,
      }).catch((err) => console.error("Reply email failed:", err));

      const updated = await messages.findOne({ _id });
      return res.status(200).json(updated);
    }

    if (req.method === "PATCH") {
      const { status } = req.body || {};
      if (!["new", "read", "replied"].includes(status)) {
        return res.status(400).json({ message: "Invalid status" });
      }
      await messages.updateOne(
        { _id },
        { $set: { status, updatedAt: new Date() } }
      );
      const updated = await messages.findOne({ _id });
      return res.status(200).json(updated);
    }

    if (req.method === "DELETE") {
      await messages.deleteOne({ _id });
      return res.status(200).json({ message: "Deleted" });
    }

    return res.status(405).json({ message: "Method not allowed" });
  } catch (err) {
    console.error("Admin message [id] error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};

// backend/api/admin/offers/[id].cjs
const { ObjectId } = require("mongodb");
const { connectToDatabase } = require("../../lib/db.cjs");
const { getAdminFromRequest } = require("../../lib/auth.cjs");
const { setCors, handleOptions } = require("../../lib/cors.cjs");
const { VALID_CATEGORY_IDS } = require("../../lib/categories.cjs");

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  if (!getAdminFromRequest(req))
    return res.status(401).json({ message: "Unauthorized" });

  const id = req.query?.id || req.params?.id;
  if (!id || !ObjectId.isValid(id))
    return res.status(400).json({ message: "Invalid ID" });

  const { db } = await connectToDatabase();
  const offers = db.collection("offers");
  const _id = new ObjectId(id);

  try {
    const existing = await offers.findOne({ _id });
    if (!existing) return res.status(404).json({ message: "Offer not found" });

    if (req.method === "GET") return res.status(200).json(existing);

    if (req.method === "PATCH" || req.method === "PUT") {
      const {
        name,
        percent,
        startsAt,
        endsAt,
        categories,
        active,
      } = req.body || {};
      const update = { updatedAt: new Date() };

      if (name !== undefined) {
        const t = String(name).trim();
        if (!t) return res.status(400).json({ message: "Name required" });
        update.name = t.slice(0, 80);
      }
      if (percent !== undefined) {
        const p = Number(percent);
        if (!p || p < 1 || p > 90)
          return res.status(400).json({ message: "Percent must be 1–90" });
        update.percent = p;
      }
      if (startsAt !== undefined) {
        const s = new Date(startsAt);
        if (isNaN(s)) return res.status(400).json({ message: "Invalid start" });
        update.startsAt = s;
      }
      if (endsAt !== undefined) {
        const e = new Date(endsAt);
        if (isNaN(e)) return res.status(400).json({ message: "Invalid end" });
        update.endsAt = e;
      }
      if (update.startsAt && update.endsAt && update.endsAt <= update.startsAt) {
        return res
          .status(400)
          .json({ message: "End must be after start" });
      }
      if (categories !== undefined) {
        update.categories = Array.isArray(categories)
          ? categories.filter((c) => VALID_CATEGORY_IDS.includes(c))
          : [];
      }
      if (active !== undefined) update.active = !!active;

      await offers.updateOne({ _id }, { $set: update });
      const updated = await offers.findOne({ _id });
      return res.status(200).json(updated);
    }

    if (req.method === "DELETE") {
      await offers.deleteOne({ _id });
      return res.status(200).json({ message: "Offer deleted" });
    }

    return res.status(405).json({ message: "Method not allowed" });
  } catch (err) {
    console.error("Admin offer [id] error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};

const { connectToDatabase } = require("../../lib/db.cjs");

const { setCors, handleOptions } = require("../../lib/cors.cjs");
const { VALID_CATEGORY_IDS } = require("../../lib/categories.cjs");

function cleanOffer(body) {
  const {
    name,
    percent,
    startsAt,
    endsAt,
    categories,
    active,
  } = body || {};

  if (!name || !String(name).trim()) return { error: "Name is required" };
  const p = Number(percent);
  if (!p || p < 1 || p > 90) return { error: "Percent must be 1–90" };

  const s = new Date(startsAt);
  const e = new Date(endsAt);
  if (isNaN(s) || isNaN(e)) return { error: "Invalid dates" };
  if (e <= s) return { error: "End date must be after start date" };

  const cats = Array.isArray(categories)
    ? categories.filter((c) => VALID_CATEGORY_IDS.includes(c))
    : [];

  return {
    data: {
      name: String(name).trim().slice(0, 80),
      percent: p,
      startsAt: s,
      endsAt: e,
      categories: cats,
      active: active !== false,
    },
  };
}

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  if (!getAdminFromRequest(req))
    return res.status(401).json({ message: "Unauthorized" });

  const { db } = await connectToDatabase();
  const offers = db.collection("offers");

  try {
    if (req.method === "GET") {
      const list = await offers.find({}).sort({ createdAt: -1 }).toArray();
      return res.status(200).json(list);
    }

    if (req.method === "POST") {
      const { data, error } = cleanOffer(req.body);
      if (error) return res.status(400).json({ message: error });

      const now = new Date();
      const doc = {
        ...data,
        createdAt: now,
        updatedAt: now,
      };
      const result = await offers.insertOne(doc);
      return res.status(201).json({ _id: result.insertedId, ...doc });
    }

    return res.status(405).json({ message: "Method not allowed" });
  } catch (err) {
    console.error("Admin offers error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};












// backend/api/admin/orders/[id].cjs
const { ObjectId } = require("mongodb");
const { connectToDatabase } = require("../../lib/db.cjs");
const { getAdminFromRequest } = require("../../lib/auth.cjs");
const { setCors, handleOptions } = require("../../lib/cors.cjs");

const VALID_STATUSES = ["pending", "confirmed", "shipped", "delivered", "cancelled"];
const VALID_PAYMENT = [
  "unpaid",
  "cod-pending",
  "pending-verification",
  "verified",
  "rejected",
  "cod-paid",
];

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  const decoded = getAdminFromRequest(req);
  if (!decoded) return res.status(401).json({ message: "Unauthorized" });

  const id = req.query?.id || req.params?.id;
  if (!id || !ObjectId.isValid(id)) {
    return res.status(400).json({ message: "Invalid order ID" });
  }

  const { db } = await connectToDatabase();
  const orders = db.collection("orders");
  const books = db.collection("books");
  const _id = new ObjectId(id);

  try {
    const order = await orders.findOne({ _id });
    if (!order) return res.status(404).json({ message: "Order not found" });

    if (req.method === "GET") {
      return res.status(200).json(order);
    }

    if (req.method === "PUT" || req.method === "PATCH") {
      const { status, paymentStatus, note } = req.body || {};
      const update = { updatedAt: new Date() };
      const historyPush = [];

      // ---------- STATUS CHANGE ----------
      if (status !== undefined) {
        if (!VALID_STATUSES.includes(status)) {
          return res.status(400).json({ message: "Invalid status" });
        }

        // 💡 If cancelling, restore stock (only if it was NOT already cancelled)
        if (status === "cancelled" && order.status !== "cancelled") {
          for (const item of order.items || []) {
            if (!ObjectId.isValid(item.bookId)) continue;
            await books.updateOne(
              { _id: new ObjectId(item.bookId) },
              {
                $inc: { stock: item.quantity },
                $set: { updatedAt: new Date() },
              }
            );
          }
        }

        // 💡 If un-cancelling (moving out of cancelled), decrement stock again
        if (status !== "cancelled" && order.status === "cancelled") {
          for (const item of order.items || []) {
            if (!ObjectId.isValid(item.bookId)) continue;
            await books.updateOne(
              { _id: new ObjectId(item.bookId) },
              {
                $inc: { stock: -item.quantity },
                $set: { updatedAt: new Date() },
              }
            );
          }
        }

        update.status = status;

        // 💡 When delivered, if payment was COD, mark as paid
        if (status === "delivered" && order.paymentMethod === "cod") {
          update.paymentStatus = "cod-paid";
        }

        historyPush.push({
          status,
          at: new Date(),
          note: note || "Status updated by admin",
        });
      }

      // ---------- PAYMENT STATUS CHANGE ----------
      if (paymentStatus !== undefined) {
        if (!VALID_PAYMENT.includes(paymentStatus)) {
          return res.status(400).json({ message: "Invalid payment status" });
        }
        update.paymentStatus = paymentStatus;
        historyPush.push({
          status: `payment-${paymentStatus}`,
          at: new Date(),
          note: note || `Payment marked ${paymentStatus}`,
        });
      }

      const updateOp = { $set: update };
      if (historyPush.length > 0) {
        updateOp.$push = { statusHistory: { $each: historyPush } };
      }

      await orders.updateOne({ _id }, updateOp);

      const updated = await orders.findOne({ _id });
      return res.status(200).json(updated);
    }

    return res.status(405).json({ message: "Method not allowed" });
  } catch (err) {
    console.error("Admin order update error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};

const { getAdminFromRequest } = require("../../lib/auth.cjs");

  if (!decoded) return res.status(401).json({ message: "Unauthorized" });

  if (req.method !== "GET") {
    return res.status(405).json({ message: "Method not allowed" });
  }

  try {
    const { db } = await connectToDatabase();
    const { search } = req.query || {};

    const filter = {};
    if (search) {
      const rx = new RegExp(String(search).trim(), "i");
      filter.$or = [{ name: rx }, { email: rx }, { phone: rx }];
    }

    const users = await db
      .collection("users")
      .find(filter, {
        projection: {
          name: 1, email: 1, phone: 1, avatar: 1, createdAt: 1,
          wishlist: 1, addresses: 1, blocked: 1,
        },
      })
      .sort({ createdAt: -1 })
      .limit(500)
      .toArray();

    // For each user, count orders + total spent
    const orders = await db
      .collection("orders")
      .find({})
      .project({ userId: 1, total: 1, status: 1 })
      .toArray();

    const orderStats = {};
    orders.forEach((o) => {
      if (!o.userId) return;
      if (!orderStats[o.userId]) {
        orderStats[o.userId] = { count: 0, spent: 0 };
      }
      orderStats[o.userId].count += 1;
      if (o.status !== "cancelled") {
        orderStats[o.userId].spent += Number(o.total) || 0;
      }
    });

    const enriched = users.map((u) => ({
      _id: u._id,
      name: u.name,
      email: u.email,
      phone: u.phone || "",
      avatar: u.avatar || null,
      createdAt: u.createdAt,
      blocked: !!u.blocked,
      wishlistCount: (u.wishlist || []).length,
      addressesCount: (u.addresses || []).length,
      orderCount: orderStats[u._id.toString()]?.count || 0,
      totalSpent: orderStats[u._id.toString()]?.spent || 0,
    }));

    return res.status(200).json(enriched);
  } catch (err) {
    console.error("Admin users error:", err);

// backend/api/admin/settings.cjs
const { getAdminFromRequest } = require("../lib/auth.cjs");
const { setCors, handleOptions } = require("../lib/cors.cjs");
const { getSettings, updateSettings } = require("../lib/settings.cjs");

function sanitizeZones(zones) {
  if (!Array.isArray(zones)) return null;
  return zones
    .slice(0, 30) // max 30 zones
    .map((z, i) => ({
      id:
        String(z.id || "")
          .trim()
          .toLowerCase()
          .replace(/[^a-z0-9-]/g, "-")
          .replace(/-+/g, "-")
          .replace(/^-|-$/g, "") || `zone-${i + 1}`,
      name: String(z.name || "").trim().slice(0, 60),
      description: String(z.description || "").trim().slice(0, 160),
      charge: Math.max(0, Number(z.charge) || 0),
    }))
    .filter((z) => z.name);
}

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  const decoded = getAdminFromRequest(req);
  if (!decoded) return res.status(401).json({ message: "Unauthorized" });

  try {
    if (req.method === "GET") {
      const settings = await getSettings();
      return res.status(200).json(settings);
    }

    if (req.method === "PUT") {
      const { deliveryZones, deliveryCharge, paymentQR, paymentMethods } =
        req.body || {};

      const patch = {};

      if (deliveryZones !== undefined) {
        const clean = sanitizeZones(deliveryZones);
        if (!clean || clean.length === 0) {
          return res
            .status(400)
            .json({ message: "At least one delivery zone is required" });
        }
        // Ensure unique ids
        const seen = new Set();
        clean.forEach((z, i) => {
          let base = z.id || `zone-${i + 1}`;
          let id = base;
          let n = 1;
          while (seen.has(id)) {
            id = `${base}-${n++}`;
          }
          z.id = id;
          seen.add(id);
        });
        patch.deliveryZones = clean;
      }

      if (deliveryCharge !== undefined) {
        patch.deliveryCharge = Math.max(0, Number(deliveryCharge) || 0);
      }

      if (paymentQR !== undefined) {
        patch.paymentQR = paymentQR
          ? {
              url: String(paymentQR.url || ""),
              publicId: String(paymentQR.publicId || ""),
            }
          : null;
      }

      if (paymentMethods !== undefined && typeof paymentMethods === "object") {
        patch.paymentMethods = {
          qr: !!paymentMethods.qr,
          cod: !!paymentMethods.cod,
        };
      }

      const updated = await updateSettings(patch);
      return res.status(200).json(updated);
    }

    return res.status(405).json({ message: "Method not allowed" });
  } catch (err) {
    console.error("Admin settings error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};











// backend/api/admin/stats.cjs
const { connectToDatabase } = require("../lib/db.cjs");
const { getAdminFromRequest } = require("../lib/auth.cjs");
const { setCors, handleOptions } = require("../lib/cors.cjs");

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  const decoded = getAdminFromRequest(req);
  if (!decoded) return res.status(401).json({ message: "Unauthorized" });

  try {
    const { db } = await connectToDatabase();

    const [
      booksCount,
      usersCount,
      orders,
      reviewsCount,
      blogsCount,
      announcementsCount,
      materialsCount,
      backEditionsCount,
    ] = await Promise.all([
      db.collection("books").countDocuments({ isBackEdition: { $ne: true } }),
      db.collection("users").countDocuments(),
      db.collection("orders").find({}).toArray(),
      db.collection("reviews").countDocuments(),
      db.collection("blogs").countDocuments().catch(() => 0),
      db.collection("announcements").countDocuments().catch(() => 0),
      db.collection("materials").countDocuments().catch(() => 0),
      db.collection("books").countDocuments({ isBackEdition: true }),
    ]);

    const totalOrders = orders.length;
    const pendingOrders = orders.filter((o) => o.status === "pending").length;
    const confirmedOrders = orders.filter((o) => o.status === "confirmed").length;
    const shippedOrders = orders.filter((o) => o.status === "shipped").length;
    const deliveredOrders = orders.filter((o) => o.status === "delivered").length;
    const cancelledOrders = orders.filter((o) => o.status === "cancelled").length;

    // 💡 Revenue = orders NOT cancelled and NOT delivered (in the pipeline)
    const totalRevenue = orders
      .filter((o) => o.status !== "cancelled" && o.status !== "delivered")
      .reduce((sum, o) => sum + (Number(o.total) || 0), 0);

    // 💰 Earning = only delivered orders (money actually earned)
    const totalEarning = orders
      .filter((o) => o.status === "delivered")
      .reduce((sum, o) => sum + (Number(o.total) || 0), 0);

    // 📈 Total sales of all non-cancelled orders (delivered + in pipeline)
    const grossSales = orders
      .filter((o) => o.status !== "cancelled")
      .reduce((sum, o) => sum + (Number(o.total) || 0), 0);

    const paymentPending = orders.filter(
      (o) => o.paymentStatus === "pending-verification"
    ).length;

    // Today's stats
    const now = new Date();
    const todayStart = new Date(now);
    todayStart.setHours(0, 0, 0, 0);
    const todayEnd = new Date(todayStart);
    todayEnd.setDate(todayEnd.getDate() + 1);

    const todaysOrders = orders.filter(
      (o) =>
        new Date(o.createdAt) >= todayStart &&
        new Date(o.createdAt) < todayEnd &&
        o.status !== "cancelled"
    );
    const todayRevenue = todaysOrders.reduce(
      (s, o) => s + (Number(o.total) || 0),
      0
    );
    const todayOrdersCount = todaysOrders.length;

    // 7-day chart
    const dailyRevenue = [];
    for (let i = 6; i >= 0; i--) {
      const dayStart = new Date(now);
      dayStart.setHours(0, 0, 0, 0);
      dayStart.setDate(dayStart.getDate() - i);
      const dayEnd = new Date(dayStart);
      dayEnd.setDate(dayEnd.getDate() + 1);

      const dayOrders = orders.filter(
        (o) =>
          o.status !== "cancelled" &&
          new Date(o.createdAt) >= dayStart &&
          new Date(o.createdAt) < dayEnd
      );
      const dayTotal = dayOrders.reduce(
        (sum, o) => sum + (Number(o.total) || 0),
        0
      );

      dailyRevenue.push({
        date: dayStart.toISOString().slice(0, 10),
        label: dayStart.toLocaleDateString("en-GB", {
          day: "2-digit",
          month: "short",
        }),
        total: dayTotal,
        orders: dayOrders.length,
      });
    }

    // Low stock
    const lowStock = await db
      .collection("books")
      .find({ stock: { $lte: 5 }, isBackEdition: { $ne: true } })
      .project({ title: 1, stock: 1, author: 1, images: 1 })
      .limit(10)
      .toArray();

    // 💡 Top selling books (by quantity across non-cancelled orders)
    const salesMap = new Map();
    for (const o of orders) {
      if (o.status === "cancelled") continue;
      for (const item of o.items || []) {
        const key = item.bookId || item.title;
        if (!salesMap.has(key)) {
          salesMap.set(key, {
            bookId: item.bookId,
            title: item.title,
            author: item.author,
            image: item.image,
            quantity: 0,
            revenue: 0,
          });
        }
        const agg = salesMap.get(key);
        agg.quantity += Number(item.quantity) || 0;
        agg.revenue += (Number(item.price) || 0) * (Number(item.quantity) || 0);
      }
    }
    const topBooks = Array.from(salesMap.values())
      .sort((a, b) => b.quantity - a.quantity)
      .slice(0, 5);

    // 💡 Top customers by total spent
    const userSpend = new Map();
    for (const o of orders) {
      if (o.status === "cancelled") continue;
      if (!o.userId) continue;
      const prev = userSpend.get(o.userId) || { spent: 0, orders: 0 };
      userSpend.set(o.userId, {
        spent: prev.spent + (Number(o.total) || 0),
        orders: prev.orders + 1,
      });
    }
    const topUserIds = Array.from(userSpend.entries())
      .sort((a, b) => b[1].spent - a[1].spent)
      .slice(0, 5);

    let topCustomers = [];
    if (topUserIds.length > 0) {
      const { ObjectId } = require("mongodb");
      const validIds = topUserIds
        .map(([id]) => (ObjectId.isValid(id) ? new ObjectId(id) : null))
        .filter(Boolean);
      const users = await db
        .collection("users")
        .find(
          { _id: { $in: validIds } },
          { projection: { name: 1, email: 1, avatar: 1 } }
        )
        .toArray();
      const byId = new Map(users.map((u) => [u._id.toString(), u]));
      topCustomers = topUserIds.map(([id, stats]) => ({
        _id: id,
        name: byId.get(id)?.name || "Unknown",
        email: byId.get(id)?.email || "",
        avatar: byId.get(id)?.avatar || null,
        spent: stats.spent,
        orderCount: stats.orders,
      }));
    }

    // 💡 Stock value (cost of inventory at current prices)
    const allBooks = await db
      .collection("books")
      .find({}, { projection: { price: 1, stock: 1, isBackEdition: 1 } })
      .toArray();
    const inventoryValue = allBooks.reduce(
      (sum, b) => sum + (Number(b.price) || 0) * (Number(b.stock) || 0),
      0
    );
    const totalStock = allBooks.reduce(
      (sum, b) => sum + (Number(b.stock) || 0),
      0
    );
    const outOfStock = allBooks.filter((b) => b.stock === 0).length;

    // 💡 Category breakdown
    const categoryMap = new Map();
    for (const b of allBooks) {
      const c = b.category || "General";
      categoryMap.set(c, (categoryMap.get(c) || 0) + 1);
    }
    const categories = Array.from(categoryMap.entries())
      .map(([name, count]) => ({ name, count }))
      .sort((a, b) => b.count - a.count);

    const recentOrders = await db
      .collection("orders")
      .find({})
      .sort({ createdAt: -1 })
      .limit(5)
      .project({
        orderNumber: 1,
        total: 1,
        status: 1,
        createdAt: 1,
        shippingAddress: 1,
        items: 1,
        paymentStatus: 1,
      })
      .toArray();

    return res.status(200).json({
      counts: {
        books: booksCount,
        users: usersCount,
        reviews: reviewsCount,
        blogs: blogsCount,
        announcements: announcementsCount,
        materials: materialsCount,
        backEditions: backEditionsCount,
      },
      orders: {
        total: totalOrders,
        pending: pendingOrders,
        confirmed: confirmedOrders,
        shipped: shippedOrders,
        delivered: deliveredOrders,
        cancelled: cancelledOrders,
      },
      revenue: {
        total: totalRevenue,
        gross: grossSales,
        last7Days: dailyRevenue,
        today: todayRevenue,
        todayOrders: todayOrdersCount,
      },
      earning: {
        total: totalEarning,
      },
      payment: { pendingVerification: paymentPending },
      inventory: {
        totalStock,
        value: inventoryValue,
        outOfStock,
      },
      topBooks,
      topCustomers,
      categories,
      lowStock,
      recentOrders,
    });
  } catch (err) {
    console.error("Stats error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};

const bcrypt = require("bcryptjs");
const { connectToDatabase } = require("../lib/db.cjs");
const { signToken } = require("../lib/auth.cjs");
const { setCors, handleOptions } = require("../lib/cors.cjs");

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  if (req.method !== "POST") {
    return res.status(405).json({ message: "Method not allowed" });
  }

  try {
    const { email, password } = req.body || {};
    if (!email || !password) {
      return res
        .status(400)
        .json({ message: "Email and password are required" });
    }

    const { db } = await connectToDatabase();
    const admins = db.collection("admins");

    let admin = await admins.findOne({ email: email.toLowerCase() });

    // Bootstrap first admin
    if (!admin) {
      const count = await admins.countDocuments();
      if (count === 0 && email.toLowerCase() === process.env.ADMIN_EMAIL) {
        const hashed = await bcrypt.hash(process.env.ADMIN_PASSWORD, 10);
        const doc = {
          name: "Super Admin",
          email: process.env.ADMIN_EMAIL,
          password: hashed,
          role: "admin",
          createdAt: new Date(),
        };
        const result = await admins.insertOne(doc);
        admin = { ...doc, _id: result.insertedId };
      }
    }

    if (!admin) {
      return res.status(401).json({ message: "Invalid credentials" });
    }

    const valid = await bcrypt.compare(password, admin.password);
    if (!valid) {
      return res.status(401).json({ message: "Invalid credentials" });
    }

    const token = signToken({
      id: admin._id.toString(),
      email: admin.email,
      role: admin.role,
    });

    return res.status(200).json({
      token,
      admin: {
        id: admin._id.toString(),
        name: admin.name,
        email: admin.email,
        role: admin.role,
      },
    });
  } catch (err) {
    console.error("Login error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};

// backend/api/auth/register.cjs
const bcrypt = require("bcryptjs");
const { connectToDatabase } = require("../lib/db.cjs");
const { signToken } = require("../lib/auth.cjs");
const { setCors, handleOptions } = require("../lib/cors.cjs");

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  if (req.method !== "POST") {
    return res.status(405).json({ message: "Method not allowed" });
  }

  try {
    const { name, email, password, phone } = req.body || {};

    if (!name || !email || !password) {
      return res
        .status(400)
        .json({ message: "Name, email and password are required" });
    }
    if (String(password).length < 6) {
      return res
        .status(400)
        .json({ message: "Password must be at least 6 characters" });
    }

    const { db } = await connectToDatabase();
    const users = db.collection("users");

    const existing = await users.findOne({ email: email.toLowerCase() });
    if (existing) {
      return res.status(409).json({ message: "Email already registered" });
    }

    const hashed = await bcrypt.hash(password, 10);
    const doc = {
      name: String(name).trim(),
      email: String(email).toLowerCase().trim(),
      phone: phone ? String(phone).trim() : "",
      password: hashed,
      role: "user",
      wishlist: [],
      cart: [],

// backend/api/auth/signup-request.cjs
const bcrypt = require("bcryptjs");
const { connectToDatabase } = require("../lib/db.cjs");
const { setCors, handleOptions } = require("../lib/cors.cjs");
const { sendOtpEmail } = require("../lib/mailer.cjs");
const { issueOtp } = require("../lib/otp.cjs");

const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  if (req.method !== "POST") {
    return res.status(405).json({ message: "Method not allowed" });
  }

  try {
    const { name, email, password, phone } = req.body || {};

    if (!name || !email || !password) {
      return res
        .status(400)
        .json({ message: "Name, email and password are required" });
    }
    if (String(password).length < 6) {
      return res
        .status(400)
        .json({ message: "Password must be at least 6 characters" });
    }
    const cleanEmail = String(email).toLowerCase().trim();
    if (!EMAIL_RE.test(cleanEmail)) {
      return res.status(400).json({ message: "Invalid email address" });
    }

    const { db } = await connectToDatabase();
    const users = db.collection("users");

    const existing = await users.findOne({ email: cleanEmail });
    if (existing) {
      return res.status(409).json({ message: "Email already registered" });
    }

    const passwordHash = await bcrypt.hash(password, 10);

    const pendingUser = {
      name: String(name).trim(),
      email: cleanEmail,
      phone: phone ? String(phone).trim() : "",
      passwordHash,
      role: "user",
    };

    const { code, expiresAt, ttlMin } = await issueOtp({
      email: cleanEmail,
      pendingUser,
    });

    // Send the OTP email
    try {
      await sendOtpEmail({ to: cleanEmail, name: pendingUser.name, code });
    } catch (mailErr) {
      console.error("OTP email send failed:", mailErr);
      return res.status(500).json({
        message:
          "Failed to send verification email. Please check your email address or try again shortly.",
      });
    }

    return res.status(200).json({
      message: "Verification code sent to your email",
      email: cleanEmail,
      expiresAt,
      ttlMinutes: ttlMin,
    });
  } catch (err) {
    console.error("signup-request error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};















// backend/api/auth/signup-verify.cjs
const { connectToDatabase } = require("../lib/db.cjs");
const { signToken } = require("../lib/auth.cjs");
const { setCors, handleOptions } = require("../lib/cors.cjs");
const { verifyOtp, consumeOtp } = require("../lib/otp.cjs");

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  if (req.method !== "POST") {
    return res.status(405).json({ message: "Method not allowed" });
  }

  try {
    const { email, code } = req.body || {};
    if (!email || !code) {
      return res.status(400).json({ message: "Email and code are required" });
    }

    const result = await verifyOtp({ email, code });
    if (!result.ok) {
      const map = {
        "no-code": "No verification code found. Please request a new one.",
        expired: "Verification code has expired. Please request a new one.",
        "too-many-attempts":
          "Too many incorrect attempts. Please request a new code.",
        "wrong-code": "Incorrect code. Please try again.",
      };
      return res
        .status(400)
        .json({ message: map[result.reason] || "Invalid code" });
    }

    const { db } = await connectToDatabase();
    const users = db.collection("users");

    const pu = result.pendingUser;
    if (!pu || !pu.email) {
      return res.status(400).json({ message: "Invalid pending signup" });
    }

    // Email might have been claimed in the meantime
    const existing = await users.findOne({ email: pu.email });
    if (existing) {
      await consumeOtp(result.otpId);
      return res.status(409).json({ message: "Email already registered" });
    }

    const doc = {
      name: pu.name,
      email: pu.email,
      phone: pu.phone || "",
      password: pu.passwordHash,
      role: "user",
      emailVerified: true,
      wishlist: [],
      cart: [],
      createdAt: new Date(),
    };

    const insert = await users.insertOne(doc);
    await consumeOtp(result.otpId);

    const token = signToken({
      id: insert.insertedId.toString(),
      email: doc.email,
      role: "user",
    });

    return res.status(201).json({
      token,
      user: {
        id: insert.insertedId.toString(),
        name: doc.name,
        email: doc.email,
        phone: doc.phone,
        role: "user",
      },
    });
  } catch (err) {
    console.error("signup-verify error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};

// backend/api/books/[id].cjs — ADMIN
const { ObjectId } = require("mongodb");
const { connectToDatabase } = require("../lib/db.cjs");
const { getAdminFromRequest } = require("../lib/auth.cjs");
const { setCors, handleOptions } = require("../lib/cors.cjs");
const { deleteImages } = require("../lib/cloudinary.cjs");

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  const decoded = getAdminFromRequest(req);
  if (!decoded) return res.status(401).json({ message: "Unauthorized" });

  const id = req.query?.id || req.params?.id;
  if (!id || !ObjectId.isValid(id)) {
    return res.status(400).json({ message: "Invalid book ID" });
  }

  const { db } = await connectToDatabase();
  const books = db.collection("books");
  const _id = new ObjectId(id);

  try {
    if (req.method === "GET") {
      const book = await books.findOne({ _id });
      if (!book) return res.status(404).json({ message: "Book not found" });
      return res.status(200).json(book);
    }

    if (req.method === "PUT" || req.method === "PATCH") {
      const {
        title,
        author,
        isbn,
        price,
        stock,
        category,
        images,
        discountPercent,
        discountEndsAt,
        isBackEdition,
        backEditionNote,
      } = req.body || {};

      const existing = await books.findOne({ _id });
      if (!existing) return res.status(404).json({ message: "Book not found" });

      let removedPublicIds = [];
      if (Array.isArray(images)) {
        const incomingIds = images.map((i) => i.publicId).filter(Boolean);
        removedPublicIds = (existing.images || [])
          .map((i) => i.publicId)
          .filter((pid) => pid && !incomingIds.includes(pid));
      }

      const update = {
        ...(title !== undefined && { title: String(title).trim() }),
        ...(author !== undefined && { author: String(author).trim() }),
        ...(isbn !== undefined && { isbn: String(isbn).trim() }),
        ...(price !== undefined && { price: Number(price) || 0 }),
        ...(stock !== undefined && { stock: Number(stock) || 0 }),
        ...(category !== undefined && { category }),
        ...(images !== undefined && {
          images: Array.isArray(images)
            ? images.slice(0, 3).map((img) => ({
                url: String(img.url || ""),
                publicId: String(img.publicId || ""),
              }))
            : [],
        }),

        // 💡 per-book discount
        ...(discountPercent !== undefined && {
          discountPercent: Math.max(
            0,
            Math.min(90, Number(discountPercent) || 0)
          ),
        }),
        ...(discountEndsAt !== undefined && {
          discountEndsAt: discountEndsAt
            ? new Date(discountEndsAt)
            : null,
        }),

        // 💡 Back Edition / clearance
        ...(isBackEdition !== undefined && {
          isBackEdition: !!isBackEdition,
        }),
        ...(backEditionNote !== undefined && {
          backEditionNote: String(backEditionNote || "").trim().slice(0, 200),
        }),

        updatedAt: new Date(),
      };

      const result = await books.updateOne({ _id }, { $set: update });
      if (result.matchedCount === 0) {
        return res.status(404).json({ message: "Book not found" });
      }

      if (removedPublicIds.length > 0) {
        deleteImages(removedPublicIds).catch((e) =>
          console.error("Cloudinary cleanup error:", e)
        );
      }

      const updated = await books.findOne({ _id });
      return res.status(200).json(updated);
    }

    if (req.method === "DELETE") {
      const book = await books.findOne({ _id });
      if (!book) return res.status(404).json({ message: "Book not found" });

      const publicIds = (book.images || []).map((i) => i.publicId);
      if (publicIds.length > 0) {
        deleteImages(publicIds).catch((e) =>
          console.error("Cloudinary cleanup error:", e)
        );
      }

      await books.deleteOne({ _id });
      return res.status(200).json({ message: "Book deleted" });
    }

    return res.status(405).json({ message: "Method not allowed" });
  } catch (err) {
    console.error("Book [id] error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};

// backend/api/cart/index.cjs
const { ObjectId } = require("mongodb");
const { connectToDatabase } = require("../lib/db.cjs");
const { getUserFromRequest } = require("../lib/auth.cjs");
const { setCors, handleOptions } = require("../lib/cors.cjs");
const { withActiveOffers } = require("../lib/discount.cjs");

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  const decoded = getUserFromRequest(req);
  if (!decoded) return res.status(401).json({ message: "Unauthorized" });

  const { db } = await connectToDatabase();
  const carts = db.collection("carts");

  try {
    if (req.method === "GET") {
      const cart = await carts.findOne({ userId: decoded.id });
      const items = cart?.items || [];

      if (items.length === 0) {
        return res.status(200).json({ items: [], total: 0, count: 0 });
      }

      const ids = items
        .filter((i) => ObjectId.isValid(i.bookId))
        .map((i) => new ObjectId(i.bookId));

      const books = await db
        .collection("books")
        .find(
          { _id: { $in: ids } },
          {
            projection: {
              title: 1,
              author: 1,
              price: 1,
              images: 1,
              stock: 1,
              category: 1,
              // 💡 needed by the discount helper
              discountPercent: 1,
              discountEndsAt: 1,
            },
          }
        )
        .toArray();

      // 💡 Apply discount
      const priced = await withActiveOffers(db, books);
      const byId = new Map(priced.map((b) => [b._id.toString(), b]));

      const enriched = items
        .map((i) => {
          const b = byId.get(i.bookId);
          if (!b) return null;
          return {
            bookId: i.bookId,
            quantity: i.quantity,
            title: b.title,
            author: b.author,
            price: b.finalPrice,
            originalPrice: b.originalPrice,
            discountPercent: b.discountPercent || 0,
            image: b.images?.[0]?.url || null,
            stock: b.stock,
          };
        })
        .filter(Boolean);

      const total = enriched.reduce(
        (sum, i) => sum + i.price * i.quantity,
        0
      );
      const count = enriched.reduce((sum, i) => sum + i.quantity, 0);

      return res.status(200).json({ items: enriched, total, count });
    }

    if (req.method === "POST") {
      const { bookId, quantity = 1 } = req.body || {};
      if (!bookId || !ObjectId.isValid(bookId)) {
        return res.status(400).json({ message: "Invalid book ID" });
      }
      const qty = Math.max(1, Number(quantity) || 1);

      const cart = await carts.findOne({ userId: decoded.id });
      const items = cart?.items || [];

      const existingIdx = items.findIndex((i) => i.bookId === bookId);
      if (existingIdx >= 0) {
        items[existingIdx].quantity += qty;
      } else {
        items.push({ bookId, quantity: qty });
      }

      if (cart) {
        await carts.updateOne(
          { userId: decoded.id },
          { $set: { items, updatedAt: new Date() } }
        );
      } else {
        await carts.insertOne({
          userId: decoded.id,
          items,
          createdAt: new Date(),
          updatedAt: new Date(),
        });
      }

      return res.status(200).json({ message: "Added to cart", items });
    }

    if (req.method === "PUT") {
      const { bookId, quantity } = req.body || {};
      if (!bookId || !ObjectId.isValid(bookId)) {
        return res.status(400).json({ message: "Invalid book ID" });
      }
      const qty = Number(quantity);
      if (!qty || qty < 1) {
        return res.status(400).json({ message: "Invalid quantity" });
      }

      await carts.updateOne(
        { userId: decoded.id, "items.bookId": bookId },
        {
          $set: {
            "items.$.quantity": qty,
            updatedAt: new Date(),
          },
        }
      );

      return res.status(200).json({ message: "Cart updated" });
    }

    if (req.method === "DELETE") {
      const { bookId } = req.body || {};
      if (!bookId || !ObjectId.isValid(bookId)) {
        return res.status(400).json({ message: "Invalid book ID" });
      }

      await carts.updateOne(
        { userId: decoded.id },
        {
          $pull: { items: { bookId } },
          $set: { updatedAt: new Date() },
        }
      );

      return res.status(200).json({ message: "Removed from cart" });
    }

    return res.status(405).json({ message: "Method not allowed" });
  } catch (err) {
    console.error("Cart error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};

const jwt = require("jsonwebtoken");

function signToken(payload) {
  return jwt.sign(payload, process.env.JWT_SECRET, { expiresIn: "30d" });
}

function verifyToken(token) {
  try {
    return jwt.verify(token, process.env.JWT_SECRET);
  } catch (err) {
    return null;
  }
}

// ---------- ADMIN ----------
function getAdminFromRequest(req) {
  const auth = req.headers.authorization || "";
  const token = auth.startsWith("Bearer ") ? auth.slice(7) : null;
  if (!token) return null;
  const decoded = verifyToken(token);
  if (!decoded || decoded.role !== "admin") return null;
  return decoded;
}

// ---------- USER (customer) ----------
function getUserFromRequest(req) {
  const auth = req.headers.authorization || "";
  const token = auth.startsWith("Bearer ") ? auth.slice(7) : null;
  if (!token) return null;
  const decoded = verifyToken(token);
  if (!decoded || decoded.role !== "user") return null;
  return decoded;
}

// ---------- EITHER ----------
function getAnyAuthFromRequest(req) {
  const auth = req.headers.authorization || "";
  const token = auth.startsWith("Bearer ") ? auth.slice(7) : null;
  if (!token) return null;
  return verifyToken(token);
}

// backend/api/lib/blogSlug.cjs
function slugify(input) {
  return String(input || "")
    .toLowerCase()
    .normalize("NFKD")
    .replace(/[\u0300-\u036f]/g, "")     // strip diacritics
    .replace(/[^a-z0-9\s-]/g, "")        // keep letters/digits/space/dash
    .trim()
    .replace(/\s+/g, "-")
    .replace(/-+/g, "-")
    .replace(/^-|-$/g, "")
    .slice(0, 80);
}

async function ensureUniqueSlug(db, baseSlug, excludeId = null) {
  const blogs = db.collection("blogs");
  let slug = baseSlug || `post-${Date.now()}`;
  let n = 1;
  // Collision-safe loop
  while (
    await blogs.findOne({
      slug,
      ...(excludeId ? { _id: { $ne: excludeId } } : {}),
    })
  ) {
    slug = `${baseSlug}-${n++}`;
  }
  return slug;
}

module.exports = { slugify, ensureUniqueSlug };

// backend/api/lib/cloudinary.cjs
const crypto = require("crypto");

/**
 * Delete a Cloudinary asset.
 * @param {string} publicId
 * @param {"image"|"raw"|"video"} resourceType - default "image"
 */
async function deleteImage(publicId, resourceType = "image") {
  const cloudName = process.env.CLOUDINARY_CLOUD_NAME;
  const apiKey = process.env.CLOUDINARY_API_KEY;
  const apiSecret = process.env.CLOUDINARY_API_SECRET;

  if (!cloudName || !apiKey || !apiSecret || !publicId) return;

  const timestamp = Math.round(Date.now() / 1000);
  const stringToSign = `public_id=${publicId}&timestamp=${timestamp}`;
  const signature = crypto
    .createHash("sha1")
    .update(stringToSign + apiSecret)
    .digest("hex");

  const body = new URLSearchParams({
    public_id: publicId,
    api_key: apiKey,
    timestamp: String(timestamp),
    signature,
  });

  const resp = await fetch(
    `https://api.cloudinary.com/v1_1/${cloudName}/${resourceType}/destroy`,
    {
      method: "POST",
      headers: { "Content-Type": "application/x-www-form-urlencoded" },
      body,
    }
  );

  if (!resp.ok) {
    const text = await resp.text();
    throw new Error(`Cloudinary delete failed: ${resp.status} ${text}`);
  }

  return resp.json();
}

async function deleteImages(publicIds = []) {
  await Promise.all(
    publicIds.filter(Boolean).map((id) => deleteImage(id).catch(() => {}))
  );
}

module.exports = { deleteImage, deleteImages };

// ------------------------------------------------------------
// Helper — sanitize a `documents` array (PDFs)
// ------------------------------------------------------------
function cleanDocuments(documents) {
  if (!Array.isArray(documents)) return [];
  return documents
    .slice(0, 10)
    .map((d) => ({
      url: String(d?.url || ""),
      publicId: String(d?.publicId || ""),
      name: String(d?.name || "file.pdf").slice(0, 200),
      size: Math.max(0, Number(d?.size) || 0),
    }))
    .filter((d) => d.url && d.publicId);
}

// ------------------------------------------------------------
// ADMIN: LIST + CREATE
// ------------------------------------------------------------
function buildAdminListHandler(
  collectionName,
  { allowWebsiteUrl = false, allowDocuments = false } = {}
) {
  return async function handler(req, res) {
    if (handleOptions(req, res)) return;
    setCors(req, res);

    const decoded = getAdminFromRequest(req);
    if (!decoded) return res.status(401).json({ message: "Unauthorized" });

    try {
      const { db } = await connectToDatabase();
      const col = db.collection(collectionName);

      if (req.method === "GET") {
        const { category, status, search } = req.query || {};
        const filter = {};
        if (category && category !== "all") filter.category = category;
        if (status && status !== "all") filter.status = status;
        if (search) {
          const rx = new RegExp(String(search).trim(), "i");
          filter.$or = [{ title: rx }, { description: rx }];
        }
        const list = await col
          .find(filter, {
            projection: {
              title: 1,
              category: 1,
              description: 1,
              image: 1,
              ...(allowWebsiteUrl && { websiteUrl: 1 }),
              ...(allowDocuments && { documents: 1 }),
              status: 1,
              publishedAt: 1,
              createdAt: 1,
              views: 1,
            },
          })
          .sort({ createdAt: -1 })
          .limit(500)
          .toArray();
        return res.status(200).json(list);
      }

      if (req.method === "POST") {
        const {
          title,
          category,
          description,
          content,
          image,
          websiteUrl,
          status,
          documents,
        } = req.body || {};

        if (!title || !String(title).trim()) {
          return res.status(400).json({ message: "Title is required" });
        }
        if (!VALID_CATEGORY_IDS.includes(category)) {
          return res.status(400).json({ message: "Invalid category" });
        }

        const now = new Date();
        const isPublished = status === "published";

        const doc = {
          title: String(title).trim().slice(0, 200),
          category,
          description: String(description || "").trim().slice(0, 500),
          content: cleanHtml(content),
          image: image
            ? {
                url: String(image.url || ""),
                publicId: String(image.publicId || ""),
              }
            : null,
          ...(allowWebsiteUrl && {
            websiteUrl: websiteUrl
              ? String(websiteUrl).trim().slice(0, 500)
              : "",
          }),
          ...(allowDocuments && { documents: cleanDocuments(documents) }),
          status: isPublished ? "published" : "draft",
          publishedAt: isPublished ? now : null,
          views: 0,
          createdAt: now,
          updatedAt: now,
        };

        const result = await col.insertOne(doc);
        return res.status(201).json({ _id: result.insertedId, ...doc });
      }

      return res.status(405).json({ message: "Method not allowed" });
    } catch (err) {
      console.error(`Admin ${collectionName} error:`, err);
      return res.status(500).json({ message: "Internal server error" });
    }
  };
}

// ------------------------------------------------------------
// ADMIN: GET ONE / UPDATE / DELETE
// ------------------------------------------------------------
function buildAdminDetailHandler(
  collectionName,
  { allowWebsiteUrl = false, allowDocuments = false } = {}
) {
  return async function handler(req, res) {
    if (handleOptions(req, res)) return;
    setCors(req, res);

    const decoded = getAdminFromRequest(req);
    if (!decoded) return res.status(401).json({ message: "Unauthorized" });

    const id = req.query?.id || req.params?.id;
    if (!id || !ObjectId.isValid(id)) {
      return res.status(400).json({ message: "Invalid ID" });
    }

    const { db } = await connectToDatabase();
    const col = db.collection(collectionName);
    const _id = new ObjectId(id);

    try {
      const existing = await col.findOne({ _id });
      if (!existing) return res.status(404).json({ message: "Not found" });

      if (req.method === "GET") {
        return res.status(200).json(existing);
      }

      if (req.method === "PUT" || req.method === "PATCH") {
        const {
          title,
          category,
          description,
          content,
          image,
          websiteUrl,
          status,
          documents,
        } = req.body || {};

        const update = { updatedAt: new Date() };

        if (title !== undefined) {
          const t = String(title).trim();
          if (!t)
            return res.status(400).json({ message: "Title cannot be empty" });
          update.title = t.slice(0, 200);
        }
        if (category !== undefined) {
          if (!VALID_CATEGORY_IDS.includes(category)) {
            return res.status(400).json({ message: "Invalid category" });
          }
          update.category = category;
        }
        if (description !== undefined) {
          update.description = String(description).trim().slice(0, 500);
        }
        if (content !== undefined) {
          update.content = cleanHtml(content);
        }
        if (allowWebsiteUrl && websiteUrl !== undefined) {
          update.websiteUrl = String(websiteUrl || "").trim().slice(0, 500);
        }
        if (image !== undefined) {
          update.image = image
            ? {
                url: String(image.url || ""),
                publicId: String(image.publicId || ""),
              }
            : null;

          if (
            existing.image?.publicId &&
            existing.image.publicId !== update.image?.publicId
          ) {
            deleteImage(existing.image.publicId).catch((e) =>
              console.error("Cloudinary cleanup (image):", e)
            );
          }
        }

        // -------- PDF documents --------
        if (allowDocuments && documents !== undefined) {
          const incoming = cleanDocuments(documents);
          const incomingIds = new Set(incoming.map((d) => d.publicId));
          const removed = (existing.documents || []).filter(
            (d) => d.publicId && !incomingIds.has(d.publicId)
          );
          for (const doc of removed) {
            deleteImage(doc.publicId, "raw").catch((e) =>
              console.error("Cloudinary cleanup (doc):", e)
            );
          }
          update.documents = incoming;
        }

        if (status !== undefined) {
          const isPublished = status === "published";
          update.status = isPublished ? "published" : "draft";
          if (isPublished && !existing.publishedAt) {
            update.publishedAt = new Date();
          }
        }

        await col.updateOne({ _id }, { $set: update });
        const updated = await col.findOne({ _id });
        return res.status(200).json(updated);
      }

      if (req.method === "DELETE") {
        if (existing.image?.publicId) {
          deleteImage(existing.image.publicId).catch((e) =>
            console.error("Cloudinary cleanup (image):", e)
          );
        }
        for (const doc of existing.documents || []) {
          if (doc.publicId) {
            deleteImage(doc.publicId, "raw").catch((e) =>
              console.error("Cloudinary cleanup (doc):", e)
            );
          }
        }
        await col.deleteOne({ _id });
        return res.status(200).json({ message: "Deleted" });
      }

      return res.status(405).json({ message: "Method not allowed" });
    } catch (err) {
      console.error(`Admin ${collectionName} [id] error:`, err);
      return res.status(500).json({ message: "Internal server error" });
    }
  };
}

// backend/api/lib/cors.cjs
function setCors(req, res) {
  const allowedOrigins = [
    "http://localhost:5173",
    "http://localhost:5174",
    "http://localhost:3000",
    process.env.FRONTEND_URL,
    process.env.CLIENT_URL,
  ].filter(Boolean);

  const origin = req.headers.origin;

  if (origin && allowedOrigins.includes(origin)) {
    res.setHeader("Access-Control-Allow-Origin", origin);
  } else if (!origin) {
    res.setHeader("Access-Control-Allow-Origin", "*");
  } else {
    res.setHeader("Access-Control-Allow-Origin", origin);
  }

  res.setHeader("Access-Control-Allow-Credentials", "true");
  res.setHeader(
    "Access-Control-Allow-Methods",
    "GET,POST,PUT,DELETE,OPTIONS,PATCH"
  );
  res.setHeader(
    "Access-Control-Allow-Headers",
    "Content-Type, Authorization"
  );
  res.setHeader("Access-Control-Max-Age", "86400");
}

function handleOptions(req, res) {
  if (req.method === "OPTIONS") {
    setCors(req, res);
    res.status(204).end();
    return true;
  }
  return false;
}

module.exports = { setCors, handleOptions };

// backend/api/lib/discount.cjs
const { connectToDatabase } = require("./db.cjs");

/**
 * Get currently-active store-wide offers.
 * An offer is active if: active === true AND now between startsAt and endsAt.
 */
async function getActiveOffers(db) {
  const now = new Date();
  const offers = await db
    .collection("offers")
    .find({
      active: true,
      startsAt: { $lte: now },
      endsAt: { $gte: now },
    })
    .toArray();
  return offers;
}

/**
 * Given a book and the list of active offers, return the best applicable discount.
 * Returns { percent, source, endsAt, offerId, offerName } or null.
 */
function bestDiscountForBook(book, activeOffers) {
  const candidates = [];

  // Per-book discount
  if (book.discountPercent && book.discountPercent > 0) {
    const endsAt = book.discountEndsAt ? new Date(book.discountEndsAt) : null;
    const stillActive = !endsAt || endsAt > new Date();
    if (stillActive) {
      candidates.push({
        percent: Math.min(90, Number(book.discountPercent)),
        source: "book",
        endsAt: endsAt,
        offerId: null,
        offerName: null,
      });
    }
  }

  // Store-wide offers
  for (const offer of activeOffers) {
    // Category filter: empty array (or missing) = all books
    const cats = Array.isArray(offer.categories) ? offer.categories : [];
    if (cats.length > 0 && !cats.includes(book.category)) continue;

    candidates.push({
      percent: Math.min(90, Number(offer.percent) || 0),
      source: "offer",
      endsAt: new Date(offer.endsAt),
      offerId: offer._id.toString(),
      offerName: offer.name || "Offer",
    });
  }

  if (candidates.length === 0) return null;

  // Pick highest percent. On tie, prefer per-book (more specific).
  candidates.sort((a, b) => {
    if (b.percent !== a.percent) return b.percent - a.percent;
    return a.source === "book" ? -1 : 1;
  });
  return candidates[0];
}

/**
 * Decorate a book doc with computed pricing fields.
 * IMPORTANT: does NOT mutate the stored DB doc — returns a new object.
 */
function applyPricing(book, activeOffers) {
  const basePrice = Number(book.price) || 0;
  const discount = bestDiscountForBook(book, activeOffers);
  const percent = discount ? discount.percent : 0;
  const finalPrice = Math.round(basePrice * (1 - percent / 100) * 100) / 100;

  return {
    ...book,
    originalPrice: basePrice,
    finalPrice,
    discountPercent: percent,
    discountSource: discount?.source || null,
    discountEndsAt: discount?.endsAt || null,
    offerName: discount?.offerName || null,
    hasDiscount: percent > 0,
  };
}

async function withActiveOffers(db, books) {
  const activeOffers = await getActiveOffers(db);
  const list = Array.isArray(books) ? books : [books];
  const out = list.map((b) => applyPricing(b, activeOffers));
  return Array.isArray(books) ? out : out[0];
}

module.exports = {
  getActiveOffers,
  bestDiscountForBook,
  applyPricing,
  withActiveOffers,
};

// backend/api/lib/otp.cjs
const crypto = require("crypto");
const { connectToDatabase } = require("./db.cjs");

const COLLECTION = "signup_otps";

function generateCode() {
  // 6-digit numeric
  return String(crypto.randomInt(100000, 1000000));
}

function hashCode(code) {
  return crypto
    .createHash("sha256")
    .update(String(code) + (process.env.JWT_SECRET || ""))
    .digest("hex");
}

/**
 * Create (or replace) an OTP for an email, tied to the pending signup payload.
 * The pending signup is stored alongside so that the final /verify endpoint
 * doesn't need to trust the client again.
 */
async function issueOtp({ email, pendingUser }) {
  const { db } = await connectToDatabase();
  const otps = db.collection(COLLECTION);

  const cleanEmail = String(email).toLowerCase().trim();
  const code = generateCode();
  const ttlMin = Number(process.env.OTP_TTL_MINUTES || 10);
  const expiresAt = new Date(Date.now() + ttlMin * 60 * 1000);

  await otps.deleteMany({ email: cleanEmail }); // one active code per email

  await otps.insertOne({
    email: cleanEmail,
    codeHash: hashCode(code),
    pendingUser,              // { name, email, phone, passwordHash }
    attempts: 0,
    maxAttempts: Number(process.env.OTP_MAX_ATTEMPTS || 5),
    createdAt: new Date(),
    expiresAt,
  });

  return { code, expiresAt, ttlMin };
}

/**
 * Verify an OTP. Returns { ok, pendingUser } or { ok:false, reason }.
 * Does NOT delete the OTP — caller decides (signup endpoint will delete on success).
 */
async function verifyOtp({ email, code }) {
  const { db } = await connectToDatabase();
  const otps = db.collection(COLLECTION);
  const cleanEmail = String(email).toLowerCase().trim();

  const doc = await otps.findOne({ email: cleanEmail });
  if (!doc) return { ok: false, reason: "no-code" };

  if (new Date(doc.expiresAt) < new Date()) {
    await otps.deleteOne({ _id: doc._id });
    return { ok: false, reason: "expired" };
  }

  if (doc.attempts >= (doc.maxAttempts || 5)) {
    await otps.deleteOne({ _id: doc._id });
    return { ok: false, reason: "too-many-attempts" };
  }

  if (hashCode(code) !== doc.codeHash) {
    await otps.updateOne({ _id: doc._id }, { $inc: { attempts: 1 } });
    return { ok: false, reason: "wrong-code" };
  }

  return { ok: true, pendingUser: doc.pendingUser, otpId: doc._id };
}

async function consumeOtp(otpId) {
  const { db } = await connectToDatabase();
  await db.collection(COLLECTION).deleteOne({ _id: otpId });
}

module.exports = { issueOtp, verifyOtp, consumeOtp };

// backend/api/orders/[id]/payment.cjs
const { ObjectId } = require("mongodb");
const { connectToDatabase } = require("../../lib/db.cjs");
const { getUserFromRequest } = require("../../lib/auth.cjs");
const { setCors, handleOptions } = require("../../lib/cors.cjs");

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  const decoded = getUserFromRequest(req);
  if (!decoded) return res.status(401).json({ message: "Unauthorized" });

  const id = req.query?.id || req.params?.id;
  if (!id || !ObjectId.isValid(id)) {
    return res.status(400).json({ message: "Invalid order ID" });
  }

  if (req.method !== "POST") {
    return res.status(405).json({ message: "Method not allowed" });
  }

  try {
    const { files } = req.body || {};
    if (!Array.isArray(files) || files.length === 0) {
      return res.status(400).json({ message: "No files provided" });
    }

    const { db } = await connectToDatabase();
    const orders = db.collection("orders");

    const order = await orders.findOne({
      _id: new ObjectId(id),
      userId: decoded.id,
    });
    if (!order) return res.status(404).json({ message: "Order not found" });

    const proof = files.slice(0, 5).map((f) => ({
      url: String(f.url || ""),
      publicId: String(f.publicId || ""),
      name: String(f.name || ""),
    }));

    await orders.updateOne(
      { _id: new ObjectId(id) },
      {
        $set: {
          paymentProof: proof,
          paymentStatus: "pending-verification",
          updatedAt: new Date(),
        },
        $push: {
          statusHistory: {
            status: "payment-submitted",
            at: new Date(),
            note: `${proof.length} file(s) uploaded`,
          },
        },
      }
    );

    return res.status(200).json({ message: "Payment proof uploaded", proof });
  } catch (err) {
    console.error("Payment upload error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};

// backend/api/orders/index.cjs
const { ObjectId } = require("mongodb");
const { connectToDatabase } = require("../lib/db.cjs");
const { getUserFromRequest } = require("../lib/auth.cjs");
const { setCors, handleOptions } = require("../lib/cors.cjs");
const { getSettings } = require("../lib/settings.cjs");
const { withActiveOffers } = require("../lib/discount.cjs");
const { sendOrderNotificationToAdmin } = require("../lib/mailer.cjs");

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  const decoded = getUserFromRequest(req);
  if (!decoded) return res.status(401).json({ message: "Unauthorized" });

  const { db } = await connectToDatabase();
  const orders = db.collection("orders");
  const books = db.collection("books");
  const settings = await getSettings();

  try {
    if (req.method === "GET") {
      const list = await orders
        .find({ userId: decoded.id })
        .sort({ createdAt: -1 })
        .toArray();
      return res.status(200).json(list);
    }

    if (req.method === "POST") {
      console.log("📥 POST /api/orders body:", JSON.stringify(req.body));

      const {
        items,
        shippingAddress,
        notes,
        paymentMethod,
        deliveryZoneId,
      } = req.body || {};

      if (!Array.isArray(items) || items.length === 0) {
        return res.status(400).json({ message: "Cart is empty" });
      }

      if (
        !shippingAddress ||
        !shippingAddress.name ||
        !shippingAddress.phone ||
        !shippingAddress.address
      ) {
        return res
          .status(400)
          .json({ message: "Shipping address is required" });
      }

      const method = paymentMethod === "cod" ? "cod" : "qr";
      const allowQR = settings.paymentMethods?.qr !== false;
      const allowCOD = settings.paymentMethods?.cod !== false;

      if (method === "qr" && !allowQR) {
        return res.status(400).json({
          message: "QR payment is currently disabled. Please choose COD.",
        });
      }
      if (method === "cod" && !allowCOD) {
        return res.status(400).json({
          message: "Cash on Delivery is currently disabled. Please choose QR.",
        });
      }

      const zones = Array.isArray(settings.deliveryZones)
        ? settings.deliveryZones
        : [];

      let chosenZone = null;
      if (zones.length > 0) {
        if (deliveryZoneId) {
          chosenZone = zones.find((z) => z.id === deliveryZoneId);
          if (!chosenZone) {
            return res
              .status(400)
              .json({ message: "Invalid delivery zone" });
          }
        } else {
          console.warn(
            "⚠️ No deliveryZoneId received — using first zone:",
            zones[0]?.id
          );
          chosenZone = zones[0];
        }
      }

      const deliveryFee = chosenZone
        ? Number(chosenZone.charge) || 0
        : Number(settings.deliveryCharge) || 0;

      const validBookIds = items
        .filter((i) => i && i.bookId && ObjectId.isValid(i.bookId))
        .map((i) => new ObjectId(i.bookId));

      if (validBookIds.length === 0) {
        return res.status(400).json({ message: "No valid books in order" });
      }

      const bookDocs = await books
        .find(
          { _id: { $in: validBookIds } },
          {
            projection: {
              title: 1,
              author: 1,
              price: 1,
              images: 1,
              stock: 1,
              category: 1,
              discountPercent: 1,
              discountEndsAt: 1,
            },
          }
        )
        .toArray();

      const pricedBooks = await withActiveOffers(db, bookDocs);
      const byId = new Map(pricedBooks.map((b) => [b._id.toString(), b]));

      for (const item of items) {
        const b = byId.get(item.bookId);
        if (!b) {
          return res
            .status(400)
            .json({ message: `Book not found: ${item.bookId}` });
        }
        const qty = Number(item.quantity) || 1;
        if (qty < 1) {
          return res.status(400).json({ message: "Invalid quantity" });
        }
        if (b.stock < qty) {
          return res.status(400).json({
            message: `Not enough stock for "${b.title}". Available: ${b.stock}`,
          });
        }
      }

      const enrichedItems = items.map((i) => {
        const b = byId.get(i.bookId);
        return {
          bookId: i.bookId,
          title: b.title,
          author: b.author,
          price: b.finalPrice,
          originalPrice: b.originalPrice,
          discountPercent: b.discountPercent || 0,
          image: b.images?.[0]?.url || null,
          quantity: Number(i.quantity) || 1,
        };
      });

      const subtotal = enrichedItems.reduce(
        (s, i) => s + i.price * i.quantity,
        0
      );
      const total = subtotal + deliveryFee;

      const orderNumber = `HBC-${Date.now()
        .toString()
        .slice(-8)}-${Math.floor(Math.random() * 9000 + 1000)}`;

      const doc = {
        orderNumber,
        userId: decoded.id,
        items: enrichedItems,
        subtotal,
        deliveryFee,
        deliveryZone: chosenZone
          ? {
              id: chosenZone.id,
              name: chosenZone.name,
              charge: Number(chosenZone.charge) || 0,
            }
          : null,
        total,
        shippingAddress: {
          name: String(shippingAddress.name).trim(),
          phone: String(shippingAddress.phone).trim(),
          address: String(shippingAddress.address).trim(),
          city: String(shippingAddress.city || "").trim(),
        },
        notes: String(notes || "").trim(),
        status: "pending",
        paymentMethod: method,
        paymentStatus: method === "cod" ? "cod-pending" : "unpaid",
        paymentProof: null,
        statusHistory: [
          {
            status: "pending",
            at: new Date(),
            note: chosenZone
              ? `Order placed via ${
                  method === "cod" ? "Cash on Delivery" : "QR Payment"
                } — Zone: ${chosenZone.name}`
              : `Order placed via ${
                  method === "cod" ? "Cash on Delivery" : "QR Payment"
                }`,
          },
        ],
        createdAt: new Date(),
        updatedAt: new Date(),
      };

      for (const item of enrichedItems) {
        await books.updateOne(
          { _id: new ObjectId(item.bookId) },
          {
            $inc: { stock: -item.quantity },
            $set: { updatedAt: new Date() },
          }
        );
      }

      const result = await orders.insertOne(doc);

      await db
        .collection("carts")
        .updateOne(
          { userId: decoded.id },
          { $set: { items: [], updatedAt: new Date() } }
        );

      // 💡 Fire-and-forget admin notification — never blocks the response
      try {
        const user = await db
          .collection("users")
          .findOne(
            { _id: new ObjectId(decoded.id) },
            { projection: { email: 1, name: 1 } }
          );

        sendOrderNotificationToAdmin({
          orderNumber: doc.orderNumber,
          customerName: doc.shippingAddress.name,
          customerPhone: doc.shippingAddress.phone,
          customerEmail: user?.email || "",
          shippingAddress: doc.shippingAddress.address,
          city: doc.shippingAddress.city,
          paymentMethod: doc.paymentMethod,
          paymentStatus: doc.paymentStatus,
          items: doc.items,
          subtotal: doc.subtotal,
          deliveryFee: doc.deliveryFee,
          deliveryZoneName: doc.deliveryZone?.name || null,
          total: doc.total,
          notes: doc.notes,
          createdAt: doc.createdAt,
        }).catch((e) => console.error("Order notify email failed:", e));
      } catch (e) {

// backend/api/reviews/[bookId].cjs
const { ObjectId } = require("mongodb");
const { connectToDatabase } = require("../lib/db.cjs");
const { getUserFromRequest } = require("../lib/auth.cjs");
const { setCors, handleOptions } = require("../lib/cors.cjs");

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  const bookId = req.query?.bookId || req.params?.bookId;
  if (!bookId || !ObjectId.isValid(bookId)) {
    return res.status(400).json({ message: "Invalid book ID" });
  }

  const { db } = await connectToDatabase();
  const reviews = db.collection("reviews");

  try {
    // ---------- GET reviews ----------
    if (req.method === "GET") {
      const list = await reviews
        .find({ bookId: new ObjectId(bookId) })
        .sort({ createdAt: -1 })
        .toArray();

      const count = list.length;
      const avg =
        count === 0
          ? 0
          : list.reduce((s, r) => s + (r.rating || 0), 0) / count;

      return res.status(200).json({
        reviews: list.map((r) => ({
          _id: r._id,
          userId: r.userId,
          userName: r.userName,
          rating: r.rating,
          comment: r.comment,
          createdAt: r.createdAt,
        })),
        averageRating: Number(avg.toFixed(1)),
        totalReviews: count,
      });
    }

    // ---------- POST a review ----------
    if (req.method === "POST") {
      const decoded = getUserFromRequest(req);
      if (!decoded) return res.status(401).json({ message: "Unauthorized" });

      const { rating, comment } = req.body || {};
      const numRating = Number(rating);
      if (!numRating || numRating < 1 || numRating > 5) {
        return res
          .status(400)
          .json({ message: "Rating must be between 1 and 5" });
      }

      // One review per user per book
      const existing = await reviews.findOne({
        bookId: new ObjectId(bookId),
        userId: decoded.id,
      });
      if (existing) {
        return res
          .status(409)
          .json({ message: "You have already reviewed this book" });
      }

      const user = await db
        .collection("users")
        .findOne({ _id: new ObjectId(decoded.id) });

      const doc = {
        bookId: new ObjectId(bookId),
        userId: decoded.id,
        userName: user?.name || "Anonymous",
        rating: numRating,
        comment: String(comment || "").trim(),
        createdAt: new Date(),
      };

      const result = await reviews.insertOne(doc);

      return res.status(201).json({
        _id: result.insertedId,
        ...doc,
      });
    }

    return res.status(405).json({ message: "Method not allowed" });
  } catch (err) {
    console.error("Reviews error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};

  const users = db.collection("users");
  const _id = new ObjectId(decoded.id);

  try {
    // ---------- GET ----------
    if (req.method === "GET") {
      const user = await users.findOne({ _id });
      return res.status(200).json(user?.addresses || []);
    }

    // ---------- POST add ----------
    if (req.method === "POST") {
      const { label, name, phone, address, city, isDefault } = req.body || {};
      if (!name || !phone || !address) {
        return res
          .status(400)
          .json({ message: "Name, phone and address are required" });
      }

      const newAddress = {
        _id: new ObjectId(),
        label: String(label || "Home").trim(),
        name: String(name).trim(),
        phone: String(phone).trim(),
        address: String(address).trim(),
        city: String(city || "").trim(),
        isDefault: !!isDefault,
        createdAt: new Date(),
      };

      const user = await users.findOne({ _id });
      let addresses = user?.addresses || [];

      if (newAddress.isDefault) {
        addresses = addresses.map((a) => ({ ...a, isDefault: false }));
      }
      addresses.push(newAddress);

      await users.updateOne(
        { _id },
        { $set: { addresses, updatedAt: new Date() } }
      );

      return res.status(201).json(newAddress);
    }

    // ---------- DELETE ----------
    if (req.method === "DELETE") {
      const { addressId } = req.body || {};
      if (!addressId) {
        return res.status(400).json({ message: "addressId is required" });
      }

      await users.updateOne(
        { _id },
        { $pull: { addresses: { _id: new ObjectId(addressId) } } }
      );

      return res.status(200).json({ message: "Address removed" });
    }

    return res.status(405).json({ message: "Method not allowed" });
  } catch (err) {
    console.error("Addresses error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};











// backend/api/user/messages.cjs
const { connectToDatabase } = require("../lib/db.cjs");
const { getUserFromRequest } = require("../lib/auth.cjs");
const { setCors, handleOptions } = require("../lib/cors.cjs");
const { ObjectId } = require("mongodb");

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  const decoded = getUserFromRequest(req);
  if (!decoded) return res.status(401).json({ message: "Unauthorized" });

  const { db } = await connectToDatabase();
  const messages = db.collection("contact_messages");

  try {
    if (req.method === "GET") {
      // Get user's email so we can also match messages they sent while logged out
      const user = await db
        .collection("users")
        .findOne({ _id: new ObjectId(decoded.id) }, { projection: { email: 1 } });
      if (!user) return res.status(404).json({ message: "User not found" });

      const list = await messages
        .find({
          $or: [
            { userId: decoded.id },
            { email: user.email },
          ],
        })
        .sort({ createdAt: -1 })
        .limit(100)
        .toArray();

      return res.status(200).json(list);
    }

    if (req.method === "POST") {
      // User follows up in an existing thread
      const { threadId, text } = req.body || {};
      if (!threadId || !ObjectId.isValid(threadId)) {
        return res.status(400).json({ message: "Invalid thread ID" });
      }
      if (!text || String(text).trim().length < 2) {
        return res.status(400).json({ message: "Message is too short" });
      }

      const msg = await messages.findOne({ _id: new ObjectId(threadId) });
      if (!msg) return res.status(404).json({ message: "Thread not found" });

      const user = await db
        .collection("users")
        .findOne({ _id: new ObjectId(decoded.id) }, { projection: { email: 1 } });
      if (!user) return res.status(404).json({ message: "User not found" });

      // Only the owner can append
      if (msg.userId !== decoded.id && msg.email !== user.email) {
        return res.status(403).json({ message: "Forbidden" });
      }

      const now = new Date();
      await messages.updateOne(
        { _id: msg._id },
        {
          $push: {
            thread: { from: "user", text: String(text).trim().slice(0, 4000), at: now, adminName: null },
          },
          $set: { status: "new", updatedAt: now },
        }
      );

      const updated = await messages.findOne({ _id: msg._id });
      return res.status(200).json(updated);
    }

    return res.status(405).json({ message: "Method not allowed" });
  } catch (err) {
    console.error("User messages error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};










// backend/api/user/password.cjs
const { ObjectId } = require("mongodb");
const bcrypt = require("bcryptjs");
const { connectToDatabase } = require("../lib/db.cjs");
const { getUserFromRequest } = require("../lib/auth.cjs");
const { setCors, handleOptions } = require("../lib/cors.cjs");

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  if (req.method !== "PUT") {
    return res.status(405).json({ message: "Method not allowed" });
  }

  const decoded = getUserFromRequest(req);
  if (!decoded) return res.status(401).json({ message: "Unauthorized" });

  try {
    const { currentPassword, newPassword } = req.body || {};
    if (!currentPassword || !newPassword) {
      return res
        .status(400)
        .json({ message: "Current and new password are required" });
    }
    if (String(newPassword).length < 6) {
      return res
        .status(400)
        .json({ message: "New password must be at least 6 characters" });
    }

    const { db } = await connectToDatabase();
    const users = db.collection("users");
    const _id = new ObjectId(decoded.id);

    const user = await users.findOne({ _id });
    if (!user) return res.status(404).json({ message: "User not found" });

    const valid = await bcrypt.compare(currentPassword, user.password);
    if (!valid) {
      return res
        .status(401)
        .json({ message: "Current password is incorrect" });
    }

    const hashed = await bcrypt.hash(newPassword, 10);
    await users.updateOne(
      { _id },
      { $set: { password: hashed, updatedAt: new Date() } }
    );

    return res.status(200).json({ message: "Password updated" });
  } catch (err) {
    console.error("Password error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};

// backend/api/wishlist/index.cjs
const { ObjectId } = require("mongodb");
const { connectToDatabase } = require("../lib/db.cjs");
const { getUserFromRequest } = require("../lib/auth.cjs");
const { setCors, handleOptions } = require("../lib/cors.cjs");
const { withActiveOffers } = require("../lib/discount.cjs");

module.exports = async function handler(req, res) {
  if (handleOptions(req, res)) return;
  setCors(req, res);

  const decoded = getUserFromRequest(req);
  if (!decoded) return res.status(401).json({ message: "Unauthorized" });

  const { db } = await connectToDatabase();
  const users = db.collection("users");
  const _id = new ObjectId(decoded.id);

  try {
    if (req.method === "GET") {
      const user = await users.findOne({ _id });
      const ids = (user?.wishlist || [])
        .filter((id) => ObjectId.isValid(id))
        .map((id) => new ObjectId(id));

      if (ids.length === 0) return res.status(200).json([]);

      const books = await db
        .collection("books")
        .find(
          { _id: { $in: ids } },
          {
            projection: {
              title: 1,
              author: 1,
              price: 1,
              category: 1,
              images: 1,
              stock: 1,
              isbn: 1,
              discountPercent: 1,
              discountEndsAt: 1,
            },
          }
        )
        .toArray();

      const priced = await withActiveOffers(db, books);
      return res.status(200).json(priced);
    }

    if (req.method === "POST") {
      const { bookId } = req.body || {};
      if (!bookId || !ObjectId.isValid(bookId)) {
        return res.status(400).json({ message: "Invalid book ID" });
      }

      const user = await users.findOne({ _id });
      const current = user?.wishlist || [];
      const exists = current.includes(bookId);

      if (exists) {
        await users.updateOne({ _id }, { $pull: { wishlist: bookId } });
      } else {
        await users.updateOne({ _id }, { $addToSet: { wishlist: bookId } });
      }

      return res.status(200).json({
        inWishlist: !exists,
        message: exists ? "Removed from wishlist" : "Added to wishlist",
      });
    }

    return res.status(405).json({ message: "Method not allowed" });
  } catch (err) {
    console.error("Wishlist error:", err);
    return res.status(500).json({ message: "Internal server error" });
  }
};

require("dotenv").config();

const express = require("express");
const cors = require("cors");

const app = express();

app.use(express.json({ limit: "10mb" }));

// Build allowed origins from env — dev + LAN
const allowedOrigins = [
  "http://localhost:5173",
  "http://localhost:5174",
  "http://localhost:3000",
  process.env.FRONTEND_URL,
  process.env.CLIENT_URL,
  process.env.LAN_FRONTEND_URL,
  process.env.LAN_CLIENT_URL,
].filter(Boolean);

console.log("✅ CORS allowed origins:");
allowedOrigins.forEach((o) => console.log("   -", o));

app.use(
  cors({
    origin: function (origin, callback) {
      // Allow requests with no origin (curl, mobile apps)
      if (!origin) return callback(null, true);

      // Allow any localhost/127.0.0.1 port
      if (/^https?:\/\/(localhost|127\.0\.0\.1)(:\d+)?$/.test(origin)) {
        return callback(null, true);
      }

      // Allow any private LAN IP (192.168.x.x, 10.x.x.x, 172.16-31.x.x)
      if (
        /^https?:\/\/(192\.168\.\d+\.\d+|10\.\d+\.\d+\.\d+|172\.(1[6-9]|2\d|3[01])\.\d+\.\d+)(:\d+)?$/.test(
          origin
        )
      ) {
        return callback(null, true);
      }

      // Allow explicit whitelist
      if (allowedOrigins.includes(origin)) {
        return callback(null, true);
      }

      console.warn("⛔ CORS blocked origin:", origin);
      return callback(null, false);
    },
    credentials: true,
    methods: ["GET", "POST", "PUT", "DELETE", "OPTIONS", "PATCH"],
    allowedHeaders: ["Content-Type", "Authorization"],
  })
);

// Adapter for Vercel-style handlers
function adapt(handler) {
  return async (req, res) => {
    req.query = { ...req.query, ...req.params };
    try {
      await handler(req, res);
    } catch (err) {
      console.error("Handler error:", err);
      if (!res.headersSent) {
        res.status(500).json({ message: "Internal server error" });
      }
    }
  };
}
