// client/src/components/BookCard.jsx
import { Link } from "react-router-dom";

export default function BookCard({ book }) {
  const cover = book.images?.[0]?.url;

  const stockStatus =
    book.stock === 0
      ? { label: "Out of Stock", cls: "bg-red-100 text-red-700" }
      : book.stock <= 5
      ? { label: "Only few left", cls: "bg-yellow-100 text-yellow-800" }
      : { label: "In Stock", cls: "bg-green-100 text-green-700" };

  return (
    <Link
      to={`/books/${book._id}`}
      className="group bg-white rounded-2xl shadow-md hover:shadow-2xl border-t-4 border-t-yellow-400 hover:-translate-y-1.5 transition-all duration-300 overflow-hidden flex flex-col"
    >
      <div className="relative h-56 bg-gradient-to-br from-red-100 via-yellow-50 to-green-100 flex items-center justify-center overflow-hidden">
        {cover ? (
          <img
            src={cover}
            alt={book.title}
            className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500"
            loading="lazy"
          />
        ) : (
          <div className="w-20 h-28 bg-white rounded-lg shadow-xl flex items-center justify-center border-2 border-red-200">
            <svg
              className="w-10 h-10 text-red-600"
              fill="none"
              stroke="currentColor"
              viewBox="0 0 24 24"
            >
              <path
                strokeLinecap="round"
                strokeLinejoin="round"
                strokeWidth={2}
                d="M12 6.253v13m0-13C10.832 5.477 9.246 5 7.5 5S4.168 5.477 3 6.253v13C4.168 18.477 5.754 18 7.5 18s3.332.477 4.5 1.253m0-13C13.168 5.477 14.754 5 16.5 5c1.747 0 3.332.477 4.5 1.253v13C19.832 18.477 18.247 18 16.5 18c-1.746 0-3.332.477-4.5 1.253"
              />
            </svg>
          </div>
        )}

        <span className="absolute top-3 left-3 bg-white/90 backdrop-blur text-red-700 text-[11px] font-bold uppercase px-2.5 py-1 rounded-full shadow">
          {book.category || "General"}
        </span>

        {book.images?.length > 1 && (
          <span className="absolute top-3 right-3 bg-black/60 backdrop-blur text-white text-[11px] font-semibold px-2 py-1 rounded-full flex items-center gap-1">
            <svg
              className="w-3 h-3"
              fill="none"
              stroke="currentColor"
              viewBox="0 0 24 24"
            >
              <path
                strokeLinecap="round"
                strokeLinejoin="round"
                strokeWidth={2}
                d="M4 16l4.586-4.586a2 2 0 012.828 0L16 16m-2-2l1.586-1.586a2 2 0 012.828 0L20 14m-6-6h.01M6 20h12a2 2 0 002-2V6a2 2 0 00-2-2H6a2 2 0 00-2 2v12a2 2 0 002 2z"
              />
            </svg>
            {book.images.length}
          </span>
        )}
      </div>

      <div className="p-4 flex-1 flex flex-col">
        <h3
          className="font-bold text-gray-900 text-base leading-snug line-clamp-2 mb-1 group-hover:text-red-700 transition"
          title={book.title}
        >
          {book.title}
        </h3>
        <p className="text-sm text-gray-500 line-clamp-1 mb-3">
          by {book.author}
        </p>

        <div className="mt-auto flex items-end justify-between">
          <div>
            <p className="text-lg font-extrabold text-green-700">
              NPR {Number(book.price).toLocaleString()}
            </p>
          </div>
          <span
            className={`text-[10px] font-bold uppercase px-2 py-1 rounded-full ${stockStatus.cls}`}
          >
            {stockStatus.label}
          </span>
        </div>
      </div>
    </Link>
  );
}

// client/src/components/Navbar.jsx
import { Link, NavLink, useNavigate } from "react-router-dom";
import { useEffect, useRef, useState } from "react";
import { useAuth } from "../context/AuthContext";
import { useCart } from "../context/CartContext";
import { useWishlist } from "../context/WishlistContext";

export default function Navbar() {
  const [open, setOpen] = useState(false);
  const [userMenu, setUserMenu] = useState(false);
  const menuRef = useRef(null);

  const { user, isAuthenticated, logout } = useAuth();
  const { count: cartCount } = useCart();
  const { count: wishCount } = useWishlist();
  const navigate = useNavigate();

  const links = [
    { to: "/", label: "Home" },
    { to: "/books", label: "Books" },
    { to: "/about", label: "About" },
    { to: "/contact", label: "Contact" },
  ];

  // Close user menu when clicking outside
  useEffect(() => {
    const onClick = (e) => {
      if (menuRef.current && !menuRef.current.contains(e.target)) {
        setUserMenu(false);
      }
    };
    document.addEventListener("mousedown", onClick);
    return () => document.removeEventListener("mousedown", onClick);
  }, []);

  const initials = (user?.name || "U")
    .split(" ")
    .filter(Boolean)
    .map((s) => s[0])
    .slice(0, 2)
    .join("")
    .toUpperCase();

  const handleLogout = () => {
    logout();
    setUserMenu(false);
    navigate("/");
  };

  return (
    <header className="sticky top-0 z-40 bg-white/95 backdrop-blur border-b border-gray-100 shadow-sm">
      <div className="container-x flex items-center justify-between h-16 gap-4">
        {/* ============= LOGO ============= */}
        <Link to="/" className="flex items-center gap-3 flex-shrink-0">
          <div className="w-10 h-10 rounded-xl border-2 border-yellow-400 bg-white overflow-hidden flex items-center justify-center">
            <img
              src="/logo.jpeg"
              alt="Halesi Books Centre"

              onError={(e) => {
                e.target.style.display = "none";
                e.target.parentElement.innerHTML =
                  '<span class="text-sm font-bold text-green-700">HB</span>';
              }}
            />
          </div>
          <div className="hidden sm:block leading-tight">
            <p className="font-extrabold text-base tracking-tight">
              <span className="text-red-700">Halesi</span>{" "}
              <span className="text-green-700">Books</span>{" "}
              <span className="text-yellow-500">Centre</span>
            </p>
            <p className="text-[10px] text-gray-500 font-medium tracking-wide">
              BOOKS · STATIONERY · MORE
            </p>
          </div>
        </Link>

        {/* ============= DESKTOP NAV ============= */}
        <nav className="hidden md:flex items-center gap-1">

          {/* Wishlist badge */}

// client/src/context/AuthContext.jsx
import { createContext, useContext, useState, useEffect } from "react";
import api from "../utils/api";

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const stored = localStorage.getItem("hbc_user");
    const token = localStorage.getItem("hbc_user_token");
    if (stored && token) {
      try {
        setUser(JSON.parse(stored));
      } catch {
        localStorage.removeItem("hbc_user");
        localStorage.removeItem("hbc_user_token");
      }
    }
    setLoading(false);
  }, []);

  const login = async (email, password) => {
    const { data } = await api.post("/auth/user-login", { email, password });
    localStorage.setItem("hbc_user_token", data.token);
    localStorage.setItem("hbc_user", JSON.stringify(data.user));
    setUser(data.user);
    return data.user;
  };

  const updateUser = (patch) => {
  setUser((prev) => {
    const next = { ...prev, ...patch };
    localStorage.setItem("hbc_user", JSON.stringify(next));
    return next;
  });
};

  const register = async (payload) => {
    const { data } = await api.post("/auth/register", payload);
    localStorage.setItem("hbc_user_token", data.token);
    localStorage.setItem("hbc_user", JSON.stringify(data.user));
    setUser(data.user);
    return data.user;
  };

  const logout = () => {
    localStorage.removeItem("hbc_user_token");
    localStorage.removeItem("hbc_user");
    setUser(null);
  };

  const isAuthenticated = !!user;

  return (
    <AuthContext.Provider
      value={{ user, loading, login, register, logout, isAuthenticated }}
    >
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error("useAuth must be inside AuthProvider");
  return ctx;
}

  const [count, setCount] = useState(0);

  const refresh = async () => {
    if (!isAuthenticated) {
      setItems([]);
      setTotal(0);
      setCount(0);
      return;
    }
    try {
      const { data } = await api.get("/cart");
      setItems(data.items || []);
      setTotal(data.total || 0);
      setCount(data.count || 0);
    } catch {
      /* ignore */
    }
  };

  useEffect(() => {
    refresh();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [isAuthenticated]);

  const add = async (bookId, quantity = 1) => {
    await api.post("/cart", { bookId, quantity });
    await refresh();
  };

  const remove = async (bookId) => {
    await api.delete("/cart", { data: { bookId } });
    await refresh();
  };

  const updateQty = async (bookId, quantity) => {
    await api.put("/cart", { bookId, quantity });
    await refresh();
  };

  return (
    <CartContext.Provider
      value={{ items, total, count, add, remove, updateQty, refresh }}
    >
      {children}
    </CartContext.Provider>
  );
}

export function useCart() {
  const ctx = useContext(CartContext);
  if (!ctx) throw new Error("useCart must be inside CartProvider");
  return ctx;
}

// client/src/context/WishlistContext.jsx
import { createContext, useContext, useState, useEffect } from "react";
import api from "../utils/api";
import { useAuth } from "./AuthContext";

const WishlistContext = createContext(null);

export function WishlistProvider({ children }) {
  const { isAuthenticated } = useAuth();
  const [ids, setIds] = useState(new Set());
  const [items, setItems] = useState([]);

  const refresh = async () => {
    if (!isAuthenticated) {
      setIds(new Set());
      setItems([]);
      return;
    }
    try {
      const { data } = await api.get("/wishlist");
      setItems(data);
      setIds(new Set(data.map((b) => b._id)));
    } catch {
      /* ignore */
    }
  };

  useEffect(() => {
    refresh();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [isAuthenticated]);

  const toggle = async (bookId) => {
    const { data } = await api.post("/wishlist", { bookId });
    if (data.inWishlist) {
      setIds((prev) => new Set(prev).add(bookId));
    } else {
      setIds((prev) => {
        const next = new Set(prev);
        next.delete(bookId);
        return next;
      });
    }
    return data.inWishlist;
  };

  const has = (bookId) => ids.has(bookId);

  return (
    <WishlistContext.Provider
      value={{ ids, items, has, toggle, refresh, count: ids.size }}
    >
      {children}
    </WishlistContext.Provider>
  );
}

export function useWishlist() {
  const ctx = useContext(WishlistContext);
  if (!ctx) throw new Error("useWishlist must be inside WishlistProvider");
  return ctx;
}

import { useNavigate, useLocation } from "react-router-dom";
import { useAuth } from "../context/AuthContext";

export function useProtectedAction() {
  const { isAuthenticated } = useAuth();
  const navigate = useNavigate();
  const location = useLocation();

  return (action) => {
    if (!isAuthenticated) {
      const redirect = encodeURIComponent(location.pathname);
      navigate(`/login?redirect=${redirect}`, { replace: false });
      return false;
    }
    action?.();
    return true;
  };
}

  const { isAuthenticated } = useAuth();
  const { has: inWishlist, toggle: toggleWishlist } = useWishlist();
  const { add: addToCart } = useCart();
  const withAuth = useProtectedAction();

  const [book, setBook] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [activeImage, setActiveImage] = useState(0);
  const [activeTab, setActiveTab] = useState("details");
  const [copied, setCopied] = useState(false);
  const [addingCart, setAddingCart] = useState(false);
  const [buyingNow, setBuyingNow] = useState(false);
  const [toast, setToast] = useState("");
  const [reviewSummary, setReviewSummary] = useState({
    averageRating: 0,
    totalReviews: 0,
  });

  useEffect(() => {
    if (!id) return;
    let cancelled = false;
    (async () => {
      setLoading(true);
      setError("");
      try {
        const { data } = await api.get(`/public/books/${id}`);
        if (cancelled) return;
        setBook(data);
        setActiveImage(0);
        setActiveTab("details");

        // Fetch review summary for star display in the header
        try {
          const rev = await api.get(`/reviews/${id}`);
          if (!cancelled) {
            setReviewSummary({
              averageRating: rev.data.averageRating || 0,
              totalReviews: rev.data.totalReviews || 0,
            });
          }
        } catch {
          /* ignore */
        }
      } catch (err) {
        if (cancelled) return;
        setError(
          err.response?.data?.message || "Failed to load book details."
        );
      } finally {
        if (!cancelled) setLoading(false);
      }
    })();
    return () => {
      cancelled = true;
    };
  }, [id]);

  // Scroll to top on book change
  useEffect(() => {
    window.scrollTo({ top: 0, behavior: "smooth" });
  }, [id]);

  const showToast = (msg) => {
    setToast(msg);
    setTimeout(() => setToast(""), 2500);
  };

  // ---------- SHARE ----------
  const shareUrl = typeof window !== "undefined" ? window.location.href : "";
  const shareText = book
    ? `Check out "${book.title}" by ${book.author} at Halesi Books Centre!`
    : "";

  const handleCopy = async () => {
    try {
      await navigator.clipboard.writeText(shareUrl);
      setCopied(true);
      showToast("Link copied!");
      setTimeout(() => setCopied(false), 2000);
    } catch {
      /* ignore */
    }
  };

  // ---------- ACTIONS ----------
  const handleWishlist = () =>
    withAuth(async () => {
      try {
        const added = await toggleWishlist(id);
        showToast(added ? "Added to wishlist" : "Removed from wishlist");
      } catch {
        showToast("Failed to update wishlist");
      }
    });

  const handleAddToCart = () =>
    withAuth(async () => {
      setAddingCart(true);
      try {
        await addToCart(id, 1);
        showToast("Added to cart");
      } catch {
        showToast("Failed to add to cart");
      } finally {
        setAddingCart(false);
      }
    });

  const handleBuyNow = () =>
  withAuth(() => {
    navigate(`/checkout?bookId=${id}&qty=1`);
  });

  // ---------- LOADING ----------
  if (loading) {
    return (
      <div className="container-x py-10 animate-pulse">
        <div className="h-4 bg-gray-100 rounded w-56 mb-8" />
        <div className="grid grid-cols-1 lg:grid-cols-12 gap-10">
          <div className="lg:col-span-6">
            <div className="aspect-[3/4] bg-gray-100 rounded-3xl" />
          </div>
          <div className="lg:col-span-6 space-y-4">
            <div className="h-8 bg-gray-100 rounded w-3/4" />
            <div className="h-5 bg-gray-100 rounded w-1/2" />
            <div className="h-24 bg-gray-100 rounded-xl mt-6" />
            <div className="h-40 bg-gray-100 rounded-xl" />
          </div>
        </div>
      </div>
    );
  }

  // ---------- ERROR ----------
  if (error || !book) {
    return (
      <div className="container-x py-20">
        <div className="bg-white rounded-2xl shadow-md p-12 max-w-md mx-auto text-center">
          <h2 className="text-2xl font-bold text-gray-900 mb-2">
            Book Not Found
          </h2>
          <p className="text-sm text-gray-500 mb-6">
            {error || "The book you're looking for may have been removed."}
          </p>
          <Link
            to="/books"
            className="inline-flex items-center gap-2 bg-gradient-to-r from-red-700 to-red-600 hover:from-red-800 hover:to-red-700 text-white font-semibold py-2.5 px-5 rounded-xl shadow-md transition"
          >
            Browse All Books
          </Link>
        </div>
      </div>
    );
  }

  // ---------- DERIVED ----------
  const images = book.images || [];
  const mainImage = images[activeImage]?.url;
  const hasMultipleImages = images.length > 1;
  const wishlisted = inWishlist(id);

  const stockInfo = (() => {
    if (book.stock === 0)
      return {
        label: "Out of Stock",
        cls: "bg-red-100 text-red-700 border-red-200",
        dot: "bg-red-500",
        msg: "Currently unavailable — check back soon.",
      };
    if (book.stock <= 5)
      return {
        label: `Only ${book.stock} left`,
        cls: "bg-yellow-100 text-yellow-800 border-yellow-200",
        dot: "bg-yellow-500",
        msg: "Hurry, only a few copies remain!",
      };
    return {
      label: "In Stock",
      cls: "bg-green-100 text-green-700 border-green-200",
      dot: "bg-green-500",
      msg: "Available for immediate pickup.",
    };
  })();

  // ---------- RENDER ----------
  return (
    <div className="animate-fade-in relative">
      {/* Toast */}
      {toast && (
        <div className="fixed top-20 left-1/2 -translate-x-1/2 z-50 bg-gray-900 text-white text-sm font-semibold px-5 py-2.5 rounded-full shadow-xl animate-fade-in">
          {toast}
        </div>
      )}

      {/* Breadcrumb */}
      <div className="bg-gradient-to-r from-red-50 via-yellow-50 to-green-50 border-b border-yellow-100">
        <div className="container-x py-3">
          <nav className="flex items-center gap-2 text-xs text-gray-600 flex-wrap">
            <Link to="/" className="hover:text-red-700 font-medium transition">
              Home
            </Link>
            <span className="text-gray-400">/</span>
            <Link
              to="/books"
              className="hover:text-red-700 font-medium transition"
            >
              Books
            </Link>
            <span className="text-gray-400">/</span>
            <span className="text-red-700 font-semibold">
              {book.category || "General"}
            </span>
            <span className="text-gray-400">/</span>
            <span className="text-gray-800 font-medium truncate max-w-[180px] sm:max-w-[300px]">
              {book.title}
            </span>
          </nav>
        </div>
      </div>

      <div className="container-x py-8 md:py-12">
        <div className="grid grid-cols-1 lg:grid-cols-12 gap-8 lg:gap-10">
          {/* ============ LEFT: IMAGES (STICKY) ============ */}
          <div className="lg:col-span-6 lg:sticky lg:top-24 lg:self-start">
            <div className="relative bg-gradient-to-br from-red-50 via-yellow-50 to-green-50 rounded-3xl border-4 border-yellow-400 overflow-hidden aspect-[3/4] shadow-xl">
              {mainImage ? (
                <img
                  src={mainImage}
                  alt={book.title}
                  className="w-full h-full object-cover"
                />
              ) : (
                <div className="w-full h-full flex items-center justify-center">
                  <div className="w-32 h-40 bg-white rounded-xl shadow-xl flex items-center justify-center border-2 border-red-200">
                    <svg
                      className="w-16 h-16 text-red-600"
                      fill="none"
                      stroke="currentColor"
                      viewBox="0 0 24 24"
                    >
                      <path
                        strokeLinecap="round"
                        strokeLinejoin="round"
                        strokeWidth={2}
                        d="M12 6.253v13m0-13C10.832 5.477 9.246 5 7.5 5S4.168 5.477 3 6.253v13C4.168 18.477 5.754 18 7.5 18s3.332.477 4.5 1.253m0-13C13.168 5.477 14.754 5 16.5 5c1.747 0 3.332.477 4.5 1.253v13C19.832 18.477 18.247 18 16.5 18c-1.746 0-3.332.477-4.5 1.253"
                      />
                    </svg>
                  </div>
                </div>
              )}

              <div className="absolute top-4 left-0 bg-red-600 text-white text-xs font-bold uppercase tracking-wider px-4 py-2 rounded-r-full shadow-lg">
                {book.category || "General"}
              </div>

              {hasMultipleImages && (
                <div className="absolute top-4 right-4 bg-black/70 backdrop-blur text-white text-xs font-semibold px-3 py-1.5 rounded-full flex items-center gap-1.5">
                  <svg
                    className="w-3.5 h-3.5"
                    fill="none"
                    stroke="currentColor"
                    viewBox="0 0 24 24"
                  >
                    <path
                      strokeLinecap="round"
                      strokeLinejoin="round"
                      strokeWidth={2}
                      d="M4 16l4.586-4.586a2 2 0 012.828 0L16 16m-2-2l1.586-1.586a2 2 0 012.828 0L20 14m-6-6h.01M6 20h12a2 2 0 002-2V6a2 2 0 00-2-2H6a2 2 0 00-2 2v12a2 2 0 002 2z"
                    />
                  </svg>
                  {activeImage + 1} / {images.length}
                </div>
              )}
            </div>

            {hasMultipleImages && (
              <div className="mt-4 grid grid-cols-4 gap-3">
                {images.map((img, i) => (
                  <button
                    key={img.publicId || i}
                    onClick={() => setActiveImage(i)}
                    className={`aspect-square rounded-xl overflow-hidden border-2 transition ${
                      activeImage === i
                        ? "border-red-600 ring-2 ring-red-200"
                        : "border-gray-200 hover:border-yellow-400"
                    }`}
                  >
                    <img
                      src={img.url}
                      alt={`${book.title} ${i + 1}`}
                      className="w-full h-full object-cover"
                    />
                  </button>
                ))}
              </div>
            )}

            {/* Share row (under images) */}
            <div className="mt-6 flex items-center gap-2 flex-wrap">
              <span className="text-xs font-semibold text-gray-500 uppercase tracking-wide mr-1">
                Share
              </span>
              <button
                onClick={handleCopy}
                className="w-9 h-9 rounded-full bg-gray-100 hover:bg-gray-200 flex items-center justify-center text-gray-600 transition"
                title="Copy link"
              >
                {copied ? (
                  <svg
                    className="w-4 h-4 text-green-600"
                    fill="none"
                    stroke="currentColor"
                    viewBox="0 0 24 24"
                  >
                    <path
                      strokeLinecap="round"
                      strokeLinejoin="round"
                      strokeWidth={2.5}
                      d="M5 13l4 4L19 7"
                    />
                  </svg>
                ) : (
                  <svg
                    className="w-4 h-4"
                    fill="none"
                    stroke="currentColor"
                    viewBox="0 0 24 24"
                  >
                    <path
                      strokeLinecap="round"
                      strokeLinejoin="round"
                      strokeWidth={2}
                      d="M8 16H6a2 2 0 01-2-2V6a2 2 0 012-2h8a2 2 0 012 2v2m-6 12h8a2 2 0 002-2v-8a2 2 0 00-2-2h-8a2 2 0 00-2 2v8a2 2 0 002 2z"
                    />
                  </svg>
                )}
              </button>
              <a
                href={`https://wa.me/?text=${encodeURIComponent(
                  shareText + " " + shareUrl
                )}`}
                target="_blank"
                rel="noopener noreferrer"
                className="w-9 h-9 rounded-full bg-green-100 hover:bg-green-200 flex items-center justify-center text-green-700 transition"
                title="Share on WhatsApp"
              >
                <svg className="w-4 h-4" fill="currentColor" viewBox="0 0 24 24">
                  <path d="M17.472 14.382c-.297-.149-1.758-.867-2.03-.967-.273-.099-.471-.148-.67.15-.197.297-.767.966-.94 1.164-.173.199-.347.223-.644.075-.297-.15-1.255-.463-2.39-1.475-.883-.788-1.48-1.761-1.653-2.059-.173-.297-.018-.458.13-.606.134-.133.298-.347.446-.52.149-.174.198-.298.298-.497.099-.198.05-.371-.025-.52-.075-.149-.669-1.612-.916-2.207-.242-.579-.487-.5-.669-.51-.173-.008-.371-.01-.57-.01-.198 0-.52.074-.792.372-.272.297-1.04 1.016-1.04 2.479 0 1.462 1.065 2.875 1.213 3.074.149.198 2.096 3.2 5.077 4.487.709.306 1.262.489 1.694.625.712.227 1.36.195 1.871.118.571-.085 1.758-.719 2.006-1.413.248-.694.248-1.289.173-1.413-.074-.124-.272-.198-.57-.347m-5.421 7.403h-.004a9.87 9.87 0 01-5.031-1.378l-.361-.214-3.741.982.998-3.648-.235-.374a9.86 9.86 0 01-1.51-5.26c.001-5.45 4.436-9.884 9.888-9.884 2.64 0 5.122 1.03 6.988 2.898a9.825 9.825 0 012.893 6.994c-.003 5.45-4.437 9.884-9.885 9.884m8.413-18.297A11.815 11.815 0 0012.05 0C5.495 0 .16 5.335.157 11.892c0 2.096.547 4.142 1.588 5.945L.057 24l6.305-1.654a11.882 11.882 0 005.683 1.448h.005c6.554 0 11.89-5.335 11.893-11.893a11.821 11.821 0 00-3.48-8.413z" />
                </svg>
              </a>
              <a
                href={`https://www.facebook.com/sharer/sharer.php?u=${encodeURIComponent(
                  shareUrl
                )}`}
                target="_blank"
                rel="noopener noreferrer"
                className="w-9 h-9 rounded-full bg-blue-100 hover:bg-blue-200 flex items-center justify-center text-blue-700 transition"
                title="Share on Facebook"
              >
                <svg className="w-4 h-4" fill="currentColor" viewBox="0 0 24 24">
                  <path d="M22 12c0-5.523-4.477-10-10-10S2 6.477 2 12c0 4.991 3.657 9.128 8.438 9.878v-6.987h-2.54V12h2.54V9.797c0-2.506 1.492-3.89 3.777-3.89 1.094 0 2.238.195 2.238.195v2.46h-1.26c-1.243 0-1.63.771-1.63 1.562V12h2.773l-.443 2.89h-2.33v6.988C18.343 21.128 22 16.991 22 12z" />
                </svg>
              </a>
            </div>
          </div>

          {/* ============ RIGHT: INFO PANEL ============ */}
          <div className="lg:col-span-6">
            <h1 className="text-3xl md:text-4xl lg:text-[2.75rem] font-extrabold tracking-tight leading-[1.1] text-gray-900">
              {book.title}
            </h1>
            <p className="text-lg text-gray-600 mt-3">
              by{" "}
              <span className="font-semibold text-red-700">
                {book.author}
              </span>
            </p>

            {/* Rating + reviews summary */}
            <div className="flex items-center gap-3 mt-3 flex-wrap">
              <StarRating value={reviewSummary.averageRating} size={18} />
              <span className="text-sm font-semibold text-gray-700">
                {reviewSummary.averageRating || 0}
              </span>
              <a
                href="#reviews"
                className="text-sm text-gray-500 hover:text-red-700 hover:underline"
              >
                ({reviewSummary.totalReviews} review
                {reviewSummary.totalReviews !== 1 ? "s" : ""})
              </a>
            </div>

            {/* Stock + ISBN row */}
            <div className="flex items-center gap-2 flex-wrap mt-5">
              <span
                className={`inline-flex items-center gap-1.5 text-xs font-bold uppercase tracking-wide px-3 py-1.5 rounded-full border ${stockInfo.cls}`}
              >
                <span
                  className={`w-1.5 h-1.5 rounded-full ${stockInfo.dot} ${
                    book.stock > 0 && book.stock <= 5 ? "animate-pulse" : ""
                  }`}
                />
                {stockInfo.label}
              </span>
              {book.isbn && (
                <span className="text-xs text-gray-500 bg-gray-100 px-3 py-1.5 rounded-full font-medium">
                  ISBN: {book.isbn}
                </span>
              )}
              {book.language && (
                <span className="text-xs text-gray-500 bg-gray-100 px-3 py-1.5 rounded-full font-medium">
                  {book.language}
                </span>
              )}
            </div>

            {/* Price card */}
            <div className="bg-gradient-to-br from-red-50 to-yellow-50 rounded-2xl border-2 border-yellow-200 p-5 my-6">
              <div className="flex items-end justify-between gap-4 flex-wrap">
                <div>
                  <p className="text-xs font-semibold text-gray-500 uppercase tracking-wide mb-1">
                    Price
                  </p>
                  <p className="text-4xl font-extrabold text-green-700 leading-none">
                    NPR {Number(book.price).toLocaleString()}
                  </p>
                </div>
                <p className="text-xs text-gray-600 max-w-[180px] text-right">
                  {stockInfo.msg}
                </p>
              </div>
            </div>

            {/* Action buttons */}
            <div className="space-y-3 mb-6">
              {/* Add to Cart + Buy Now row */}
              <div className="grid grid-cols-2 gap-3">
                <button
                  onClick={handleAddToCart}
                  disabled={addingCart || book.stock === 0}
                  className="flex items-center justify-center gap-2 bg-white border-2 border-red-600 text-red-600 hover:bg-red-50 font-bold py-3.5 rounded-xl transition disabled:opacity-60 disabled:cursor-not-allowed"
                >
                  {addingCart ? (
                    <div className="w-4 h-4 border-2 border-red-300 border-t-red-700 rounded-full animate-spin" />
                  ) : (
                    <svg
                      className="w-5 h-5"
                      fill="none"
                      stroke="currentColor"
                      viewBox="0 0 24 24"
                    >
                      <path
                        strokeLinecap="round"
                        strokeLinejoin="round"
                        strokeWidth={2}
                        d="M3 3h2l.4 2M7 13h10l4-8H5.4M7 13L5.4 5M7 13l-2.293 2.293c-.63.63-.184 1.707.707 1.707H17m0 0a2 2 0 100 4 2 2 0 000-4zm-8 2a2 2 0 11-4 0 2 2 0 014 0z"
                      />
                    </svg>
                  )}
                  Add to Cart
                </button>

                <button
                  onClick={handleBuyNow}
                  disabled={buyingNow || book.stock === 0}
                  className="flex items-center justify-center gap-2 bg-gradient-to-r from-red-700 to-red-600 hover:from-red-800 hover:to-red-700 text-white font-bold py-3.5 rounded-xl shadow-lg shadow-red-500/30 hover:shadow-red-500/50 transition disabled:opacity-60 disabled:cursor-not-allowed"
                >
                  {buyingNow ? (
                    <div className="w-4 h-4 border-2 border-white/40 border-t-white rounded-full animate-spin" />
                  ) : (
                    <svg
                      className="w-5 h-5"
                      fill="none"
                      stroke="currentColor"
                      viewBox="0 0 24 24"
                    >
                      <path
                        strokeLinecap="round"
                        strokeLinejoin="round"
                        strokeWidth={2}
                        d="M13 10V3L4 14h7v7l9-11h-7z"
                      />
                    </svg>
                  )}
                  Buy Now
                </button>
              </div>

              {/* Wishlist + Call row */}
              <div className="grid grid-cols-2 gap-3">
                <button
                  onClick={handleWishlist}
                  className={`flex items-center justify-center gap-2 font-bold py-3 rounded-xl border-2 transition ${
                    wishlisted
                      ? "bg-red-600 border-red-600 text-white hover:bg-red-700"
                      : "bg-white border-gray-200 text-gray-700 hover:border-red-300 hover:text-red-700"
                  }`}
                >
                  <svg
                    className="w-5 h-5"
                    fill={wishlisted ? "currentColor" : "none"}
                    stroke="currentColor"
                    viewBox="0 0 24 24"
                  >
                    <path
                      strokeLinecap="round"
                      strokeLinejoin="round"
                      strokeWidth={2}
                      d="M4.318 6.318a4.5 4.5 0 000 6.364L12 20.364l7.682-7.682a4.5 4.5 0 00-6.364-6.364L12 7.636l-1.318-1.318a4.5 4.5 0 00-6.364 0z"
                    />
                  </svg>
                  {wishlisted ? "Wishlisted" : "Add to Wishlist"}
                </button>

                <a
                  href="tel:+9779800000000"
                  className="flex items-center justify-center gap-2 bg-green-600 hover:bg-green-700 text-white font-bold py-3 rounded-xl shadow-md hover:shadow-lg transition"
                >
                  <svg
                    className="w-5 h-5"
                    fill="none"
                    stroke="currentColor"
                    viewBox="0 0 24 24"
                  >
                    <path
                      strokeLinecap="round"
                      strokeLinejoin="round"
                      strokeWidth={2}
                      d="M3 5a2 2 0 012-2h3.28a1 1 0 01.948.684l1.498 4.493a1 1 0 01-.502 1.21l-2.257 1.13a11.042 11.042 0 005.516 5.516l1.13-2.257a1 1 0 011.21-.502l4.493 1.498a1 1 0 01.684.949V19a2 2 0 01-2 2h-1C9.716 21 3 14.284 3 6V5z"
                    />
                  </svg>
                  Call to Order
                </a>
              </div>

              {/* Signin nudge for guests */}
              {!isAuthenticated && (
                <div className="bg-yellow-50 border border-yellow-200 rounded-xl px-4 py-3 text-xs text-yellow-800 flex items-center justify-between gap-2 flex-wrap">
                  <span>
                    🔒 Sign in to add to cart, wishlist and review this book.
                  </span>
                  <div className="flex gap-2">
                    <Link
                      to={`/login?redirect=${encodeURIComponent(
                        window.location.pathname
                      )}`}
                      className="font-bold text-red-700 hover:underline"
                    >
                      Sign In
                    </Link>
                    <span className="text-yellow-400">·</span>
                    <Link
                      to={`/signup?redirect=${encodeURIComponent(
                        window.location.pathname
                      )}`}
                      className="font-bold text-green-700 hover:underline"
                    >
                      Sign Up
                    </Link>
                  </div>
                </div>
              )}
            </div>

            {/* TABS */}
            <div className="border-b border-gray-200 mb-4">
              <div className="flex gap-1">
                {[
                  { id: "details", label: "Details" },
                  { id: "description", label: "Description" },
                  { id: "shipping", label: "Shipping" },
                ].map((tab) => (
                  <button
                    key={tab.id}
                    onClick={() => setActiveTab(tab.id)}
                    className={`relative px-4 py-3 text-sm font-semibold transition ${
                      activeTab === tab.id
                        ? "text-red-700"
                        : "text-gray-500 hover:text-gray-800"
                    }`}
                  >
                    {tab.label}
                    {activeTab === tab.id && (
                      <span className="absolute left-0 right-0 -bottom-px h-0.5 bg-red-600 rounded-t" />
                    )}
                  </button>
                ))}
              </div>
            </div>

            <div className="min-h-[180px]">
              {activeTab === "details" && (
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                  <DetailRow label="Title" value={book.title} />
                  <DetailRow label="Author" value={book.author} />
                  {book.isbn && <DetailRow label="ISBN" value={book.isbn} />}
                  <DetailRow
                    label="Category"
                    value={book.category || "General"}
                  />
                  {book.publishedYear && (
                    <DetailRow
                      label="Published Year"
                      value={book.publishedYear}
                    />
                  )}
                  {book.edition && (
                    <DetailRow label="Edition" value={book.edition} />
                  )}
                  {book.language && (
                    <DetailRow label="Language" value={book.language} />
                  )}
                  {book.colorFormat && (
                    <DetailRow label="Format" value={book.colorFormat} />
                  )}
                  <DetailRow
                    label="Availability"
                    value={
                      book.stock > 0
                        ? `${book.stock} copies in stock`
                        : "Out of stock"
                    }
                  />
                </div>
              )}

              {activeTab === "description" && (
                <div>

// client/src/pages/Checkout.jsx
import { useEffect, useState, useMemo, useRef } from "react";
import { Link, useNavigate, useSearchParams } from "react-router-dom";
import api from "../utils/api";
import { useAuth } from "../context/AuthContext";
import { useCart } from "../context/CartContext";

export default function Checkout() {
  const navigate = useNavigate();
  const [params] = useSearchParams();
  const { isAuthenticated, loading: authLoading } = useAuth();
  const { refresh: refreshCart } = useCart();

  const bookId = params.get("bookId");
  const initialQty = Math.max(1, parseInt(params.get("qty") || "1", 10));

  // ---------- STATE ----------
  const [book, setBook] = useState(null);
  const [quantity, setQuantity] = useState(initialQty);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  // Address
  const [addresses, setAddresses] = useState([]);
  const [selectedAddressId, setSelectedAddressId] = useState(null);
  const [showAddressForm, setShowAddressForm] = useState(false);
  const [saveAddressToProfile, setSaveAddressToProfile] = useState(true);
  const [newAddress, setNewAddress] = useState({
    label: "Home",
    name: "",
    phone: "",
    address: "",
    city: "",
    isDefault: false,
  });

  // Order
  const [notes, setNotes] = useState("");
  const [submitting, setSubmitting] = useState(false);
  const [order, setOrder] = useState(null);

  // Payment proof
  const [paymentFiles, setPaymentFiles] = useState([]);
  const [uploadingProof, setUploadingProof] = useState(false);
  const [proofUploaded, setProofUploaded] = useState(false);
  const fileInputRef = useRef(null);

  // Settings
  const [settings, setSettings] = useState({
    deliveryZones: [],
    deliveryCharge: 0,
    paymentQR: null,
    paymentMethods: { qr: true, cod: true },
  });
  const [paymentMethod, setPaymentMethod] = useState("qr");
  const [deliveryZoneId, setDeliveryZoneId] = useState(null);

  // ---------- AUTH GUARD ----------
  useEffect(() => {
    if (!authLoading && !isAuthenticated) {
      navigate(
        `/login?redirect=${encodeURIComponent(
          window.location.pathname + window.location.search
        )}`,
        { replace: true }
      );
    }
  }, [authLoading, isAuthenticated, navigate]);

  // ---------- FETCH DATA ----------
  useEffect(() => {
    if (!bookId || !isAuthenticated) return;
    let cancelled = false;

    (async () => {
      setLoading(true);
      setError("");
      try {
        const [bookRes, addrRes, settingsRes] = await Promise.all([
          api.get(`/public/books/${bookId}`),
          api.get("/user/addresses"),
          api.get("/settings/public"),
        ]);

        if (cancelled) return;

        setBook(bookRes.data);
        setAddresses(addrRes.data || []);

        const s = settingsRes.data || {};
        const fetched = {
          deliveryZones: Array.isArray(s.deliveryZones)
            ? s.deliveryZones
            : [],
          deliveryCharge: Number(s.deliveryCharge) || 0,
          paymentQR: s.paymentQR || null,
          paymentMethods: s.paymentMethods || { qr: true, cod: true },
        };
        setSettings(fetched);
        console.log("✅ Checkout settings:", fetched);

        // Payment method default
        if (!fetched.paymentMethods.qr && fetched.paymentMethods.cod) {
          setPaymentMethod("cod");
        }

        // 👇 Auto-select first delivery zone — CRITICAL
        if (fetched.deliveryZones.length > 0) {
          setDeliveryZoneId(fetched.deliveryZones[0].id);
        }

        // Address default
        const def =
          (addrRes.data || []).find((a) => a.isDefault) ||
          (addrRes.data || [])[0];
        if (def) {
          setSelectedAddressId(def._id);
          setShowAddressForm(false);
        } else {
          setShowAddressForm(true);
        }
      } catch (err) {
        if (cancelled) return;
        console.error("Checkout load error:", err);
        setError(
          err.response?.data?.message ||
            "Failed to load checkout data. Please refresh."
        );
      } finally {
        if (!cancelled) setLoading(false);
      }
    })();

    return () => {
      cancelled = true;
    };
  }, [bookId, isAuthenticated]);

  // ---------- QUANTITY CLAMP ----------
  const maxQty = book?.stock || 99;
  useEffect(() => {
    if (quantity > maxQty) setQuantity(maxQty);
    if (quantity < 1) setQuantity(1);
  }, [maxQty]); // eslint-disable-line

  // ---------- LIVE PRICING ----------
  const pricing = useMemo(() => {
    if (!book)
      return { unit: 0, subtotal: 0, delivery: 0, total: 0, zone: null };

    const unit = Number(book.price) || 0;
    const subtotal = unit * quantity;

    const zones = settings.deliveryZones || [];
    const zone =
      zones.find((z) => z.id === deliveryZoneId) || zones[0] || null;

    const delivery = zone
      ? Number(zone.charge) || 0
      : Number(settings.deliveryCharge) || 0;

    return { unit, subtotal, delivery, total: subtotal + delivery, zone };
  }, [book, quantity, settings, deliveryZoneId]);

  // ---------- PLACE ORDER ----------
  const handleSubmitOrder = async (e) => {
    e.preventDefault();
    setError("");

    // Address payload
    let addressPayload = null;

    if (showAddressForm || addresses.length === 0) {
      if (
        !newAddress.name.trim() ||
        !newAddress.phone.trim() ||
        !newAddress.address.trim()
      ) {
        setError("Please fill in name, phone and address");
        return;
      }
      addressPayload = { ...newAddress };

      if (saveAddressToProfile) {
        try {
          await api.post("/user/addresses", newAddress);
        } catch {
          /* ignore */
        }
      }
    } else {
      const selected = addresses.find((a) => a._id === selectedAddressId);
      if (!selected) {
        setError("Please select an address");
        return;
      }
      addressPayload = {
        name: selected.name,
        phone: selected.phone,
        address: selected.address,
        city: selected.city || "",
      };
    }

    // Compute zone fallback at submit time
    const zones = settings.deliveryZones || [];
    const zoneToSend =
      deliveryZoneId || (zones.length > 0 ? zones[0].id : null);

    console.log("📦 Submitting order:", {
      bookId,
      quantity,
      paymentMethod,
      deliveryZoneId: zoneToSend,
      zonesAvailable: zones.map((z) => z.id),
    });

    setSubmitting(true);
    try {
      const { data } = await api.post("/orders", {
        items: [{ bookId, quantity }],
        shippingAddress: addressPayload,
        notes,
        paymentMethod,
        deliveryZoneId: zoneToSend,
      });

      setOrder(data);
      await refreshCart();
      window.scrollTo({ top: 0, behavior: "smooth" });
    } catch (err) {
      console.error("Order failed:", err);
      setError(
        err.response?.data?.message ||
          "Failed to place order. Please try again."
      );
    } finally {
      setSubmitting(false);
    }
  };

    if (arr.length === 0) return;

    const valid = arr.filter((f) => {
      const isImage = f.type.startsWith("image/");
      const isPdf = f.type === "application/pdf";

    });

    if (valid.length !== arr.length) {
      setError(
        "Some files were skipped — only images or PDF under 10 MB are allowed."
      );
    } else {
      setError("");
    }

    setPaymentFiles((prev) => [...prev, ...valid].slice(0, 5));

    setPaymentFiles((prev) => prev.filter((_, i) => i !== idx));

      setError("Please add at least one file");
      return;

    setUploadingProof(true);
    setError("");

      const { data: sign } = await api.post("/upload/payment-sign");

      const uploaded = [];
      for (const file of paymentFiles) {
        const fd = new FormData();
        fd.append("file", file);
        fd.append("api_key", sign.apiKey);
        fd.append("timestamp", sign.timestamp);
        fd.append("signature", sign.signature);
        fd.append("folder", sign.folder);

        const endpoint = file.type === "application/pdf" ? "raw" : "image";
        const url = `https://api.cloudinary.com/v1_1/${sign.cloudName}/${endpoint}/upload`;

        const resp = await fetch(url, { method: "POST", body: fd });
        if (!resp.ok) throw new Error("Upload failed");
        const result = await resp.json();

        uploaded.push({
          url: result.secure_url,
          publicId: result.public_id,
          name: file.name,
        });
      }

      await api.post(`/orders/${order._id}/payment`, { files: uploaded });
      setProofUploaded(true);
    } catch (err) {
      console.error(err);
      setError("Failed to upload payment proof. Try again.");
    } finally {
      setUploadingProof(false);
    }
  };

  // ==========================================================
  // LOADING
  // ==========================================================
  if (loading || authLoading) {
    return (
      <div className="container-x py-20 flex justify-center">
        <div className="w-12 h-12 border-4 border-red-200 border-t-red-600 rounded-full animate-spin" />
      </div>
    );
  }

  // ==========================================================
  // NOT FOUND
  // ==========================================================
  if (!book) {
    return (
      <div className="container-x py-20 text-center">
        <h2 className="text-xl font-bold text-gray-900">Book not found</h2>
        <Link
          to="/books"
          className="inline-block mt-4 bg-red-600 text-white font-semibold py-2.5 px-5 rounded-xl"
        >
          Browse Books
        </Link>
      </div>
    );
  }

  // ==========================================================
  // PAYMENT SCREEN
  // ==========================================================
  if (order) {
    const isCOD = order.paymentMethod === "cod";

    return (
      <div className="container-x py-8 md:py-12 animate-fade-in">
        <div className="bg-green-50 border border-green-200 rounded-2xl p-6 mb-8 flex items-start gap-4">
          <div className="w-12 h-12 rounded-full bg-green-100 text-green-700 flex items-center justify-center flex-shrink-0">
            <svg
              className="w-6 h-6"
              fill="none"
              stroke="currentColor"
              viewBox="0 0 24 24"
            >
              <path
                strokeLinecap="round"
                strokeLinejoin="round"
                strokeWidth={2.5}
                d="M5 13l4 4L19 7"
              />
            </svg>
          </div>
          <div className="flex-1">
            <h1 className="text-2xl font-extrabold text-green-900">
              Order Placed!
            </h1>
            <p className="text-sm text-green-800 mt-1">
              Order{" "}

            {!isCOD ? (
              <>
                <div className="bg-white rounded-2xl shadow-lg border-2 border-yellow-200 p-6 mb-6">

                  <div className="bg-white border-4 border-gray-100 rounded-2xl p-3 max-w-sm mx-auto aspect-square flex items-center justify-center">

                      </p>
                    )}
                  </div>

                  <p className="text-xs text-gray-500 text-center mt-4">
                    Scan with your banking / eSewa / Khalti app to pay{" "}
                    <span className="font-bold text-gray-800">
                      NPR {Number(order.total).toLocaleString()}
                    </span>
                    .
                  </p>
                </div>

                <div className="bg-white rounded-2xl shadow-md border border-gray-100 p-6">
                  <h3 className="font-bold text-gray-900 mb-1 flex items-center gap-2">
                    <span className="w-1.5 h-5 bg-red-600 rounded-full" />
                    Upload Payment Proof
                  </h3>
                  <p className="text-xs text-gray-500 mb-4">
                    Upload a screenshot or PDF of your payment confirmation.
                  </p>

                  {proofUploaded ? (
                    <div className="bg-green-50 border border-green-200 rounded-xl p-4 flex items-center gap-3">
                      <div className="w-9 h-9 rounded-full bg-green-100 text-green-700 flex items-center justify-center flex-shrink-0">
                        <svg
                          className="w-5 h-5"
                          fill="none"
                          stroke="currentColor"
                          viewBox="0 0 24 24"

                          <path
                            strokeLinecap="round"
                            strokeLinejoin="round"

                  ) : (
                    <>
                      <label className="block border-2 border-dashed border-yellow-400 hover:border-red-500 bg-yellow-50/50 rounded-xl p-6 cursor-pointer text-center transition mb-3">
                        <input
                          ref={fileInputRef}
                          type="file"
                          accept="image/*,application/pdf"
                          multiple
                          hidden
                          onChange={(e) => handlePaymentFiles(e.target.files)}

                          className="w-10 h-10 text-yellow-600 mx-auto mb-2"

                          viewBox="0 0 24 24"
                        >
                          <path
                            strokeLinecap="round"
                            strokeLinejoin="round"
                            strokeWidth={2}
                            d="M7 16a4 4 0 01-.88-7.903A5 5 0 1115.9 6L16 6a5 5 0 011 9.9M15 13l-3-3m0 0l-3 3m3-3v12"

                        </svg>
                        <p className="text-sm font-semibold text-gray-700">
                          Click to upload files

                        <p className="text-xs text-gray-500 mt-1">
                          PNG, JPG, PDF up to 10 MB
                        </p>

                      {paymentFiles.length > 0 && (
                        <div className="space-y-2 mb-3">
                          {paymentFiles.map((f, i) => (

                              key={i}
                              className="flex items-center gap-3 bg-gray-50 rounded-lg p-2 border border-gray-100"

                              <div className="w-9 h-9 rounded-lg bg-white border border-gray-200 flex items-center justify-center flex-shrink-0 overflow-hidden">

                                  <img
                                    src={URL.createObjectURL(f)}

                                  <svg
                                    className="w-5 h-5 text-red-500"

                                    <path d="M4 4a2 2 0 012-2h4.586A2 2 0 0112 2.586L15.414 6A2 2 0 0116 7.414V16a2 2 0 01-2 2H6a2 2 0 01-2-2V4z" />
                                  </svg>

                              </div>
                              <p className="flex-1 text-sm text-gray-700 truncate">
                                {f.name}
                              </p>
                              <button
                                type="button"
                                onClick={() => removePaymentFile(i)}

                      >

                          "Submit Payment Proof"

                      </button>

              </>
            ) : (
              <div className="bg-white rounded-2xl shadow-lg border-2 border-green-200 p-6">
                <div className="w-14 h-14 rounded-full bg-green-100 text-green-700 flex items-center justify-center mb-4">
                  <svg
                    className="w-7 h-7"
                    fill="none"
                    stroke="currentColor"
                    viewBox="0 0 24 24"
                  >
                    <path
                      strokeLinecap="round"
                      strokeLinejoin="round"
                      strokeWidth={2}

              <h3 className="font-bold text-lg text-gray-900 mb-4 flex items-center gap-2">
                <span className="w-1.5 h-5 bg-green-600 rounded-full" />
                Order Summary
              </h3>

              {order.items.map((item, i) => (
                <div
                  key={i}
                  className="flex items-center gap-3 pb-3 mb-3 border-b border-gray-100 last:border-0"
                >
                  <div className="w-14 h-18 rounded-lg bg-gray-100 overflow-hidden flex-shrink-0">
                    {item.image ? (
                      <img
                        src={item.image}
                        alt={item.title}
                        className="w-full h-full object-cover"
                      />
                    ) : (
                      <div className="w-full h-full flex items-center justify-center text-gray-300">
                        <svg
                          className="w-6 h-6"
                          fill="none"
                          stroke="currentColor"
                          viewBox="0 0 24 24"
                        >
                          <path
                            strokeLinecap="round"
                            strokeLinejoin="round"
                            strokeWidth={2}
                            d="M12 6.253v13"
                          />
                        </svg>
                      </div>
                    )}
                  </div>
                  <div className="flex-1 min-w-0">
                    <p className="text-sm font-bold text-gray-900 line-clamp-2">
                      {item.title}
                    </p>
                    <p className="text-xs text-gray-500">
                      Qty: {item.quantity} × NPR{" "}
                      {Number(item.price).toLocaleString()}

                  <p className="text-sm font-bold text-green-700">
                    NPR {Number(item.price * item.quantity).toLocaleString()}

              <div className="space-y-2 pt-3 text-sm">
                <div className="flex justify-between text-gray-600">
                  <span>Subtotal</span>
                  <span className="font-semibold">
                    NPR {Number(order.subtotal).toLocaleString()}
                  </span>
                </div>
                <div className="flex justify-between text-gray-600">
                  <span>
                    Delivery
                    {order.deliveryZone?.name && (
                      <span className="block text-[10px] text-gray-400 font-medium mt-0.5">
                        {order.deliveryZone.name}
                      </span>
                    )}
                  </span>
                  <span className="font-semibold">
                    {Number(order.deliveryFee) === 0
                      ? "Free"
                      : `NPR ${Number(order.deliveryFee).toLocaleString()}`}

                </div>
                <div className="flex justify-between pt-2 border-t border-gray-100">
                  <span className="font-bold text-gray-900">Total</span>
                  <span className="font-extrabold text-green-700 text-lg">
                    NPR {Number(order.total).toLocaleString()}
                  </span>
                </div>
                <div className="pt-2 mt-2 border-t border-gray-100 flex justify-between text-xs">
                  <span className="text-gray-500">Payment Method</span>
                  <span className="font-bold text-gray-700">
                    {order.paymentMethod === "cod"
                      ? "Cash on Delivery"
                      : "QR Code"}
                  </span>
                </div>
              </div>
            </div>

            <div className="bg-white rounded-2xl shadow-md border border-gray-100 p-6 mb-6">
              <h3 className="font-bold text-lg text-gray-900 mb-3 flex items-center gap-2">
                <span className="w-1.5 h-5 bg-yellow-400 rounded-full" />
                Shipping To
              </h3>
              <p className="text-sm font-bold text-gray-900">
                {order.shippingAddress.name}
              </p>
              <p className="text-sm text-gray-600">
                {order.shippingAddress.phone}
              </p>
              <p className="text-sm text-gray-600 mt-1">
                {order.shippingAddress.address}
                {order.shippingAddress.city &&
                  `, ${order.shippingAddress.city}`}
              </p>
              {order.deliveryZone?.name && (
                <p className="text-xs text-gray-500 mt-2 inline-flex items-center gap-1 bg-green-50 border border-green-200 rounded-full px-2.5 py-1">
                  <span className="w-1.5 h-1.5 rounded-full bg-green-500" />
                  {order.deliveryZone.name}
                </p>
              )}
            </div>

            <div className="flex flex-col gap-3">
              <Link
                to="/profile?tab=orders"
                className="w-full text-center bg-gray-100 hover:bg-gray-200 text-gray-800 font-semibold py-3 rounded-xl transition"

                View My Orders
              </Link>
              <Link
                to="/books"
                className="w-full text-center bg-gradient-to-r from-red-700 to-red-600 hover:from-red-800 hover:to-red-700 text-white font-semibold py-3 rounded-xl shadow-md transition"

  // ==========================================================

      <nav className="flex items-center gap-2 text-xs text-gray-500 mb-6">
        <Link to="/" className="hover:text-red-700">

        </Link>
        <span>/</span>
        <Link to={`/books/${bookId}`} className="hover:text-red-700">
          Book Details
        </Link>
        <span>/</span>
        <span className="text-gray-800 font-medium">Checkout</span>
      </nav>

      <h1 className="text-3xl md:text-4xl font-extrabold tracking-tight text-gray-900 mb-8">
        Checkout
      </h1>

      {error && (
        <div className="bg-red-50 border-l-4 border-red-500 text-red-700 px-4 py-3 rounded-lg text-sm mb-6">
          {error}
        </div>
      )}

      <form onSubmit={handleSubmitOrder}>
        <div className="grid grid-cols-1 lg:grid-cols-12 gap-8">
          {/* LEFT */}
          <div className="lg:col-span-7 space-y-6">
            {/* QUANTITY */}
            <div className="bg-white rounded-2xl shadow-md border border-gray-100 p-6">
              <h2 className="font-bold text-lg text-gray-900 mb-4 flex items-center gap-2">
                <span className="w-1.5 h-5 bg-red-600 rounded-full" />

                    type="button"

                    −

                  <input

                    max={maxQty}
                    value={quantity}
                    onChange={(e) =>
                      setQuantity(
                        Math.max(1, Math.min(maxQty, Number(e.target.value)))
                      )
                    }
                    className="w-16 text-center font-bold text-lg bg-transparent focus:outline-none"

                  <button
                    type="button"
                    onClick={() => setQuantity((q) => Math.min(maxQty, q + 1))}
                    disabled={quantity >= maxQty}
                    className="w-10 h-10 flex items-center justify-center rounded-lg bg-white text-gray-700 shadow-sm hover:bg-green-50 hover:text-green-600 disabled:opacity-40 transition"
                  >
                    +
                  </button>
                </div>

                <div className="flex-1">
                  <p className="text-sm text-gray-500">
                    Stock available:{" "}
                    <span className="font-bold text-gray-800">{maxQty}</span>
                  </p>
                  <p className="text-sm text-gray-500 mt-0.5">
                    Unit price:{" "}
                    <span className="font-bold text-gray-800">
                      NPR {Number(book.price).toLocaleString()}
                    </span>

            {(settings.deliveryZones || []).length > 0 && (

                  Choose the area where you want your order delivered.
                </p>

                <div className="space-y-2">
                  {settings.deliveryZones.map((zone) => {

                    return (
                      <label

                        className={`flex items-start gap-3 p-4 rounded-xl border-2 cursor-pointer transition ${
                          selected
                            ? "border-red-500 bg-red-50/50"

                        <div className="flex-1 min-w-0">
                          <div className="flex items-center justify-between gap-2 flex-wrap">
                            <p className="text-sm font-bold text-gray-900">
                              {zone.name}
                            </p>
                            <span className="text-sm font-extrabold text-green-700">
                              {Number(zone.charge) === 0

                          {zone.description && (

                              {zone.description}
                            </p>

                      </label>
                    );
                  })}
                </div>

            )}

            <div className="bg-white rounded-2xl shadow-md border border-gray-100 p-6">
              <h2 className="font-bold text-lg text-gray-900 mb-4 flex items-center gap-2">
                <span className="w-1.5 h-5 bg-yellow-400 rounded-full" />
                Payment Method
              </h2>

              <div className="space-y-3">
                {settings.paymentMethods?.qr && (
                  <label
                    className={`flex items-start gap-3 p-4 rounded-xl border-2 cursor-pointer transition ${
                      paymentMethod === "qr"
                        ? "border-red-500 bg-red-50/50"
                        : "border-gray-200 hover:border-yellow-400"
                    }`}
                  >
                    <input
                      type="radio"
                      name="paymentMethod"
                      checked={paymentMethod === "qr"}
                      onChange={() => setPaymentMethod("qr")}
                      className="mt-1 w-4 h-4 text-red-600 focus:ring-red-500"
                    />
                    <div className="flex-1">
                      <p className="text-sm font-bold text-gray-900 flex items-center gap-2">
                        QR Code Payment
                        <span className="text-[10px] font-bold uppercase bg-green-100 text-green-700 px-2 py-0.5 rounded-full">
                          Recommended
                        </span>
                      </p>
                      <p className="text-xs text-gray-500 mt-1">

                        proof.
                      </p>
                    </div>
                  </label>
                )}

                {settings.paymentMethods?.cod && (
                  <label
                    className={`flex items-start gap-3 p-4 rounded-xl border-2 cursor-pointer transition ${
                      paymentMethod === "cod"
                        ? "border-red-500 bg-red-50/50"
                        : "border-gray-200 hover:border-yellow-400"
                    }`}
                  >
                    <input
                      type="radio"
                      name="paymentMethod"
                      checked={paymentMethod === "cod"}

                      className="mt-1 w-4 h-4 text-red-600 focus:ring-red-500"
                    />
                    <div className="flex-1">
                      <p className="text-sm font-bold text-gray-900">
                        Cash on Delivery
                      </p>
                      <p className="text-xs text-gray-500 mt-1">
                        Pay in cash when your order is delivered.
                      </p>
                    </div>
                  </label>
                )}
              </div>
            </div>

            {/* ADDRESS */}
            <div className="bg-white rounded-2xl shadow-md border border-gray-100 p-6">
              <div className="flex items-center justify-between mb-4">
                <h2 className="font-bold text-lg text-gray-900 flex items-center gap-2">
                  <span className="w-1.5 h-5 bg-green-600 rounded-full" />
                  Shipping Address
                </h2>
                {addresses.length > 0 && !showAddressForm && (
                  <button
                    type="button"

                    >

                      />
                      <div className="flex-1">
                        <div className="flex items-center gap-2 flex-wrap">
                          <span className="text-sm font-bold text-gray-900">

                            {a.label}
                          </span>
                          {a.isDefault && (
                            <span className="text-[10px] font-bold uppercase bg-red-100 text-red-700 px-2 py-0.5 rounded-full">
                              Default
                            </span>
                          )}
                        </div>
                        <p className="text-xs text-gray-600 mt-1">{a.phone}</p>
                        <p className="text-xs text-gray-600 mt-0.5">
                          {a.address}
                          {a.city && `, ${a.city}`}
                        </p>
                      </div>
                    </label>
                  ))}
                </div>
              )}

              {(showAddressForm || addresses.length === 0) && (
                <div className="space-y-4">
                  {addresses.length > 0 && (
                    <button
                      type="button"
                      onClick={() => setShowAddressForm(false)}
                      className="text-xs font-semibold text-gray-500 hover:text-red-700"

                      ← Back to saved addresses
                    </button>
                  )}

                  <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                    <Field
                      label="Label"
                      value={newAddress.label}
                      onChange={(v) =>
                        setNewAddress({ ...newAddress, label: v })
                      }
                      placeholder="Home / Office"
                    />
                    <Field
                      label="Full Name *"
                      value={newAddress.name}
                      onChange={(v) =>
                        setNewAddress({ ...newAddress, name: v })
                      }
                      placeholder="John Doe"

                    />
                    <Field
                      label="Phone *"
                      value={newAddress.phone}
                      onChange={(v) =>
                        setNewAddress({ ...newAddress, phone: v })

                      placeholder="+977-98XXXXXXXX"
                      required
                    />
                    <Field
                      label="City"
                      value={newAddress.city}
                      onChange={(v) =>
                        setNewAddress({ ...newAddress, city: v })
                      }
                      placeholder="Halesi"
                    />
                  </div>

                  <div>
                    <label className="block text-sm font-semibold text-gray-700 mb-1.5">
                      Full Address *
                    </label>
                    <textarea
                      rows={3}
                      required
                      value={newAddress.address}
                      onChange={(e) =>
                        setNewAddress({
                          ...newAddress,
                          address: e.target.value,
                        })
                      }
                      className="w-full px-4 py-2.5 rounded-xl border-2 border-yellow-300 bg-yellow-50/50 focus:outline-none focus:border-red-500 focus:ring-2 focus:ring-red-200 focus:bg-white transition resize-none text-sm"
                      placeholder="Street, ward, landmark..."
                    />
                  </div>

                  <label className="flex items-start gap-2 text-sm text-gray-700 cursor-pointer">
                    <input
                      type="checkbox"
                      checked={saveAddressToProfile}
                      onChange={(e) =>
                        setSaveAddressToProfile(e.target.checked)
                      }
                      className="mt-0.5 w-4 h-4 rounded text-red-600 focus:ring-red-500"
                    />
                    <span>
                      Save this address to my profile for future orders
                    </span>
                  </label>
                </div>

            </div>

            <div className="bg-white rounded-2xl shadow-md border border-gray-100 p-6">
              <h2 className="font-bold text-lg text-gray-900 mb-3 flex items-center gap-2">

                Order Notes (optional)
              </h2>
              <textarea
                rows={3}
                value={notes}
                onChange={(e) => setNotes(e.target.value)}
                className="w-full px-4 py-2.5 rounded-xl border-2 border-yellow-300 bg-yellow-50/50 focus:outline-none focus:border-red-500 focus:ring-2 focus:ring-red-200 focus:bg-white transition resize-none text-sm"
                placeholder="Any special instructions for delivery..."
              />
            </div>
          </div>

          {/* RIGHT: SUMMARY */}

// client/src/pages/Login.jsx
import { useState } from "react";
import { Link, useNavigate, useSearchParams } from "react-router-dom";
import { useAuth } from "../context/AuthContext";

export default function Login() {
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [showPassword, setShowPassword] = useState(false);
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);
  const { login } = useAuth();
  const navigate = useNavigate();
  const [params] = useSearchParams();
  const redirect = params.get("redirect") || "/";

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError("");
    setLoading(true);
    try {
      await login(email, password);
      navigate(redirect, { replace: true });
    } catch (err) {
      setError(
        err.response?.data?.message ||
          "Login failed. Please check your credentials."
      );
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="min-h-[80vh] flex items-center justify-center px-4 py-12 bg-gradient-to-br from-red-50 via-yellow-50 to-green-50">
      <div className="w-full max-w-md">
        <div className="text-center mb-8">
          <div className="w-16 h-16 mx-auto mb-4 bg-gradient-to-br from-red-600 to-red-800 rounded-2xl flex items-center justify-center shadow-lg">
            <svg
              className="w-8 h-8 text-white"
              fill="none"
              stroke="currentColor"
              viewBox="0 0 24 24"
            >
              <path
                strokeLinecap="round"
                strokeLinejoin="round"
                strokeWidth={2}
                d="M16 7a4 4 0 11-8 0 4 4 0 018 0zM12 14a7 7 0 00-7 7h14a7 7 0 00-7-7z"
              />
            </svg>
          </div>
          <h1 className="text-2xl font-extrabold text-gray-900">
            Welcome Back
          </h1>
          <p className="text-sm text-gray-500 mt-1">
            Sign in to continue to Halesi Books Centre
          </p>
        </div>

        <div className="bg-white rounded-2xl shadow-xl p-7 border-t-4 border-t-red-600 border-b-4 border-b-yellow-400">
          <form onSubmit={handleSubmit} className="space-y-5">
            {error && (
              <div className="bg-red-50 border-l-4 border-red-500 text-red-700 px-4 py-3 rounded-lg text-sm">
                {error}
              </div>
            )}

            <div>
              <label className="block text-sm font-semibold text-gray-700 mb-1.5">
                Email
              </label>
              <input
                type="email"
                required
                value={email}
                onChange={(e) => setEmail(e.target.value)}
                className="w-full px-4 py-2.5 rounded-xl border-2 border-yellow-300 bg-yellow-50/50 focus:outline-none focus:border-red-500 focus:ring-2 focus:ring-red-200 focus:bg-white transition"
                placeholder="you@example.com"
              />
            </div>

            <div>
              <label className="block text-sm font-semibold text-gray-700 mb-1.5">
                Password
              </label>
              <div className="relative">
                <input
                  type={showPassword ? "text" : "password"}
                  required
                  value={password}
                  onChange={(e) => setPassword(e.target.value)}
                  className="w-full px-4 py-2.5 pr-12 rounded-xl border-2 border-yellow-300 bg-yellow-50/50 focus:outline-none focus:border-green-600 focus:ring-2 focus:ring-green-200 focus:bg-white transition"
                  placeholder="••••••••"
                />
                <button
                  type="button"
                  onClick={() => setShowPassword((s) => !s)}
                  className="absolute inset-y-0 right-0 px-3 text-gray-500 hover:text-green-700"

                >
                  {showPassword ? "Hide" : "Show"}

            <button

              disabled={loading}
              className="w-full flex items-center justify-center gap-2 bg-gradient-to-r from-red-700 to-red-600 hover:from-red-800 hover:to-red-700 text-white font-semibold py-3 rounded-xl shadow-lg shadow-red-500/30 hover:shadow-red-500/50 transition disabled:opacity-60"

// =================== CART ===================
function CartTab({ items, total, onRefresh, showToast }) {
  const [busy, setBusy] = useState(null);
  const navigate = useNavigate();

  const updateQty = async (bookId, quantity) => {
    if (quantity < 1) return;
    setBusy(bookId);
    try {
      await api.put("/cart", { bookId, quantity });
      await onRefresh();
    } catch {
      showToast("Failed to update");
    } finally {
      setBusy(null);
    }
  };

  const remove = async (bookId) => {
    if (!confirm("Remove this item from cart?")) return;
    setBusy(bookId);
    try {
      await api.delete("/cart", { data: { bookId } });
      showToast("Removed from cart");
      await onRefresh();
    } catch {
      showToast("Failed to remove");
    } finally {
      setBusy(null);
    }
  };

  if (items.length === 0) {
    return (
      <div className="bg-white rounded-2xl shadow-sm border border-gray-100 p-12">
        <EmptyState
          icon="M3 3h2l.4 2M7 13h10l4-8H5.4M7 13L5.4 5M7 13l-2.293 2.293c-.63.63-.184 1.707.707 1.707H17m0 0a2 2 0 100 4 2 2 0 000-4zm-8 2a2 2 0 11-4 0 2 2 0 014 0z"
          title="Your cart is empty"
          desc="Add books to your cart to see them here."
          cta={{ to: "/books", label: "Browse Books" }}
        />
      </div>
    );
  }

  return (
    <div className="space-y-4">
      <h2 className="text-2xl font-extrabold text-gray-900">My Cart</h2>

      <div className="bg-white rounded-2xl shadow-sm border border-gray-100 divide-y divide-gray-100">
        {items.map((item) => (
          <div key={item.bookId} className="p-4 flex items-center gap-4">
            <Link
              to={`/books/${item.bookId}`}
              className="w-16 h-20 rounded-lg overflow-hidden bg-gray-100 flex-shrink-0"
            >
              {item.image ? (
                <img
                  src={item.image}
                  alt={item.title}
                  className="w-full h-full object-cover"
                />
              ) : (
                <div className="w-full h-full flex items-center justify-center text-gray-400">
                  <svg
                    className="w-6 h-6"
                    fill="none"
                    stroke="currentColor"
                    viewBox="0 0 24 24"
                  >
                    <path
                      strokeLinecap="round"
                      strokeLinejoin="round"
                      strokeWidth={2}
                      d="M12 6.253v13m0-13C10.832 5.477 9.246 5 7.5 5S4.168 5.477 3 6.253v13C4.168 18.477 5.754 18 7.5 18s3.332.477 4.5 1.253"
                    />
                  </svg>
                </div>
              )}
            </Link>

            <div className="flex-1 min-w-0">
              <Link
                to={`/books/${item.bookId}`}
                className="text-sm font-bold text-gray-900 hover:text-red-700 line-clamp-2 transition"
              >
                {item.title}
              </Link>
              <p className="text-xs text-gray-500 mt-0.5">by {item.author}</p>
              <p className="text-base font-extrabold text-green-700 mt-1">
                NPR {Number(item.price).toLocaleString()}
              </p>
            </div>

            <div className="flex flex-col items-end gap-2 flex-shrink-0">
              <div className="flex items-center gap-1 bg-gray-100 rounded-lg">
                <button
                  onClick={() => updateQty(item.bookId, item.quantity - 1)}
                  disabled={item.quantity <= 1 || busy === item.bookId}
                  className="w-8 h-8 flex items-center justify-center text-gray-600 hover:bg-gray-200 rounded-l-lg disabled:opacity-40"
                >
                  −
                </button>
                <span className="w-8 text-center text-sm font-bold text-gray-800">
                  {busy === item.bookId ? "…" : item.quantity}
                </span>
                <button
                  onClick={() => updateQty(item.bookId, item.quantity + 1)}
                  disabled={busy === item.bookId || item.quantity >= item.stock}
                  className="w-8 h-8 flex items-center justify-center text-gray-600 hover:bg-gray-200 rounded-r-lg disabled:opacity-40"
                >
                  +
                </button>
              </div>

              <button
                onClick={() => remove(item.bookId)}
                disabled={busy === item.bookId}
                className="text-xs text-red-600 hover:text-red-800 font-semibold"
              >
                Remove
              </button>
            </div>
          </div>
        ))}

        {/* Summary */}
        <div className="p-5 bg-gradient-to-br from-red-50 to-yellow-50">
          <div className="flex items-center justify-between mb-4">
            <span className="text-sm font-semibold text-gray-700">
              Subtotal
            </span>
            <span className="text-2xl font-extrabold text-green-700">
              NPR {Number(total).toLocaleString()}
            </span>
          </div>
          <button
            onClick={() => navigate("/checkout")}
            className="w-full bg-gradient-to-r from-red-700 to-red-600 hover:from-red-800 hover:to-red-700 text-white font-bold py-3.5 rounded-xl shadow-lg shadow-red-500/30 hover:shadow-red-500/50 transition"
          >
            Proceed to Checkout
          </button>
        </div>
      </div>
    </div>
  );
}

// =================== SECURITY ===================
function SecurityTab({ showToast }) {
  const [form, setForm] = useState({
    currentPassword: "",
    newPassword: "",
    confirmPassword: "",
  });
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError("");

    if (form.newPassword !== form.confirmPassword) {
      setError("New passwords do not match");
      return;
    }
    if (form.newPassword.length < 6) {
      setError("New password must be at least 6 characters");
      return;
    }

    setSaving(true);
    try {
      await api.put("/user/password", {
        currentPassword: form.currentPassword,
        newPassword: form.newPassword,
      });
      showToast("Password changed");
      setForm({ currentPassword: "", newPassword: "", confirmPassword: "" });
    } catch (err) {
      setError(err.response?.data?.message || "Failed to change password");
    } finally {
      setSaving(false);
    }
  };

  return (
    <div className="space-y-6">
      <h2 className="text-2xl font-extrabold text-gray-900">Security</h2>

      <div className="bg-white rounded-2xl shadow-sm border border-gray-100 p-6 max-w-lg">
        <h3 className="font-bold text-gray-900 mb-4">Change Password</h3>

        {error && (
          <div className="mb-4 bg-red-50 border-l-4 border-red-500 text-red-700 px-3 py-2 rounded text-sm">
            {error}
          </div>
        )}

        <form onSubmit={handleSubmit} className="space-y-4">
          <FormInput
            label="Current Password"
            type="password"
            value={form.currentPassword}
            onChange={(v) => setForm({ ...form, currentPassword: v })}
            required
          />
          <FormInput
            label="New Password"
            type="password"
            value={form.newPassword}
            onChange={(v) => setForm({ ...form, newPassword: v })}
            placeholder="At least 6 characters"
            required
          />
          <FormInput
            label="Confirm New Password"
            type="password"
            value={form.confirmPassword}
            onChange={(v) => setForm({ ...form, confirmPassword: v })}
            required
          />

          <button
            type="submit"
            disabled={saving}
            className="bg-gradient-to-r from-red-700 to-red-600 hover:from-red-800 hover:to-red-700 text-white font-semibold px-6 py-2.5 rounded-xl shadow-md transition disabled:opacity-60"
          >
            {saving ? "Updating..." : "Update Password"}
          </button>
        </form>
      </div>
    </div>
  );
}

function getStatusStyle(status) {
  switch (status) {
    case "pending":
      return {
        label: "Pending",
        cls: "bg-yellow-100 text-yellow-800 border-yellow-200",
      };
    case "confirmed":
      return {
        label: "Confirmed",
        cls: "bg-blue-100 text-blue-800 border-blue-200",
      };
    case "shipped":
      return {
        label: "Shipped",
        cls: "bg-purple-100 text-purple-800 border-purple-200",
      };
    case "delivered":
      return {
        label: "Delivered",
        cls: "bg-green-100 text-green-800 border-green-200",
      };
    case "cancelled":
      return {
        label: "Cancelled",
        cls: "bg-red-100 text-red-800 border-red-200",
      };
    default:
      return {
        label: status || "Unknown",
        cls: "bg-gray-100 text-gray-700 border-gray-200",
      };
  }
}

// client/src/utils/api.js
import axios from "axios";

const baseURL =
  import.meta.env.VITE_API_BASE_URL ||
  `http://${window.location.hostname}:5000/api`;

const api = axios.create({
  baseURL,
  headers: { "Content-Type": "application/json" },
});

// Attach user token to every request
api.interceptors.request.use((config) => {
  const token = localStorage.getItem("hbc_user_token");
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

export default api;
