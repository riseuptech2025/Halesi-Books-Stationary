// frontend/src/components/BookFormModal.jsx
import { useEffect, useState } from "react";
import ImageUploader from "./ImageUploader";
import RichTextEditor from "./RichTextEditor";

const emptyForm = {
  title: "",
  author: "",
  isbn: "",
  price: "",
  stock: "",
  category: "General",
  images: [],
  publishedYear: "",
  edition: "",
  language: "",
  colorFormat: "",
  description: "",
};

const LANGUAGES = [
  "English",
  "Nepali",
  "Hindi",
  "Sanskrit",
  "Tibetan",
  "Other",
];

const FORMATS = [
  "Paperback",
  "Hardcover",
  "E-Book",
  "Audiobook",
  "Spiral",
  "Other",
];

export default function BookFormModal({ book, categories, onSave, onClose }) {
  const [form, setForm] = useState(emptyForm);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  const isEditing = !!book;

  useEffect(() => {
    if (book) {
      setForm({
        title: book.title || "",
        author: book.author || "",
        isbn: book.isbn || "",
        price: book.price ?? "",
        stock: book.stock ?? "",
        category: book.category || "General",
        images: Array.isArray(book.images) ? book.images : [],
        publishedYear: book.publishedYear ?? "",
        edition: book.edition || "",
        language: book.language || "",
        colorFormat: book.colorFormat || "",
        description: book.description || "",
      });
    } else {
      setForm(emptyForm);
    }
  }, [book]);

  const handleChange = (e) => {
    setForm((prev) => ({ ...prev, [e.target.name]: e.target.value }));
  };

  const handleImagesChange = (images) => {
    setForm((prev) => ({ ...prev, images }));
  };

  const handleDescriptionChange = (html) => {
    setForm((prev) => ({ ...prev, description: html }));
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError("");

    if (!form.title.trim() || !form.author.trim()) {
      setError("Title and author are required.");
      return;
    }
    if (form.price === "" || Number(form.price) < 0) {
      setError("Please enter a valid price.");
      return;
    }
    if (form.stock === "" || Number(form.stock) < 0) {
      setError("Please enter a valid stock quantity.");
      return;
    }

    setSaving(true);
    try {
      await onSave({
        title: form.title.trim(),
        author: form.author.trim(),
        isbn: form.isbn.trim(),
        price: Number(form.price),
        stock: Number(form.stock),
        category: form.category,
        images: form.images,
        publishedYear: form.publishedYear ? Number(form.publishedYear) : null,
        edition: form.edition.trim(),
        language: form.language.trim(),
        colorFormat: form.colorFormat.trim(),
        description: form.description,
      });
      onClose();
    } catch (err) {
      setError(
        err.response?.data?.message || "Failed to save book. Try again."
      );
    } finally {
      setSaving(false);
    }
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/40 backdrop-blur-sm animate-fade-in">
      <div
        className="bg-white rounded-2xl shadow-2xl w-full max-w-3xl border-t-4 border-t-red-600 border-b-4 border-b-yellow-400 max-h-[92vh] overflow-y-auto"
        onClick={(e) => e.stopPropagation()}
      >
        {/* Header */}
        <div className="flex items-center justify-between px-6 py-4 border-b border-gray-100 sticky top-0 bg-white z-10">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 rounded-xl bg-red-100 text-red-600 flex items-center justify-center">
              <svg
                className="w-5 h-5"
                fill="none"
                stroke="currentColor"
                viewBox="0 0 24 24"
              >
                <path
                  strokeLinecap="round"
                  strokeLinejoin="round"
                  strokeWidth={2}
                  d={
                    isEditing
                      ? "M11 5H6a2 2 0 00-2 2v11a2 2 0 002 2h11a2 2 0 002-2v-5m-1.414-9.414a2 2 0 112.828 2.828L11.828 15H9v-2.828l8.586-8.586z"
                      : "M12 4v16m8-8H4"
                  }
                />
              </svg>
            </div>
            <div>
              <h2 className="text-lg font-bold text-gray-900">
                {isEditing ? "Edit Book" : "Add New Book"}
              </h2>
              <p className="text-xs text-gray-500">
                {isEditing
                  ? "Update the book details below."
                  : "Fill in the book details below."}
              </p>
            </div>
          </div>
          <button
            onClick={onClose}
            className="w-9 h-9 rounded-lg hover:bg-gray-100 flex items-center justify-center text-gray-500 hover:text-gray-800 transition"
            aria-label="Close"
          >
            <svg
              className="w-5 h-5"
              fill="none"
              stroke="currentColor"
              viewBox="0 0 24 24"
            >
              <path
                strokeLinecap="round"
                strokeLinejoin="round"
                strokeWidth={2}
                d="M6 18L18 6M6 6l12 12"
              />
            </svg>
          </button>
        </div>

        {/* Form */}
        <form onSubmit={handleSubmit} className="p-6 space-y-5">
          {error && (
            <div className="flex items-start gap-2 bg-red-50 border-l-4 border-red-500 text-red-700 px-4 py-3 rounded-lg text-sm">
              <svg
                className="w-5 h-5 flex-shrink-0 mt-0.5"
                fill="currentColor"
                viewBox="0 0 20 20"
              >
                <path
                  fillRule="evenodd"
                  d="M18 10A8 8 0 11 2 10a8 8 0 0116 0zm-7 4a1 1 0 11-2 0 1 1 0 012 0zm-1-9a1 1 0 00-1 1v4a1 1 0 102 0V6a1 1 0 00-1-1z"
                  clipRule="evenodd"
                />
              </svg>
              <span>{error}</span>
            </div>
          )}

          {/* ---- BASIC INFO ---- */}
          <SectionLabel>Basic Information</SectionLabel>

          <Field label="Title *" accent="red">
            <input
              type="text"
              name="title"
              value={form.title}
              onChange={handleChange}
              required
              placeholder="e.g. Wings of Fire"
              className="input-base border-red-focus"
            />
          </Field>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <Field label="Author *" accent="red">
              <input
                type="text"
                name="author"
                value={form.author}
                onChange={handleChange}
                required
                placeholder="e.g. A.P.J. Abdul Kalam"
                className="input-base border-red-focus"
              />
            </Field>

            <Field label="ISBN" accent="red">
              <input
                type="text"
                name="isbn"
                value={form.isbn}
                onChange={handleChange}
                placeholder="978-81-7371-146-6"
                className="input-base border-red-focus"
              />
            </Field>
          </div>

          {/* ---- PUBLISHING DETAILS ---- */}
          <SectionLabel>Publishing Details</SectionLabel>

          <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
            <Field label="Published Year" accent="green">
              <input
                type="number"
                name="publishedYear"
                value={form.publishedYear}
                onChange={handleChange}
                placeholder="2024"
                min="1500"
                max="2100"
                className="input-base border-green-focus"
              />
            </Field>

            <Field label="Edition" accent="green">
              <input
                type="text"
                name="edition"
                value={form.edition}
                onChange={handleChange}
                placeholder="1st, 2nd..."
                className="input-base border-green-focus"
              />
            </Field>

            <Field label="Language" accent="green">
              <select
                name="language"
                value={form.language}
                onChange={handleChange}
                className="input-base border-green-focus"
              >
                <option value="">Select...</option>
                {LANGUAGES.map((l) => (
                  <option key={l} value={l}>
                    {l}
                  </option>
                ))}
              </select>
            </Field>

            <Field label="Format" accent="green">
              <select
                name="colorFormat"
                value={form.colorFormat}
                onChange={handleChange}
                className="input-base border-green-focus"
              >
                <option value="">Select...</option>
                {FORMATS.map((f) => (
                  <option key={f} value={f}>
                    {f}
                  </option>
                ))}
              </select>
            </Field>
          </div>

          {/* ---- PRICING & STOCK ---- */}
          <SectionLabel>Pricing & Stock</SectionLabel>

          <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
            <Field label="Price (NPR) *" accent="green">
              <input
                type="number"
                name="price"
                value={form.price}
                onChange={handleChange}
                required
                min="0"
                step="0.01"
                placeholder="0.00"
                className="input-base border-green-focus"
              />
            </Field>

            <Field label="Stock *" accent="green">
              <input
                type="number"
                name="stock"
                value={form.stock}
                onChange={handleChange}
                required
                min="0"
                step="1"
                placeholder="0"
                className="input-base border-green-focus"
              />
            </Field>

            <Field label="Category" accent="yellow" className="col-span-2">
              <select
                name="category"
                value={form.category}
                onChange={handleChange}
                className="input-base border-red-focus"
              >
                {categories.map((c) => (
                  <option key={c} value={c}>
                    {c}
                  </option>
                ))}
              </select>
            </Field>
          </div>

          {/* ---- IMAGES ---- */}
          <SectionLabel>Book Images</SectionLabel>
          <ImageUploader images={form.images} onChange={handleImagesChange} />

          {/* ---- DESCRIPTION ---- */}
          <SectionLabel>Description</SectionLabel>
          <RichTextEditor
            value={form.description}
            onChange={handleDescriptionChange}
            placeholder="Write a detailed description — use headings, bullet points, links, and more."
          />
          <p className="text-[11px] text-gray-400 -mt-3">
            Tip: use the toolbar for headings, bullet lists, links, quotes, and more.
          </p>

          {/* ---- ACTIONS ---- */}
          <div className="flex gap-3 pt-3 sticky bottom-0 bg-white border-t border-gray-100 -mx-6 px-6 py-4">
            <button
              type="button"
              onClick={onClose}
              disabled={saving}
              className="flex-1 py-2.5 rounded-xl border-2 border-gray-200 text-gray-700 font-semibold hover:bg-gray-50 transition disabled:opacity-60"
            >
              Cancel
            </button>
            <button
              type="submit"
              disabled={saving}
              className="flex-1 flex items-center justify-center gap-2 bg-gradient-to-r from-red-700 to-red-600 hover:from-red-800 hover:to-red-700 text-white font-semibold py-2.5 rounded-xl shadow-lg shadow-red-500/30 hover:shadow-red-500/50 transition disabled:opacity-60 disabled:cursor-not-allowed"
            >
              {saving ? (
                <>
                  <div className="w-4 h-4 border-2 border-white/40 border-t-white rounded-full animate-spin" />
                  Saving...
                </>
              ) : (
                <>
                  <svg
                    className="w-4 h-4"
                    fill="none"
                    stroke="currentColor"
                    viewBox="0 0 24 24"
                  >
                    <path
                      strokeLinecap="round"
                      strokeLinejoin="round"
                      strokeWidth={2}
                      d="M5 13l4 4L19 7"
                    />
                  </svg>
                  {isEditing ? "Update Book" : "Add Book"}
                </>
              )}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

// ---------- helpers ----------

function SectionLabel({ children }) {
  return (
    <div className="flex items-center gap-2 pt-2">
      <span className="w-1.5 h-4 bg-red-600 rounded-full" />
      <h3 className="text-sm font-bold text-gray-800 uppercase tracking-wide">
        {children}
      </h3>
      <span className="flex-1 h-px bg-gray-100" />
    </div>
  );
}

function Field({ label, accent = "red", className = "", children }) {
  const dotColor = {
    red: "text-red-500",
    green: "text-green-600",
    yellow: "text-yellow-500",
  }[accent];

  return (
    <div className={className}>
      <label className="block text-sm font-semibold text-gray-700 mb-1.5">
        <span className={`${dotColor} mr-1`}>●</span>
        {label}
      </label>
      {children}
    </div>
  );
}










// src/components/ImageUploader.jsx
import { useRef, useState } from "react";
import api from "../utils/api";

const MAX_IMAGES = 3;
const MAX_SIZE_MB = 5;

export default function ImageUploader({ images = [], onChange }) {
  const inputRef = useRef(null);
  const [uploading, setUploading] = useState(false);
  const [error, setError] = useState("");

  const openPicker = () => {
    if (images.length >= MAX_IMAGES) return;
    inputRef.current?.click();
  };

  const handleFiles = async (e) => {
    const files = Array.from(e.target.files || []);
    if (files.length === 0) return;

    setError("");

    const remaining = MAX_IMAGES - images.length;
    const toUpload = files.slice(0, remaining);

    // Validate
    for (const f of toUpload) {
      if (!f.type.startsWith("image/")) {
        setError("Only image files are allowed.");
        return;
      }
      if (f.size > MAX_SIZE_MB * 1024 * 1024) {
        setError(`Each image must be under ${MAX_SIZE_MB} MB.`);
        return;
      }
    }

    setUploading(true);
    try {
      // 1) Get signature from our backend
      const { data: sign } = await api.post("/upload/sign");

      // 2) Upload each file directly to Cloudinary
      const uploaded = [];
      for (const file of toUpload) {
        const formData = new FormData();
        formData.append("file", file);
        formData.append("api_key", sign.apiKey);
        formData.append("timestamp", sign.timestamp);
        formData.append("signature", sign.signature);
        formData.append("folder", sign.folder);

        const resp = await fetch(sign.uploadUrl, {
          method: "POST",
          body: formData,
        });

        if (!resp.ok) {
          throw new Error("Upload to Cloudinary failed");
        }

        const result = await resp.json();
        uploaded.push({
          url: result.secure_url,
          publicId: result.public_id,
        });
      }

      onChange([...images, ...uploaded].slice(0, MAX_IMAGES));
    } catch (err) {
      console.error(err);
      setError(
        err.response?.data?.message || "Failed to upload images. Try again."
      );
    } finally {
      setUploading(false);
      if (inputRef.current) inputRef.current.value = "";
    }
  };

  const removeImage = (publicId) => {
    onChange(images.filter((img) => img.publicId !== publicId));
  };

  return (
    <div>
      <label className="block text-sm font-semibold text-gray-700 mb-1.5">
        <span className="text-yellow-500 mr-1">●</span> Book Images{" "}
        <span className="text-xs font-normal text-gray-400">
          (up to {MAX_IMAGES} images, {MAX_SIZE_MB} MB each)
        </span>
      </label>

      {/* Image grid */}
      <div className="grid grid-cols-3 gap-3">
        {images.map((img, idx) => (
          <div
            key={img.publicId || idx}
            className="relative group aspect-square rounded-xl overflow-hidden border-2 border-yellow-300 bg-yellow-50"
          >
            <img
              src={img.url}
              alt={`Book ${idx + 1}`}
              className="w-full h-full object-cover"
            />
            {/* Primary badge */}
            {idx === 0 && (
              <span className="absolute top-1.5 left-1.5 bg-red-600 text-white text-[10px] font-bold uppercase px-1.5 py-0.5 rounded">
                Cover
              </span>
            )}
            {/* Remove button */}
            <button
              type="button"
              onClick={() => removeImage(img.publicId)}
              className="absolute top-1.5 right-1.5 w-6 h-6 rounded-full bg-red-600 hover:bg-red-700 text-white flex items-center justify-center opacity-0 group-hover:opacity-100 transition shadow-lg"
              aria-label="Remove image"
            >
              <svg
                className="w-3.5 h-3.5"
                fill="none"
                stroke="currentColor"
                viewBox="0 0 24 24"
              >
                <path
                  strokeLinecap="round"
                  strokeLinejoin="round"
                  strokeWidth={2.5}
                  d="M6 18L18 6M6 6l12 12"
                />
              </svg>
            </button>
          </div>
        ))}

        {/* Add tile */}
        {images.length < MAX_IMAGES && (
          <button
            type="button"
            onClick={openPicker}
            disabled={uploading}
            className="aspect-square rounded-xl border-2 border-dashed border-yellow-400 hover:border-red-500 bg-yellow-50/50 hover:bg-red-50 flex flex-col items-center justify-center gap-1 text-yellow-700 hover:text-red-600 transition disabled:opacity-60 disabled:cursor-not-allowed"
          >
            {uploading ? (
              <>
                <div className="w-6 h-6 border-2 border-red-200 border-t-red-600 rounded-full animate-spin" />
                <span className="text-[10px] font-semibold">Uploading</span>
              </>
            ) : (
              <>
                <svg
                  className="w-7 h-7"
                  fill="none"
                  stroke="currentColor"
                  viewBox="0 0 24 24"
                >
                  <path
                    strokeLinecap="round"
                    strokeLinejoin="round"
                    strokeWidth={2}
                    d="M12 4v16m8-8H4"
                  />
                </svg>
                <span className="text-[10px] font-semibold">Add Image</span>
              </>
            )}
          </button>
        )}
      </div>

      <input
        ref={inputRef}
        type="file"
        accept="image/*"
        multiple
        hidden
        onChange={handleFiles}
      />

      {error && (
        <p className="text-xs text-red-600 mt-2 font-medium">{error}</p>
      )}

      {images.length > 0 && (
        <p className="text-[11px] text-gray-400 mt-2">
          First image is used as the book cover.
        </p>
      )}
    </div>
  );
}

import { Navigate } from "react-router-dom";
import { useAuth } from "../context/AuthContext";

export default function ProtectedRoute({ children }) {
  const { isAuthenticated, loading } = useAuth();

  if (loading) {
    return (
      <div className="min-h-screen flex items-center justify-center">
        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-primary-600"></div>
      </div>
    );
  }

  if (!isAuthenticated) {
    return <Navigate to="/login" replace />;
  }

  return children;
}














// frontend/src/components/RichTextEditor.jsx
import { useEditor, EditorContent } from "@tiptap/react";
import StarterKit from "@tiptap/starter-kit";
import Link from "@tiptap/extension-link";
import Underline from "@tiptap/extension-underline";
import Placeholder from "@tiptap/extension-placeholder";
import { useEffect } from "react";

export default function RichTextEditor({ value, onChange, placeholder }) {
  const editor = useEditor({
    extensions: [
      StarterKit.configure({
        heading: { levels: [1, 2, 3] },
      }),
      Underline,
      Link.configure({
        openOnClick: false,
        autolink: true,
        HTMLAttributes: {
          rel: "noopener noreferrer",
          target: "_blank",
          class: "text-red-700 underline hover:text-red-800",
        },
      }),
      Placeholder.configure({
        placeholder: placeholder || "Write a description...",
      }),
    ],
    content: value || "",
    onUpdate: ({ editor }) => {
      onChange(editor.getHTML());
    },
    editorProps: {
      attributes: {
        class:
          "prose prose-sm max-w-none focus:outline-none min-h-[180px] p-4 text-gray-800",
      },
    },
  });

  // Sync external value changes (e.g. when editing a different book)
  useEffect(() => {
    if (!editor) return;
    if (value !== editor.getHTML()) {
      editor.commands.setContent(value || "", false);
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [value]);

  if (!editor) return null;

  return (
    <div className="border-2 border-yellow-300 rounded-xl bg-white overflow-hidden focus-within:border-red-500 focus-within:ring-2 focus-within:ring-red-200 transition">
      {/* Toolbar */}
      <div className="flex flex-wrap items-center gap-1 p-2 border-b border-yellow-200 bg-yellow-50/60">
        <ToolbarButton
          onClick={() => editor.chain().focus().toggleBold().run()}
          active={editor.isActive("bold")}
          title="Bold"
        >
          <span className="font-bold">B</span>
        </ToolbarButton>

        <ToolbarButton
          onClick={() => editor.chain().focus().toggleItalic().run()}
          active={editor.isActive("italic")}
          title="Italic"
        >
          <span className="italic">I</span>
        </ToolbarButton>

        <ToolbarButton
          onClick={() => editor.chain().focus().toggleUnderline().run()}
          active={editor.isActive("underline")}
          title="Underline"
        >
          <span className="underline">U</span>
        </ToolbarButton>

        <ToolbarButton
          onClick={() => editor.chain().focus().toggleStrike().run()}
          active={editor.isActive("strike")}
          title="Strikethrough"
        >
          <span className="line-through">S</span>
        </ToolbarButton>

        <Divider />

        <ToolbarButton
          onClick={() =>
            editor.chain().focus().toggleHeading({ level: 1 }).run()
          }
          active={editor.isActive("heading", { level: 1 })}
          title="Heading 1"
        >
          H1
        </ToolbarButton>
        <ToolbarButton
          onClick={() =>
            editor.chain().focus().toggleHeading({ level: 2 }).run()
          }
          active={editor.isActive("heading", { level: 2 })}
          title="Heading 2"
        >
          H2
        </ToolbarButton>
        <ToolbarButton
          onClick={() =>
            editor.chain().focus().toggleHeading({ level: 3 }).run()
          }
          active={editor.isActive("heading", { level: 3 })}
          title="Heading 3"
        >
          H3
        </ToolbarButton>

        <Divider />

        <ToolbarButton
          onClick={() => editor.chain().focus().toggleBulletList().run()}
          active={editor.isActive("bulletList")}
          title="Bullet list"
        >
          <svg
            className="w-4 h-4"
            fill="none"
            stroke="currentColor"
            viewBox="0 0 24 24"
          >
            <path
              strokeLinecap="round"
              strokeLinejoin="round"
              strokeWidth={2}
              d="M4 6h16M4 12h16M4 18h16"
            />
          </svg>
        </ToolbarButton>

        <ToolbarButton
          onClick={() => editor.chain().focus().toggleOrderedList().run()}
          active={editor.isActive("orderedList")}
          title="Numbered list"
        >
          <svg
            className="w-4 h-4"
            fill="none"
            stroke="currentColor"
            viewBox="0 0 24 24"
          >
            <path
              strokeLinecap="round"
              strokeLinejoin="round"
              strokeWidth={2}
              d="M7 20h14M7 12h14M7 4h14M3 20h.01M3 12h.01M3 4h.01"
            />
          </svg>
        </ToolbarButton>

        <ToolbarButton
          onClick={() => editor.chain().focus().toggleBlockquote().run()}
          active={editor.isActive("blockquote")}
          title="Quote"
        >
          <svg
            className="w-4 h-4"
            fill="none"
            stroke="currentColor"
            viewBox="0 0 24 24"
          >
            <path
              strokeLinecap="round"
              strokeLinejoin="round"
              strokeWidth={2}
              d="M8 10h.01M12 10h.01M16 10h.01M9 16H5a2 2 0 01-2-2V6a2 2 0 012-2h14a2 2 0 012 2v8a2 2 0 01-2 2h-5l-5 5v-5z"

import { createContext, useContext, useState, useEffect } from "react";
import api from "../utils/api";

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [admin, setAdmin] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const stored = localStorage.getItem("hbc_admin");
    const token = localStorage.getItem("hbc_token");
    if (stored && token) {
      setAdmin(JSON.parse(stored));
    }
    setLoading(false);
  }, []);

  const login = async (email, password) => {
    const { data } = await api.post("/auth/login", { email, password });
    localStorage.setItem("hbc_token", data.token);
    localStorage.setItem("hbc_admin", JSON.stringify(data.admin));
    setAdmin(data.admin);
    return data;
  };

  const logout = async () => {
    try {
      await api.post("/auth/logout");
    } catch (e) {
      // ignore
    }
    localStorage.removeItem("hbc_token");
    localStorage.removeItem("hbc_admin");
    setAdmin(null);
  };

  const isAuthenticated = !!admin;

  return (
    <AuthContext.Provider
      value={{ admin, loading, login, logout, isAuthenticated }}
    >
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error("useAuth must be inside AuthProvider");
  return ctx;
}

import { useState, useEffect, useMemo } from "react";
import Layout from "../components/Layout";
import BookFormModal from "../components/BookFormModal";
import api from "../utils/api";

const CATEGORIES = [
  "General",
  "Academic",
  "Literature",
  "Religion",
  "Children",
  "Competitive",
  "Reference",
];

export default function Books() {
  const [books, setBooks] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [search, setSearch] = useState("");
  const [categoryFilter, setCategoryFilter] = useState("All");
  const [showModal, setShowModal] = useState(false);
  const [editingBook, setEditingBook] = useState(null);
  const [deletingId, setDeletingId] = useState(null);

  // ============ FETCH ============
  const fetchBooks = async () => {
    setLoading(true);
    setError("");
    try {
      const { data } = await api.get("/books");
      setBooks(data);
    } catch (err) {
      setError(
        err.response?.data?.message || "Failed to load books. Try again."
      );
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchBooks();
  }, []);

  // ============ FILTER ============
  const filteredBooks = useMemo(() => {
    return books.filter((b) => {
      const matchesSearch =
        !search ||
        b.title?.toLowerCase().includes(search.toLowerCase()) ||
        b.author?.toLowerCase().includes(search.toLowerCase()) ||
        b.isbn?.toLowerCase().includes(search.toLowerCase());

      const matchesCategory =
        categoryFilter === "All" || b.category === categoryFilter;

      return matchesSearch && matchesCategory;
    });
  }, [books, search, categoryFilter]);

  // ============ CREATE / UPDATE ============
  const handleSave = async (formData) => {
    if (editingBook) {
      const { data } = await api.put(`/books/${editingBook._id}`, formData);
      // Replace the edited book with the server response
      setBooks((prev) =>
        prev.map((b) => (b._id === editingBook._id ? data : b))
      );
      return data;
    } else {
      const { data } = await api.post("/books", formData);
      setBooks((prev) => [data, ...prev]);
      return data;
    }
  };

  // ============ DELETE ============
  const handleDelete = async (id) => {
    if (!confirm("Are you sure you want to delete this book?")) return;
    setDeletingId(id);
    try {
      await api.delete(`/books/${id}`);
      setBooks((prev) => prev.filter((b) => b._id !== id));
    } catch (err) {
      alert(err.response?.data?.message || "Failed to delete book");
    } finally {
      setDeletingId(null);
    }
  };

  const openAddModal = () => {
    setEditingBook(null);
    setShowModal(true);
  };

  const openEditModal = (book) => {
    setEditingBook(book);
    setShowModal(true);
  };

  const closeModal = () => {
    setShowModal(false);
    setEditingBook(null);
  };

  return (
    <Layout
      title="Books Management"
      subtitle="Manage your book inventory — add, edit, or remove books."
    >
      {/* ============ TOOLBAR ============ */}
      <div className="bg-white rounded-2xl shadow-md border-l-4 border-l-red-600 p-4 mb-6">
        <div className="flex flex-col md:flex-row md:items-center gap-3">
          {/* Search */}
          <div className="flex-1 relative">
            <svg
              className="w-5 h-5 text-gray-400 absolute left-3 top-1/2 -translate-y-1/2"
              fill="none"
              stroke="currentColor"
              viewBox="0 0 24 24"

                strokeLinejoin="round"
                strokeWidth={2}
                d="M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0z"
              />
            </svg>
            <input
              type="text"
              placeholder="Search by title, author, or ISBN..."

              onChange={(e) => setSearch(e.target.value)}
              className="w-full pl-10 pr-4 py-2.5 rounded-xl border-2 border-yellow-300 bg-yellow-50/50 text-gray-800 placeholder-amber-700/50 focus:outline-none focus:border-red-500 focus:ring-2 focus:ring-red-200 focus:bg-white transition"
            />
          </div>

          {/* Category filter */}
          <select
            value={categoryFilter}
            onChange={(e) => setCategoryFilter(e.target.value)}
            className="px-4 py-2.5 rounded-xl border-2 border-yellow-300 bg-yellow-50/50 text-gray-800 font-medium focus:outline-none focus:border-green-600 focus:ring-2 focus:ring-green-200 focus:bg-white transition"
          >

function KPI({ label, value, icon, accent, sub }) {
  const colors = {
    red: { bg: "bg-red-50", icon: "text-red-600", val: "text-red-700" },
    green: { bg: "bg-green-50", icon: "text-green-600", val: "text-green-700" },
    yellow: {
      bg: "bg-yellow-50",
      icon: "text-yellow-600",
      val: "text-yellow-700",
    },
  }[accent];

  return (
    <div className="bg-white rounded-2xl shadow-sm border border-gray-100 p-5 hover:shadow-md transition">
      <div
        className={`w-11 h-11 rounded-xl ${colors.bg} ${colors.icon} flex items-center justify-center mb-3`}
      >
        <svg
          className="w-5 h-5"
          fill="none"
          stroke="currentColor"
          viewBox="0 0 24 24"
        >
          <path
            strokeLinecap="round"
            strokeLinejoin="round"
            strokeWidth={2}
            d={icon}
          />
        </svg>
      </div>
      <p className="text-xs text-gray-500 font-semibold uppercase tracking-wide">
        {label}
      </p>
      <p className={`text-2xl font-extrabold mt-1 ${colors.val}`}>{value}</p>
      {sub && <p className="text-xs text-gray-400 mt-0.5">{sub}</p>}
    </div>
  );
}

// frontend/src/pages/Orders.jsx
import { useEffect, useState } from "react";
import Layout from "../components/Layout";
import api from "../utils/api";

const STATUS_OPTIONS = [
  { value: "pending", label: "Pending" },
  { value: "confirmed", label: "Confirmed" },
  { value: "shipped", label: "Shipped" },
  { value: "delivered", label: "Delivered" },
  { value: "cancelled", label: "Cancelled" },
];

const STATUS_STYLES = {
  pending: "bg-yellow-100 text-yellow-800 border-yellow-200",
  confirmed: "bg-blue-100 text-blue-800 border-blue-200",
  shipped: "bg-purple-100 text-purple-800 border-purple-200",
  delivered: "bg-green-100 text-green-800 border-green-200",
  cancelled: "bg-red-100 text-red-800 border-red-200",
};

const PAYMENT_STYLES = {
  unpaid: "bg-gray-100 text-gray-700 border-gray-200",
  "pending-verification": "bg-yellow-100 text-yellow-800 border-yellow-200",
  verified: "bg-green-100 text-green-800 border-green-200",
  rejected: "bg-red-100 text-red-800 border-red-200",
};

export default function Orders() {
  const [orders, setOrders] = useState([]);
  const [loading, setLoading] = useState(true);
  const [search, setSearch] = useState("");
  const [statusFilter, setStatusFilter] = useState("all");
  const [paymentFilter, setPaymentFilter] = useState("all");
  const [expanded, setExpanded] = useState(null);
  const [updating, setUpdating] = useState(null);
  const [toast, setToast] = useState("");

  const showToast = (msg) => {
    setToast(msg);
    setTimeout(() => setToast(""), 2500);
  };

  const fetchOrders = async () => {
    setLoading(true);
    try {
      const { data } = await api.get("/admin/orders");
      setOrders(data);
    } catch (err) {
      console.error(err);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchOrders();
  }, []);

  const filtered = orders.filter((o) => {
    if (statusFilter !== "all" && o.status !== statusFilter) return false;
    if (paymentFilter !== "all" && o.paymentStatus !== paymentFilter)
      return false;
    if (search.trim()) {
      const q = search.toLowerCase();
      return (
        o.orderNumber?.toLowerCase().includes(q) ||
        o.shippingAddress?.name?.toLowerCase().includes(q) ||
        o.shippingAddress?.phone?.toLowerCase().includes(q) ||
        o.user?.email?.toLowerCase().includes(q)
      );
    }
    return true;
  });

  const updateOrder = async (id, patch) => {
    setUpdating(id);
    try {
      const { data } = await api.put(`/admin/orders/${id}`, patch);
      setOrders((prev) => prev.map((o) => (o._id === id ? { ...o, ...data } : o)));
      showToast("Order updated");
    } catch {
      showToast("Failed to update");
    } finally {
      setUpdating(null);
    }
  };

  return (
    <Layout
      title="Orders"
      subtitle="Manage customer orders and verify payments."
    >
      {toast && (
        <div className="fixed top-20 left-1/2 -translate-x-1/2 z-50 bg-gray-900 text-white text-sm font-semibold px-5 py-2.5 rounded-full shadow-xl">
          {toast}
        </div>
      )}

      {/* Toolbar */}
      <div className="bg-white rounded-2xl shadow-sm border-l-4 border-l-red-600 p-4 mb-6">
        <div className="flex flex-col md:flex-row md:items-center gap-3">
          <div className="flex-1 relative">
            <svg
              className="w-5 h-5 text-gray-400 absolute left-3 top-1/2 -translate-y-1/2"
              fill="none"
              stroke="currentColor"
              viewBox="0 0 24 24"
            >
              <path
                strokeLinecap="round"
                strokeLinejoin="round"
                strokeWidth={2}
                d="M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0z"
              />
            </svg>
            <input
              type="text"
              placeholder="Search by order #, name, phone, email..."
              value={search}
              onChange={(e) => setSearch(e.target.value)}
              className="w-full pl-10 pr-4 py-2.5 rounded-xl border-2 border-yellow-300 bg-yellow-50/50 text-sm focus:outline-none focus:border-red-500 focus:ring-2 focus:ring-red-200 focus:bg-white transition"
            />
          </div>

          <select
            value={statusFilter}
            onChange={(e) => setStatusFilter(e.target.value)}
            className="px-4 py-2.5 rounded-xl border-2 border-yellow-300 bg-yellow-50/50 text-sm font-medium focus:outline-none focus:border-green-600 focus:ring-2 focus:ring-green-200 focus:bg-white transition"
          >
            <option value="all">All Status</option>
            {STATUS_OPTIONS.map((s) => (
              <option key={s.value} value={s.value}>
                {s.label}
              </option>
            ))}
          </select>

          <select
            value={paymentFilter}
            onChange={(e) => setPaymentFilter(e.target.value)}
            className="px-4 py-2.5 rounded-xl border-2 border-yellow-300 bg-yellow-50/50 text-sm font-medium focus:outline-none focus:border-green-600 focus:ring-2 focus:ring-green-200 focus:bg-white transition"
          >
            <option value="all">All Payments</option>
            <option value="unpaid">Unpaid</option>
            <option value="pending-verification">Pending Verification</option>
            <option value="verified">Verified</option>
            <option value="rejected">Rejected</option>
          </select>
        </div>

        <p className="mt-3 text-xs text-gray-500 flex items-center gap-2">
          <span className="w-2 h-2 rounded-full bg-red-500" />
          {filtered.length} of {orders.length} orders
        </p>
      </div>

      {/* Orders */}
      {loading ? (
        <div className="bg-white rounded-2xl shadow-md p-16 flex flex-col items-center">
          <div className="w-12 h-12 border-4 border-red-200 border-t-red-600 rounded-full animate-spin mb-4" />
          <p className="text-sm text-gray-500">Loading orders...</p>
        </div>
      ) : filtered.length === 0 ? (
        <div className="bg-white rounded-2xl shadow-md p-16 text-center">
          <p className="text-lg font-bold text-gray-800">No orders found</p>
          <p className="text-sm text-gray-500 mt-1">
            Try adjusting your filters.
          </p>
        </div>
      ) : (
        <div className="space-y-4">
          {filtered.map((order) => (
            <OrderRow
              key={order._id}
              order={order}
              expanded={expanded === order._id}
              onToggle={() =>
                setExpanded(expanded === order._id ? null : order._id)
              }
              onUpdate={(patch) => updateOrder(order._id, patch)}
              updating={updating === order._id}
            />
          ))}
        </div>
      )}
    </Layout>
  );
}

            {order.status}
          </span>

              PAYMENT_STYLES[order.paymentStatus] || PAYMENT_STYLES.unpaid

import { useAuth } from "../context/AuthContext";

export default function Settings() {
  const { admin } = useAuth();
  const [settings, setSettings] = useState(null);
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [toast, setToast] = useState("");
  const [uploadingQR, setUploadingQR] = useState(false);
  const fileInputRef = useRef(null);

  const showToast = (msg) => {
    setToast(msg);
    setTimeout(() => setToast(""), 2500);
  };

  const load = async () => {
    setLoading(true);
    try {
      const { data } = await api.get("/admin/settings");
      setSettings(data);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    load();
  }, []);

  const save = async (patch) => {
    setSaving(true);
    try {
      const { data } = await api.put("/admin/settings", patch);
      setSettings(data);
      showToast("Settings saved");
    } catch (err) {
      showToast(err.response?.data?.message || "Failed to save");
    } finally {
      setSaving(false);
    }
  };

  // ---------- QR upload ----------
  const handleQRUpload = async (file) => {
    if (!file) return;
    if (!file.type.startsWith("image/")) {
      showToast("Please select an image file");
      return;
    }
    if (file.size > 5 * 1024 * 1024) {
      showToast("Image must be under 5 MB");
      return;
    }

    setUploadingQR(true);
    try {
      const { data: sign } = await api.post("/upload/qr-sign");

      const fd = new FormData();
      fd.append("file", file);
      fd.append("api_key", sign.apiKey);
      fd.append("timestamp", sign.timestamp);
      fd.append("signature", sign.signature);
      fd.append("folder", sign.folder);

      const resp = await fetch(sign.uploadUrl, { method: "POST", body: fd });
      if (!resp.ok) throw new Error("Cloudinary upload failed");
      const result = await resp.json();

      await save({
        paymentQR: {
          url: result.secure_url,
          publicId: result.public_id,
        },
      });
    } catch (err) {
      console.error(err);
      showToast("Failed to upload QR");
    } finally {
      setUploadingQR(false);
      if (fileInputRef.current) fileInputRef.current.value = "";
    }
  };

  const removeQR = async () => {
    if (!confirm("Remove the QR code? Customers won't be able to pay by QR.")) return;
    await save({ paymentQR: null });
  };

  if (loading || !settings) {
    return (
      <Layout title="Settings" subtitle="Loading...">
        <div className="bg-white rounded-2xl shadow-md p-16 flex justify-center">
          <div className="w-12 h-12 border-4 border-red-200 border-t-red-600 rounded-full animate-spin" />
        </div>
      </Layout>

          {toast}
        </div>
      )}

      <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
        {/* ---------- DELIVERY ZONES ---------- */}
        <div className="lg:col-span-2 bg-white rounded-2xl shadow-sm border border-gray-100 p-6">
          <h3 className="font-bold text-gray-900 mb-1 flex items-center gap-2">
            <span className="w-1.5 h-5 bg-green-600 rounded-full" />
            Delivery Zones & Charges
          </h3>
          <p className="text-xs text-gray-500 mb-5">
            Define delivery zones with their charges. Customers pick a zone at
            checkout and the fee is applied automatically.
          </p>

          <ZoneManager
            zones={settings.deliveryZones || []}
            onChange={(zones) => save({ deliveryZones: zones })}
            saving={saving}
          />
        </div>

        {/* ---------- PAYMENT METHODS ---------- */}
        <div className="bg-white rounded-2xl shadow-sm border border-gray-100 p-6">
          <h3 className="font-bold text-gray-900 mb-1 flex items-center gap-2">
            <span className="w-1.5 h-5 bg-red-600 rounded-full" />
            Payment Methods
          </h3>
          <p className="text-xs text-gray-500 mb-4">
            Enable or disable payment options for customers.
          </p>

          <div className="space-y-3">
            <ToggleRow
              label="QR Code Payment"
              desc="Customer scans QR and uploads proof."
              enabled={settings.paymentMethods?.qr}
              onToggle={(v) =>
                save({
                  paymentMethods: { ...settings.paymentMethods, qr: v },
                })
              }
              disabled={saving}
            />
            <ToggleRow
              label="Cash on Delivery"
              desc="Customer pays in cash when the order arrives."
              enabled={settings.paymentMethods?.cod}
              onToggle={(v) =>
                save({
                  paymentMethods: { ...settings.paymentMethods, cod: v },
                })
              }
              disabled={saving}
            />
          </div>
        </div>

        {/* ---------- ADMIN INFO ---------- */}
        <div className="bg-white rounded-2xl shadow-sm border border-gray-100 p-6">
          <h3 className="font-bold text-gray-900 mb-4 flex items-center gap-2">

function ZoneManager({ zones, onChange, saving }) {
  const [draft, setDraft] = useState(zones);
  const [editingIndex, setEditingIndex] = useState(null);

  useEffect(() => {
    setDraft(zones);
  }, [zones]);

  const update = (i, patch) => {
    setDraft((prev) =>
      prev.map((z, idx) => (idx === i ? { ...z, ...patch } : z))
    );
  };

  const addZone = () => {
    setDraft((prev) => [
      ...prev,
      {
        id: `zone-${Date.now()}`,
        name: "",
        description: "",
        charge: 0,
      },
    ]);
    setEditingIndex(draft.length);
  };

  const removeZone = (i) => {
    if (!confirm("Remove this zone?")) return;
    const next = draft.filter((_, idx) => idx !== i);
    setDraft(next);
    onChange(next);
    setEditingIndex(null);
  };

  const saveZones = () => {
    for (const z of draft) {
      if (!z.name?.trim()) {
        alert("Every zone must have a name.");
        return;
      }
      if (z.charge < 0 || Number.isNaN(Number(z.charge))) {
        alert("Charge must be zero or a positive number.");
        return;
      }
    }
    onChange(
      draft.map((z) => ({
        id: z.id,
        name: z.name.trim(),
        description: (z.description || "").trim(),
        charge: Number(z.charge) || 0,
      }))
    );
    setEditingIndex(null);
  };

  return (
    <div className="space-y-3">
      {draft.length === 0 ? (
        <div className="text-center py-8 bg-gray-50 rounded-xl border-2 border-dashed border-gray-200">

                        className="w-4 h-4"
                        fill="none"
                        stroke="currentColor"

                      >

                          strokeLinecap="round"
                          strokeLinejoin="round"

                        />

                        </label>

                          type="text"
                          value={zone.name}
                          onChange={(e) => update(i, { name: e.target.value })}
                          placeholder="e.g. Inside Halesi"
                          className="w-full px-3 py-2 rounded-lg border-2 border-yellow-300 bg-yellow-50/50 text-sm focus:outline-none focus:border-red-500 focus:ring-2 focus:ring-red-200 focus:bg-white transition"
                        />
                      </div>
                      <div>
                        <label className="block text-xs font-semibold text-gray-700 mb-1">
                          Charge (NPR) *
                        </label>
                        <input
                          type="number"
                          min={0}
                          value={zone.charge}
                          onChange={(e) =>
                            update(i, { charge: e.target.value })
                          }
                          className="w-full px-3 py-2 rounded-lg border-2 border-yellow-300 bg-yellow-50/50 text-sm focus:outline-none focus:border-green-600 focus:ring-2 focus:ring-green-200 focus:bg-white transition"
                        />
                      </div>
                    </div>
                    <div>
                      <label className="block text-xs font-semibold text-gray-700 mb-1">
                        Description
                      </label>
                      <input
                        type="text"
                        value={zone.description || ""}
                        onChange={(e) =>
                          update(i, { description: e.target.value })
                        }
                        placeholder="e.g. Free delivery within Halesi town"
                        className="w-full px-3 py-2 rounded-lg border-2 border-yellow-300 bg-yellow-50/50 text-sm focus:outline-none focus:border-red-500 focus:ring-2 focus:ring-red-200 focus:bg-white transition"
                      />
                    </div>
                    <div className="flex gap-2">
                      <button
                        type="button"
                        onClick={() => setEditingIndex(null)}
                        className="flex-1 py-2 rounded-lg border-2 border-gray-200 text-gray-700 font-semibold text-xs hover:bg-gray-50 transition"
                      >
                        Done
                      </button>
                      <button
                        type="button"
                        onClick={() => removeZone(i)}
                        className="py-2 px-4 rounded-lg bg-red-50 text-red-700 font-semibold text-xs hover:bg-red-100 transition"
                      >
                        Delete
                      </button>
                    </div>
                  </div>
                )}
              </div>
            );
          })}
        </div>
      )}

      {/* Actions */}
      <div className="flex gap-2 pt-2">
        <button
          type="button"
          onClick={addZone}
          className="flex-1 flex items-center justify-center gap-2 py-2.5 rounded-xl border-2 border-dashed border-green-300 text-green-700 font-semibold text-sm hover:bg-green-50 transition"
        >
          <svg
            className="w-4 h-4"
            fill="none"
            stroke="currentColor"
            viewBox="0 0 24 24"
          >
            <path
              strokeLinecap="round"
              strokeLinejoin="round"
              strokeWidth={2}
              d="M12 4v16m8-8H4"
            />
          </svg>
          Add Zone
        </button>
        <button
          type="button"
          onClick={saveZones}
          disabled={saving}
          className="flex-1 bg-gradient-to-r from-green-700 to-green-600 hover:from-green-800 hover:to-green-700 text-white font-semibold py-2.5 rounded-xl shadow-md transition disabled:opacity-50 disabled:cursor-not-allowed"
        >
          {saving ? "Saving..." : "Save Zones"}
        </button>
      </div>
    </div>
  );
}

import { useEffect, useState } from "react";
import Layout from "../components/Layout";
import api from "../utils/api";

export default function Users() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [search, setSearch] = useState("");
  const [expanded, setExpanded] = useState(null);
  const [busy, setBusy] = useState(null);

  const fetch = async () => {
    setLoading(true);
    try {
      const { data } = await api.get("/admin/users");
      setUsers(data);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetch();
  }, []);

  const filtered = users.filter((u) => {
    if (!search.trim()) return true;
    const q = search.toLowerCase();
    return (
      u.name?.toLowerCase().includes(q) ||
      u.email?.toLowerCase().includes(q) ||
      u.phone?.toLowerCase().includes(q)
    );
  });

  const toggleBlock = async (id, blocked) => {
    setBusy(id);
    try {
      await api.put(`/admin/users/${id}`, { blocked: !blocked });
      setUsers((prev) =>
        prev.map((u) => (u._id === id ? { ...u, blocked: !blocked } : u))
      );
    } finally {
      setBusy(null);
    }
  };

  const remove = async (id) => {
    if (!confirm("Delete this user permanently?")) return;
    setBusy(id);
    try {
      await api.delete(`/admin/users/${id}`);
      setUsers((prev) => prev.filter((u) => u._id !== id));
    } finally {
      setBusy(null);
    }
  };

  const initials = (name) =>
    (name || "U")
      .split(" ")
      .filter(Boolean)
      .map((s) => s[0])
      .slice(0, 2)
      .join("")
      .toUpperCase();

  return (
    <Layout title="Users" subtitle="Manage customer accounts.">
      {/* Toolbar */}
      <div className="bg-white rounded-2xl shadow-sm border-l-4 border-l-red-600 p-4 mb-6">
        <div className="relative">
          <svg
            className="w-5 h-5 text-gray-400 absolute left-3 top-1/2 -translate-y-1/2"
            fill="none"
            stroke="currentColor"
            viewBox="0 0 24 24"
          >
            <path
              strokeLinecap="round"
              strokeLinejoin="round"
              strokeWidth={2}
              d="M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0z"
            />
          </svg>
          <input
            type="text"
            placeholder="Search by name, email, phone..."
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            className="w-full pl-10 pr-4 py-2.5 rounded-xl border-2 border-yellow-300 bg-yellow-50/50 text-sm focus:outline-none focus:border-red-500 focus:ring-2 focus:ring-red-200 focus:bg-white transition"
          />
        </div>
        <p className="mt-3 text-xs text-gray-500">
          {filtered.length} of {users.length} users
        </p>
      </div>

      {/* List */}
      {loading ? (
        <div className="bg-white rounded-2xl shadow-md p-16 flex justify-center">
          <div className="w-12 h-12 border-4 border-red-200 border-t-red-600 rounded-full animate-spin" />
        </div>
      ) : filtered.length === 0 ? (
        <div className="bg-white rounded-2xl shadow-md p-16 text-center">
          <p className="text-lg font-bold text-gray-800">No users found</p>
        </div>
      ) : (
        <div className="bg-white rounded-2xl shadow-md overflow-hidden">
          {filtered.map((u) => (
            <div
              key={u._id}
              className="border-b border-gray-100 last:border-0"
            >
              <div className="p-4 flex items-center gap-4">
                {u.avatar?.url ? (
                  <img
                    src={u.avatar.url}
                    alt={u.name}
                    className="w-12 h-12 rounded-full object-cover border-2 border-red-500 flex-shrink-0"
                  />
                ) : (
                  <div className="w-12 h-12 rounded-full bg-gradient-to-br from-red-600 to-red-800 text-white flex items-center justify-center font-bold flex-shrink-0">
                    {initials(u.name)}
                  </div>
                )}

                <div className="flex-1 min-w-0">
                  <div className="flex items-center gap-2 flex-wrap">
                    <p className="font-bold text-gray-900 text-sm truncate">
                      {u.name}
                    </p>
                    {u.blocked && (
                      <span className="text-[10px] font-bold uppercase bg-red-100 text-red-700 px-2 py-0.5 rounded-full">
                        Blocked
                      </span>
                    )}
                  </div>
                  <p className="text-xs text-gray-500 truncate">{u.email}</p>
                  {u.phone && (
                    <p className="text-xs text-gray-400">{u.phone}</p>
                  )}
                </div>

                <div className="hidden sm:flex items-center gap-6 text-right">
                  <div>
                    <p className="text-xs text-gray-500">Orders</p>
                    <p className="text-sm font-bold text-gray-900">
                      {u.orderCount}
                    </p>
                  </div>
                  <div>
                    <p className="text-xs text-gray-500">Spent</p>
                    <p className="text-sm font-bold text-green-700">
                      NPR {Number(u.totalSpent).toLocaleString()}
                    </p>
                  </div>
                </div>

                <button
                  onClick={() =>
                    setExpanded(expanded === u._id ? null : u._id)
                  }
                  className="w-9 h-9 rounded-lg hover:bg-gray-100 flex items-center justify-center transition"
                >
                  <svg
                    className={`w-4 h-4 text-gray-500 transition ${
                      expanded === u._id ? "rotate-180" : ""
                    }`}
                    fill="none"
                    stroke="currentColor"
                    viewBox="0 0 24 24"
                  >
                    <path
                      strokeLinecap="round"
                      strokeLinejoin="round"
                      strokeWidth={2}
                      d="M19 9l-7 7-7-7"
                    />
                  </svg>
                </button>
              </div>

              {expanded === u._id && (
                <div className="bg-gray-50 border-t border-gray-100 p-4">
                  <div className="grid grid-cols-2 md:grid-cols-4 gap-3 mb-4">
                    <Stat label="Wishlist" value={u.wishlistCount} />
                    <Stat label="Addresses" value={u.addressesCount} />
                    <Stat label="Orders" value={u.orderCount} />
                    <Stat
                      label="Total Spent"
                      value={`NPR ${Number(u.totalSpent).toLocaleString()}`}
                    />
                  </div>

                  <div className="flex gap-2">
                    <button
                      onClick={() => toggleBlock(u._id, u.blocked)}
                      disabled={busy === u._id}
                      className={`px-4 py-2 rounded-lg font-semibold text-sm transition disabled:opacity-60 ${
                        u.blocked
                          ? "bg-green-100 text-green-800 hover:bg-green-200"
                          : "bg-yellow-100 text-yellow-800 hover:bg-yellow-200"
                      }`}
                    >
                      {u.blocked ? "Unblock User" : "Block User"}
                    </button>
                    <button
                      onClick={() => remove(u._id)}
                      disabled={busy === u._id}
                      className="px-4 py-2 rounded-lg bg-red-100 text-red-700 hover:bg-red-200 font-semibold text-sm transition disabled:opacity-60"
                    >
                      Delete
                    </button>
                  </div>
                </div>
              )}
            </div>
          ))}
        </div>
      )}
    </Layout>
  );
}

        {label}
      </p>
      <p className="text-base font-extrabold text-gray-900 mt-0.5">{value}</p>
    </div>
  );
}

import axios from "axios";

const api = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  withCredentials: true,
  headers: {
    "Content-Type": "application/json",
  },
});

// Attach token from localStorage to every request
api.interceptors.request.use((config) => {
  const token = localStorage.getItem("hbc_token");
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// Handle 401 globally
api.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response?.status === 401) {
      localStorage.removeItem("hbc_token");
      localStorage.removeItem("hbc_admin");
      if (window.location.pathname !== "/login") {
        window.location.href = "/login";
      }
    }
    return Promise.reject(error);
  }
);

export default api;
